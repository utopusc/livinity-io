```typescript
/**
 * X11 wheel button conventions:
 *   button 4 = scroll up
 *   button 5 = scroll down
 *   button 6 = scroll left
 *   button 7 = scroll right
 */
export type ScrollButton = 4 | 5 | 6 | 7

/**
 * Dispatch a scroll wheel event at (x, y) inside the given X11 wid.
 *
 * Same activate-first pattern as `dispatchPointer` (Phase 100-07.3 fix).
 * Chrome filters synthetic XSendEvent — `xdotool click --window <wid> 5`
 * dispatches the event but Chrome drops it. The reliable pattern is to
 * activate the wid first, move the cursor wid-relative, then click
 * (without --window) so xdotool sends a real button-press event to the
 * X11-focused window.
 *
 * Phase 100-09-02 closes the scroll-down bug from 100-08 deploy live test.
 * Frontend wheel handler maps deltaY > 0 → button 5 (down), < 0 → button 4
 * (up), deltaX > 0 → button 7 (right), < 0 → button 6 (left). Single click
 * = single notch; the browser's deltaY magnitude is NOT preserved (one
 * event per wheel notch is the X11 convention).
 */
export async function dispatchScroll(
    wid: number,
    x: number,
    y: number,
    button: ScrollButton,
): Promise<void> {
    if (!Number.isInteger(wid) || wid <= 0) throw new Error(`invalid wid: ${wid}`)
    if (!Number.isFinite(x) || !Number.isFinite(y)) throw new Error('invalid coords')
    if (button !== 4 && button !== 5 && button !== 6 && button !== 7) {
        throw new Error(`invalid scroll button: ${button} (must be 4/5/6/7)`)
    }
    const ix = Math.max(0, Math.round(x))
    const iy = Math.max(0, Math.round(y))
    const widStr = String(wid)
    await execFileAsync(
        'xdotool',
        [
            'windowactivate', '--sync', widStr,
            'windowfocus', '--sync', widStr,
            'mousemove', '--window', widStr, '--sync', String(ix), String(iy),
            'click', '--clearmodifiers', String(button),
        ],
        {timeout: DEFAULT_TIMEOUT_MS},
    )
}
```

```typescript
const onWheel = (ev: WheelEvent) => {
    // Suppress browser native scroll — events route through tRPC to the
    // captured Chrome via xdotool. preventDefault REQUIRES passive: false
    // on the addEventListener call below.
    ev.preventDefault()
    const {x, y} = eventToFbCoords(ev)
    // X11 scroll button convention:
    //   4 = scroll up   (deltaY < 0)
    //   5 = scroll down (deltaY > 0)
    //   6 = scroll left (deltaX < 0)
    //   7 = scroll right(deltaX > 0)
    // Browsers may emit both deltaY and deltaX simultaneously (touchpads).
    // Dispatch each axis as its own event; xdotool buttons aren't combinable.
    if (ev.deltaY !== 0) {
        const button = ev.deltaY > 0 ? 5 : 4
        // eslint-disable-next-line no-console
        console.info(`[100-09-02] wheel-Y → tRPC webappId=${webappId} btn=${button}`)
        inputScrollMutation.mutate({webappId, x, y, button: button as 4 | 5})
    }
    if (ev.deltaX !== 0) {
        const button = ev.deltaX > 0 ? 7 : 6
        inputScrollMutation.mutate({webappId, x, y, button: button as 6 | 7})
    }
}
```

```typescript
/**
 * X11 wheel button mapping (private constant — same as input-dispatcher.ts).
 */
const SCROLL_BUTTON: Record<ScrollDirection, '4' | '5' | '6' | '7'> = {
    up: '4',
    down: '5',
    left: '6',
    right: '7',
}

/**
 * Phase 100-09-02 — xdotool scroll via activate-first chain. Same Chrome
 * synthetic-event filter that broke clicks (P100-07.3) breaks scroll too.
 * Returns true on clean exit; false otherwise (caller falls back to nut-js).
 * Never throws.
 */
async function tryXdotoolScroll(
    direction: ScrollDirection,
    count: number,
    coordinates: Coords,
    windowId: number,
): Promise<boolean> {
    const widStr = String(windowId)
    const btn = SCROLL_BUTTON[direction]
    const args: string[] = [
        'windowactivate', '--sync', widStr,
        'windowfocus', '--sync', widStr,
        'mousemove', '--window', widStr, '--sync', String(coordinates.x), String(coordinates.y),
    ]
    // xdotool's `click --repeat N --delay <ms> <btn>` for N>1.
    if (count > 1) {
        args.push('click', '--clearmodifiers', '--repeat', String(count), '--delay', '50', btn)
    } else {
        args.push('click', '--clearmodifiers', btn)
    }
    return await spawnXdotool(args)
}

/**
 * Scroll the wheel in a direction at coordinates. Mirrors `_scrollTool`
 * (computer_scroll).
 *
 * 2026-05-10 P100-09-02: when `windowId` is set, dispatch via the
 * activate-first xdotool click 4/5/6/7 chain (Chrome filters nut-js's
 * synthetic XTestFakeButtonEvent the same way it filters clicks; same
 * fix as P100-07.3 `tryXdotoolClick`). When unset, the original nut-js
 * path runs (host-display, backward compat with native bytebot).
 */
export async function scroll(opts: {
    coordinates: Coords
    direction: ScrollDirection
    scrollCount: number
    holdKeys?: readonly string[]
    /** P100-09-02: target a specific X11 window via xdotool activate-first chain. */
    windowId?: number
}): Promise<void> {
    const n = Math.max(1, Math.floor(opts.scrollCount))
    if (typeof opts.windowId === 'number' && (!opts.holdKeys || opts.holdKeys.length === 0)) {
        const ok = await tryXdotoolScroll(opts.direction, n, opts.coordinates, opts.windowId)
        if (ok) return
        // Fall through to nut-js — xdotool unavailable. Host-display only.
    }
    await withHeldKeys(opts.holdKeys, async () => {
        await mouse.setPosition(new Point(opts.coordinates.x, opts.coordinates.y))
        switch (opts.direction) {
            case 'up':    await mouse.scrollUp(n); break
            case 'down':  await mouse.scrollDown(n); break
            case 'left':  await mouse.scrollLeft(n); break
            case 'right': await mouse.scrollRight(n); break
        }
    })
}
```

    ```typescript
     * X11 wheel button conventions:
     *   button 4 = scroll up
     *   button 5 = scroll down
     *   button 6 = scroll left

    export type ScrollButton = 4 | 5 | 6 | 7

    /**
     * Phase 100-09-02 — Dispatch a scroll wheel event at (x, y) inside the
     * given X11 wid. Same activate-first pattern as `dispatchPointer`.
     *
     * Chrome filters synthetic XSendEvent — `xdotool click --window <wid> 5`
     * dispatches the event but Chrome drops it (same fix as P100-07.3 for
     * regular click). The reliable pattern is: activate the wid, focus it,
     * move the cursor wid-relative, then click (without --window) so xdotool
     * sends a real button event to the X11-focused window.
     *
     * Closes 100-09 Bug 2 (scroll-down doesn't work) — frontend wheel
     * listener (webapp-stream-window.tsx) maps deltaY > 0 → button 5,
     * deltaY < 0 → button 4, deltaX > 0 → button 7, deltaX < 0 → button 6.
     */
    export async function dispatchScroll(
        wid: number,
        x: number,
        y: number,
        button: ScrollButton,
    ): Promise<void> {
        if (!Number.isInteger(wid) || wid <= 0) throw new Error(`invalid wid: ${wid}`)
        if (!Number.isFinite(x) || !Number.isFinite(y)) throw new Error('invalid coords')
        if (button !== 4 && button !== 5 && button !== 6 && button !== 7) {
            throw new Error(`invalid scroll button: ${button} (must be 4/5/6/7)`)

        const iy = Math.max(0, Math.round(y))
        const widStr = String(wid)
        await execFileAsync(
            'xdotool',
            [
                'windowactivate', '--sync', widStr,
                'windowfocus', '--sync', widStr,
                'mousemove', '--window', widStr, '--sync', String(ix), String(iy),
    ```
