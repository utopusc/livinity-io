```typescript
/**
 * Phase 100-08-05 — pure scope-filter helper.
 *
 * Decides which MCP servers to inject into `additionalMcpServers` based on
 * the request's optional `webappId`:
 *
 * - webappId present AND `bytebot:webapp:<webappId>` exists in servers:
 *     scope = 'webapp'; only that one server is included.
 * - webappId present AND no matching server (LIV-CORE RECONCILE LAG —
 *   see 100-08-04 SUMMARY): scope = 'lag-fallback'; host bytebot +
 *   user-installed MCPs included; per-WebApp instances excluded;
 *   logger.warn fires with webappScope + fallback flag.
 * - webappId absent: scope = 'host'; host bytebot + user-installed MCPs
 *   included; per-WebApp instances excluded.
 *
 * The function is pure (no I/O) so api.scope-filter.test.ts can unit-test
 * it without spinning up Express. Reserved-name skip ('nexus-tools',
 * 'chrome-devtools') stays inline at the call site (it's about logging,
 * not scope).
 */
export function filterAdditionalMcpServers(
    enabledServers: ReadonlyArray<{name: string; transport: string}>,
    reqWebappId: string | null,
    logger: {warn: (msg: string, meta?: unknown) => void},
): {scope: 'webapp' | 'host' | 'lag-fallback'; servers: typeof enabledServers} {
    if (reqWebappId) {
        const targetName = `bytebot:webapp:${reqWebappId}`;
        const matching = enabledServers.find(s => s.name === targetName);
        if (matching) {
            // scope = 'webapp': only the per-WebApp instance.
            return {scope: 'webapp', servers: [matching]};
        }
        // scope = 'lag-fallback': liv-core reconcile lag (or webappId
        // doesn't match any registered WebApp). Fall through to host scope
        // for this turn so the user's chat doesn't fail with empty tools.
        logger.warn(
            '/api/agent/stream: webappId scope-filter lag fallback — no matching `bytebot:webapp:<id>` server registered yet',
            {webappScope: reqWebappId, fallback: 'lag'},
        );
        return {
            scope: 'lag-fallback',
            servers: enabledServers.filter(s => !s.name.startsWith('bytebot:webapp:')),
        };
    }
    // scope = 'host': webappId absent (v32 chat home).
    return {
        scope: 'host',
        servers: enabledServers.filter(s => !s.name.startsWith('bytebot:webapp:')),
    };
}
```

// Phase 100-08-05 — extract optional webappId from request body for scope filtering.
const reqWebappId: string | null =
    typeof req.body?.webappId === 'string' && req.body.webappId.length > 0
        ? req.body.webappId
        : null;

let additionalMcpServers: Record<string, unknown> | undefined = undefined;
if (mcpConfigManager) {
    try {
        const servers = await mcpConfigManager.listServers();
        const enabled = servers.filter(s => s.enabled);
        if (enabled.length > 0) {
            // Phase 100-08-05 — scope filter (D-100-08-B):
            // see filterAdditionalMcpServers for the lag-fallback contract.
            const {scope, servers: scoped} = filterAdditionalMcpServers(
                enabled,
                reqWebappId,
                logger,
            );

            additionalMcpServers = {};
            for (const s of scoped) {
                if (s.name === 'nexus-tools' || s.name === 'chrome-devtools') {
                    logger.warn(`/api/agent/stream: skipping MCP server with reserved name '${s.name}'`);
                    continue;
                }
                if (s.transport === 'stdio') {
                    if (!s.command) {
                        logger.warn(`/api/agent/stream: stdio MCP server '${s.name}' missing command, skipping`);
                        continue;
                    }
                    additionalMcpServers[s.name] = {
                        type: 'stdio',
                        command: s.command,
                        args: s.args ?? [],
                        ...(s.env ? {env: s.env} : {}),
                    };
                } else if (s.transport === 'streamableHttp') {
                    if (!s.url) {
                        logger.warn(`/api/agent/stream: http MCP server '${s.name}' missing url, skipping`);
                        continue;
                    }
                    additionalMcpServers[s.name] = {
                        type: 'http',
                        url: s.url,
                        ...(s.headers ? {headers: s.headers} : {}),
                    };
                }
            }
            logger.info('/api/agent/stream: enumerated MCP servers', {
                total: servers.length,
                enabled: enabled.length,
                injected: Object.keys(additionalMcpServers).length,
                webappScope: reqWebappId ?? 'host',
                scope,
            });
        }
    } catch (err) {
        logger.warn('/api/agent/stream: MCP enumeration failed (non-fatal)', {
            error: formatErrorMessage(err),
        });
    }
}

import assert from 'node:assert/strict';
import {filterAdditionalMcpServers} from './api.js';

type Server = {name: string; transport: string};

const stubServers: Server[] = [
    {name: 'bytebot', transport: 'stdio'},
    {name: 'bytebot:webapp:abc', transport: 'stdio'},
    {name: 'bytebot:webapp:xyz', transport: 'stdio'},
    {name: 'my-mcp', transport: 'streamableHttp'},
];

class FakeLogger {
    warns: Array<{msg: string; meta?: unknown}> = [];
    warn(msg: string, meta?: unknown) {
        this.warns.push({msg, meta});
    }
}

async function testT1_webappScopeHit() {
    const logger = new FakeLogger();
    const {scope, servers} = filterAdditionalMcpServers(stubServers, 'abc', logger);
    assert.equal(scope, 'webapp');
    assert.equal(servers.length, 1);
    assert.equal(servers[0]!.name, 'bytebot:webapp:abc');
    assert.equal(logger.warns.length, 0);
    console.log('  PASS T1: scope=webapp, only bytebot:webapp:abc returned');
}

async function testT2_lagFallback() {
    const logger = new FakeLogger();
    // Stub WITHOUT bytebot:webapp:abc — simulates liv-core reconcile lag.
    const lagging = stubServers.filter(s => s.name !== 'bytebot:webapp:abc');
    const {scope, servers} = filterAdditionalMcpServers(lagging, 'abc', logger);
    assert.equal(scope, 'lag-fallback');
    const names = servers.map(s => s.name).sort();
    assert.deepEqual(names, ['bytebot', 'my-mcp']);
    assert.equal(logger.warns.length, 1, 'lag fallback should log a WARN');
    assert.match(logger.warns[0]!.msg, /lag fallback|scope-filter/i);
    const meta = logger.warns[0]!.meta as any;
    assert.equal(meta?.webappScope, 'abc');
    assert.equal(meta?.fallback, 'lag');
    console.log('  PASS T2: scope=lag-fallback, host bytebot + my-mcp returned, WARN logged');
}

async function testT3_hostScope() {
    const logger = new FakeLogger();
    const {scope, servers} = filterAdditionalMcpServers(stubServers, null, logger);
    assert.equal(scope, 'host');
    const names = servers.map(s => s.name).sort();
    assert.deepEqual(names, ['bytebot', 'my-mcp']);
    assert.equal(logger.warns.length, 0);
    console.log('  PASS T3: scope=host, per-WebApp instances filtered out');
}

async function testT4_otherWebappRegistered() {
    const logger = new FakeLogger();
