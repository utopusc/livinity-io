WINDOW-DISCOVERY.TS — xdotool fallback:
```ts
export async function listAllWindows(): Promise<WindowInfo[]> {
  let stdout: string
  try {
    const result = await execFileAsync('wmctrl', ['-lG'], {timeout: DEFAULT_TIMEOUT_MS})
    stdout = result.stdout
  } catch (err) {
    const e = err as NodeJS.ErrnoException & {stderr?: string}
    if (e.code === 'ENOENT') return []
    // P100-09-07: wmctrl failed (likely no WM exposing _NET_CLIENT_LIST).
    // Fall through to xdotool which walks the X11 window tree directly.
    const errText = (e.stderr ?? '').toString()
    if (errText.includes('Cannot get client list') || errText.includes('_NET_CLIENT_LIST')) {
      return listAllWindowsViaXdotool()
    }
    throw err
  }
  // ... existing parsing path
}

async function listAllWindowsViaXdotool(): Promise<WindowInfo[]> {
  let stdout: string
  try {
    const result = await execFileAsync('xdotool', ['search', '--name', ''], {timeout: DEFAULT_TIMEOUT_MS})
    stdout = result.stdout
  } catch (err) {
    const e = err as NodeJS.ErrnoException
    if (e.code === 'ENOENT') return []
    return []  // both wmctrl AND xdotool failed → graceful empty list
  }
  const out: WindowInfo[] = []
  const wids = stdout.split(/\r?\n/).filter(Boolean)
  for (const widDecimal of wids) {
    const wid = parseInt(widDecimal.trim(), 10)
    if (!Number.isFinite(wid)) continue
    try {
      const widHex = `0x${wid.toString(16)}`
      const [geoResult, nameResult] = await Promise.all([
        execFileAsync('xdotool', ['getwindowgeometry', '--shell', widHex], {timeout: 1000}).catch(() => ({stdout: ''})),
        execFileAsync('xdotool', ['getwindowname', widHex], {timeout: 1000}).catch(() => ({stdout: ''})),
      ])
      // xdotool getwindowgeometry --shell emits: WINDOW=<wid>\nX=<x>\nY=<y>\nWIDTH=<w>\nHEIGHT=<h>
      const geom = parseXdotoolShellGeom(geoResult.stdout)
      const title = nameResult.stdout.trim()
      if (geom) out.push({wid, title, geometry: geom})
    } catch {
      // skip windows that vanish mid-iteration
    }
  }
  return out
}

function parseXdotoolShellGeom(text: string): {x: number; y: number; w: number; h: number} | null {
  const matches = text.match(/X=(-?\d+)[\s\S]*Y=(-?\d+)[\s\S]*WIDTH=(\d+)[\s\S]*HEIGHT=(\d+)/m)
  if (!matches) return null
  return {x: parseInt(matches[1], 10), y: parseInt(matches[2], 10), w: parseInt(matches[3], 10), h: parseInt(matches[4], 10)}
}
```

Note: xdotool's `search --name ''` matches all windows (empty regex matches everything). On Mini PC Linux this returns one wid per line, decimal format.
