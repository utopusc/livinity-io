CANDIDATE PATCH 3 — skip activate-first chain when wid already focused (input-dispatcher.ts):

```typescript
async function isWidActive(widStr: string): Promise<boolean> {
    return await new Promise<boolean>((resolve) => {
        let stdout = ''
        try {
            const child = spawn('xdotool', ['getactivewindow'], {
                stdio: ['ignore', 'pipe', 'ignore'], env: process.env,
            })
            child.stdout?.on('data', (chunk: Buffer) => { stdout += chunk.toString() })
            child.on('error', () => resolve(false))
            child.on('close', (code) => {
                if (code !== 0) return resolve(false)
                const active = stdout.trim()
                // Compare both decimal AND hex forms (xdotool may emit either):
                resolve(active === widStr || active === '0x' + Number(widStr).toString(16))
            })
        } catch {
            resolve(false)
        }
    })
}
```

Then in dispatchPointer, when isWidActive(widStr) is true, skip windowactivate/windowfocus from the argv:

```typescript
const argv: string[] = []
if (!(await isWidActive(widStr))) {
    argv.push('windowactivate', '--sync', widStr, 'windowfocus', '--sync', widStr)
}
argv.push('mousemove', '--window', widStr, '--sync', String(ix), String(iy))
argv.push(kind, '--clearmodifiers', String(button))
```

NOTE: this adds an EXTRA xdotool subprocess (getactivewindow ~3ms) for the non-focused-yet case but saves ~15ms when the wid is already focused. Net positive after the FIRST click on a given window (every subsequent click is faster).

    /**
     * Phase 100-09-04 — query xdotool for the active window and compare
     * against widStr. Returns true if the wid is already focused (so the
     * activate-first chain can be skipped). Returns false on lookup failure
     * (caller falls back to full chain — safe).
     */
    async function isWidActive(widStr: string): Promise<boolean> {
        return await new Promise<boolean>((resolve) => {
            let stdout = ''
            try {
                const child = spawn('xdotool', ['getactivewindow'], {
                    stdio: ['ignore', 'pipe', 'ignore'],
                    env: {...process.env, ...WEBAPPS_X11_ENV},
                })
                child.stdout?.on('data', (chunk: Buffer) => { stdout += chunk.toString('utf-8') })
                child.on('error', () => resolve(false))
                child.on('close', (code) => {
                    if (code !== 0) return resolve(false)
                    const active = stdout.trim()
                    const widDecimal = widStr
                    const widHex = '0x' + Number(widStr).toString(16)
                    resolve(active === widDecimal || active === widHex)
                })
            } catch {
                resolve(false)
            }
        })
    }
