**Step 2 — Implement `translateLegacyBytebotToolNames` in `skill-replay-tool.ts`.**

Add an exported helper:

```typescript
/**
 * Phase 100-10-02 backwards-compat shim (D-100-10-I).
 *
 * action_log records authored before the Bytebot→Luse rename
 * (skill_version <= 2) reference tools as `mcp__bytebot__<name>`.
 * After the rename, the registered tool names are `mcp__luse__<name>`.
 * On read of legacy skills, lazy-translate the prefix so the existing
 * skill replay path continues to find the tool in the dispatcher.
 *
 * Removal target: v34 (cleanup tracked in CONTEXT.md G-100-10-F).
 */
export function translateLegacyBytebotToolNames(
  events: ReadonlyArray<{tool: string; args: unknown}>,
  skillVersion: number,
): Array<{tool: string; args: unknown}> {
  if (skillVersion > 2) return events.slice()
  return events.map((e) =>
    e.tool.startsWith('mcp__bytebot__')
      ? {...e, tool: 'mcp__luse__' + e.tool.slice('mcp__bytebot__'.length)}
      : e,
  )
}
```
