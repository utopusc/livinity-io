```tsx
import {useWebAppAgent} from '@/hooks/use-webapp-agent'
import {useWebAppDrawerStore} from '@/modules/window/webapp-drawer-store'

export function WebAppFloatingActionBar({webappId}: {webappId: string}) {
  const mode = useWebAppDrawerStore((s) => s.chatInputModeByWebappId[webappId] ?? 'icons')
  const setMode = useWebAppDrawerStore((s) => s.setChatInputMode)
  // ...existing isRecording, toggleTeachRecording, etc...

  if (mode === 'chat-input') {
    return <ChatInputBar webappId={webappId} onClose={() => setMode(webappId, 'icons')} />
  }
  return <IconBar webappId={webappId} onChatClick={() => setMode(webappId, 'chat-input')} />
}

function ChatInputBar({webappId, onClose}: {webappId: string; onClose: () => void}) {
  const {sendMessage, isStreaming} = useWebAppAgent(webappId)
  const [input, setInput] = useState('')
  const inputRef = useRef<HTMLInputElement>(null)

  useEffect(() => {
    inputRef.current?.focus()
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose()
    }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [onClose])

  const handleSend = () => {
    if (!input.trim() || isStreaming) return
    sendMessage(input)
    setInput('')
    onClose() // back to icons after send
  }

  return (
    <Magnetic intensity={0.2}>
      <motion.div
        layout
        className='fixed bottom-4 left-1/2 -translate-x-1/2 flex items-center gap-2 rounded-full bg-white/90 backdrop-blur-xl border border-black/5 shadow-lg px-3 py-2 z-50'
      >
        <Input
          ref={inputRef}
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => e.key === 'Enter' && handleSend()}
          placeholder='Mesaj yaz...'
          className='w-[360px] border-none focus-visible:ring-0 bg-transparent'
          disabled={isStreaming}
        />
        <Button size='icon' variant='ghost' onClick={handleSend} disabled={!input.trim() || isStreaming} aria-label='Send'>
          <Send className='h-4 w-4' />
        </Button>
        <Button size='icon' variant='ghost' onClick={onClose} aria-label='Close chat input'>
          <X className='h-4 w-4' />
        </Button>
      </motion.div>
    </Magnetic>
  )
}

function IconBar({webappId, onChatClick}: {webappId: string; onChatClick: () => void}) {

  // (keep Teach / Auto / Skills buttons unchanged from current state; they're rewired in 09-09)
}
```
