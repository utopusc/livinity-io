<objective>

The black space below the stream goes away. The noVNC `<canvas>` element gets `width: 100%`, `height: 100%`, `object-fit: cover` (or `contain` if user feedback after deploy says they prefer letterboxing — start with `cover` per CONTEXT default).

DELETE `webapp-auto-drawer.tsx` outright (`git rm`). Narrow `WebAppDrawerMode` type from `'chat' | 'teach' | 'auto'` to `'chat' | 'teach'`. After this plan, the floating action bar shows only Chat + Teach (2 icons). Skill button lives outside top-right. Stream fills the window.

```typescript
// Phase 100-10-05 — WebAppFloatingSkillsButton.
//
// Per D-100-10-D: Skill popover trigger MOVED from inside the WebApp
// window's top-right (where 09-06 placed it as <WebAppSkillsPopover/>)
// to OUTSIDE the window, anchored at the window's top-right corner —
// mirroring the 100-06 pattern (floating action bar moved OUTSIDE at
// the bottom). Single round icon button that opens a Radix Popover with
// the skills list (Play/Delete actions per skill).
//
// Sacred SHA: liv/packages/core/src/sdk-agent-runner.ts unchanged.

import {motion} from 'framer-motion'
import {Library} from 'lucide-react'
import {useState} from 'react'

import {Magnetic} from '@/components/motion-primitives/magnetic'
import {Popover, PopoverContent, PopoverTrigger} from '@/shadcn-components/ui/popover'
import {trpcReact} from '@/trpc/trpc'
import {cn} from '@/shadcn-lib/utils'

export interface WebAppFloatingSkillsButtonProps {
  webappId: string
  windowX: number          // viewport-relative left of the WebApp window
  windowY: number          // viewport-relative top of the WebApp window
  windowWidth: number      // WebApp window width
  zIndex: number           // z-index of parent window (button sits one above)
  onReplaySkill?: (skillId: string) => void
}

export function WebAppFloatingSkillsButton(props: WebAppFloatingSkillsButtonProps) {
  const {webappId} = props
  const [open, setOpen] = useState(false)
  const listQuery = trpcReact.webapp.skills.list.useQuery(
    {webappId},
    {enabled: !!webappId && /^[0-9a-f-]{36}$/i.test(webappId), staleTime: 30_000},
  )
  const utils = trpcReact.useUtils()
  const deleteMutation = trpcReact.webapp.skills.delete.useMutation({
    onSuccess: () => void utils.webapp.skills.list.invalidate({webappId}),
  })
  const skills = listQuery.data ?? []

  return (
    <motion.div
      className='fixed select-none'
      style={{
        left: props.windowX + props.windowWidth - 16,
        top: props.windowY + 16,
        transform: 'translateX(-100%)',  // anchor to right edge of button
        zIndex: props.zIndex + 1,
      }}
      initial={{opacity: 0, y: -10, scale: 0.9}}
      animate={{opacity: 1, y: 0, scale: 1}}
      exit={{opacity: 0, y: -10, scale: 0.9}}
      transition={{type: 'spring', stiffness: 500, damping: 35}}
      layout
    >
      <Popover open={open} onOpenChange={setOpen}>
        <PopoverTrigger asChild>
          <Magnetic intensity={0.3} range={60} springOptions={{stiffness: 200, damping: 12, mass: 0.15}}>
            <button
              type='button'
              aria-label='Skills'
              className={cn(
                'flex h-9 w-9 items-center justify-center rounded-full bg-white/90 backdrop-blur-xl border border-neutral-200/60 shadow-[0_2px_8px_rgba(0,0,0,0.08)] text-neutral-500 hover:bg-primary hover:border-primary/80 hover:text-white transition-all duration-200',
              )}
            >
              <Library className='h-4 w-4' strokeWidth={2.25} />
              {skills.length > 0 ? (
                <span className='absolute -top-1 -right-1 min-w-[18px] h-[18px] rounded-full bg-primary text-white text-[10px] font-semibold flex items-center justify-center px-1 ring-2 ring-white'>
                  {skills.length}
                </span>
              ) : null}
            </button>
          </Magnetic>
        </PopoverTrigger>
        <PopoverContent side='bottom' align='end' className='w-72 p-2'>
          {/* skills list — same JSX as the old WebAppSkillsPopover popover content */}
          {skills.length === 0 ? (
            <div className='px-2 py-3 text-caption-sm text-text-tertiary'>No saved skills yet. Click Teach to record one.</div>
          ) : (
            <div className='flex flex-col gap-1'>
              {skills.map((s) => (
                <div key={s.id} className='flex items-center gap-2 rounded-radius-sm px-2 py-1.5 hover:bg-surface-2'>
                  <span className='flex-1 truncate text-caption-sm text-text-primary'>{s.skillName}</span>
                  <button type='button' aria-label='Play' onClick={() => props.onReplaySkill?.(s.id)} className='flex h-6 w-6 items-center justify-center rounded-radius-xs text-text-secondary hover:bg-surface-1 hover:text-text-primary'>
                    {/* Play icon */}
                    ▶
                  </button>
                  <button type='button' aria-label='Delete' onClick={() => deleteMutation.mutate({skillId: s.id})} className='flex h-6 w-6 items-center justify-center rounded-radius-xs text-text-secondary hover:bg-red-500/10 hover:text-red-500'>
                    {/* Trash icon */}
                    🗑
                  </button>
                </div>
              ))}
            </div>
          )}
        </PopoverContent>
      </Popover>
    </motion.div>
  )
}
```
