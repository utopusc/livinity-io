<objective>
Grow the Phase 243 single-pane terminal panel into a multi-tab UI that reattaches on browser reload.

Purpose: The user-visible delta. v43 ships a single xterm window; v44 must show a tab strip with the tabs the user opened in this browser tab persisting across reload. Each browser-side tab maps to a server-side PtySession via a sessionId stored in localStorage.

Per CONTEXT:
- localStorage key shape: `livos.v44.terminal.session.<tabKey>` where `<tabKey>` is a browser-minted uuidv7 (stable per UI tab, NOT the server sessionId)
- value at the key: the server-side sessionId (uuidv7) returned by `{type:'ready', sessionId}` or used in `?attach=<sessionId>`
- On mount, the panel reads ALL `livos.v44.terminal.session.*` keys and tries to reattach each. Failed reattaches (4404) → entry cleared, tab shows a "session expired, click + to start a new one" empty state OR is removed entirely (CONTEXT is ambiguous — choose: entry cleared + tab removed; a fresh "+" click creates a replacement).

Backward compat: when localStorage is empty (first load OR cleared), the panel auto-opens 1 tab via `?create` — feels identical to Phase 243 single-session.

Server → Client WS messages (from 246-03):
- `{type:'ready', sessionId}` → create path; write sessionId to localStorage + tab state
- `{type:'reattached', sessionId, scrollback}` → attach path; replay scrollback via `scrollback.forEach(line => term.write(line))` BEFORE wiring live data forwarder
- `{type:'data', data}` → term.write(data)
- `{type:'exit', code, signal}` → set status='exited', terminal.writeln('[exited]'), DO NOT auto-close ws (server already closed it)

    ```ts
    /**
     * Phase 246-04 — Browser-local tab ↔ server-sessionId map.
     * Storage key shape per CONTEXT D-V44 spec:
     *   `livos.v44.terminal.session.<tabKey>` → <serverSessionId>
     */
    export const TERMINAL_SESSION_STORAGE_PREFIX = 'livos.v44.terminal.session.'

    function buildKey(tabKey: string): string {
      return TERMINAL_SESSION_STORAGE_PREFIX + tabKey
    }

    export function readAllTabSessions(storage: Pick<Storage, 'key' | 'length' | 'getItem'> = window.localStorage): Record<string, string> {
      const out: Record<string, string> = {}
      for (let i = 0; i < storage.length; i++) {
        const fullKey = storage.key(i)
        if (fullKey && fullKey.startsWith(TERMINAL_SESSION_STORAGE_PREFIX)) {
          const tabKey = fullKey.slice(TERMINAL_SESSION_STORAGE_PREFIX.length)
          const sessionId = storage.getItem(fullKey)
          if (sessionId) out[tabKey] = sessionId
        }
      }
      return out
    }

    export function writeTabSession(tabKey: string, sessionId: string, storage: Pick<Storage, 'setItem'> = window.localStorage): void {
      storage.setItem(buildKey(tabKey), sessionId)
    }

    export function removeTabSession(tabKey: string, storage: Pick<Storage, 'removeItem'> = window.localStorage): void {
      storage.removeItem(buildKey(tabKey))
    }
    ```

    ```ts
    export interface UseTerminalWsOptions {
      mode: 'create' | 'attach'
      sessionId?: string                                  // required when mode === 'attach'
      onMessage: (msg: ServerToClient) => void
      onOpen: () => void
      onClose: (event: CloseEvent) => void
    }

    export function useTerminalWs(opts: UseTerminalWsOptions): { send: (msg: ClientToServer) => void; readyState: number } {
      const wsRef = useRef<WebSocket | null>(null)
      const [readyState, setReadyState] = useState<number>(WebSocket.CONNECTING)

      useEffect(() => {
        const url = buildWsUrl(opts.mode, opts.sessionId)
        const ws = new WebSocket(url)
        wsRef.current = ws
        ws.onopen = () => { setReadyState(WebSocket.OPEN); opts.onOpen() }
        ws.onmessage = e => {
          try {
            const msg = JSON.parse(e.data) as ServerToClient
            opts.onMessage(msg)
          } catch {
            opts.onMessage({type: 'error', message: 'parse error'})
          }
        }
        ws.onclose = e => { setReadyState(WebSocket.CLOSED); opts.onClose(e) }
        return () => ws.close()
      }, [opts.mode, opts.sessionId])

      function send(msg: ClientToServer) {
        if (wsRef.current?.readyState === WebSocket.OPEN) wsRef.current.send(JSON.stringify(msg))
      }
      return {send, readyState}
    }

      const proto = window.location.protocol === 'https:' ? 'wss://' : 'ws://'
      const port = window.location.port ? `:${window.location.port}` : ''
      const base = `${proto}${window.location.hostname}${port}/livos/terminal/ws`
      if (mode === 'attach' && sessionId) return `${base}?attach=${encodeURIComponent(sessionId)}`
      return base
    ```

    ```tsx
    function TerminalTabPane({tabKey, mode, sessionId, isActive, onSessionResolved, onExited, onExpired}: TerminalTabPaneProps) {
      const containerRef = useRef<HTMLDivElement>(null)
      const termRef = useRef<Terminal | null>(null)
      const fitRef = useRef<FitAddon | null>(null)

      const {send, readyState} = useTerminalWs({
        mode,
        sessionId: sessionId ?? undefined,
        onOpen: () => {
          if (mode === 'create') {
            const dims = fitRef.current?.proposeDimensions()
            send({type: 'init', cols: dims?.cols ?? 80, rows: dims?.rows ?? 24})
          }
        },
        onMessage: msg => {
          if (msg.type === 'ready') {
            onSessionResolved(tabKey, msg.sessionId)
            writeTabSession(tabKey, msg.sessionId)
          } else if (msg.type === 'reattached') {
            // Replay scrollback FIRST, then onSessionResolved (already set in localStorage from mount).
            msg.scrollback.forEach(line => termRef.current?.write(line))
          } else if (msg.type === 'data') {
            termRef.current?.write(msg.data)
          } else if (msg.type === 'exit') {
            termRef.current?.writeln('\r\n[session exited code=' + msg.code + ']')
            onExited(tabKey)
            removeTabSession(tabKey)
          } else if (msg.type === 'error') {
            termRef.current?.writeln('\r\n[error] ' + msg.message)
          }
        },
        onClose: ev => {
          // If close happened before any ready/reattached arrived AND mode === 'attach', the session likely expired.
          if (mode === 'attach' && ev.code === 4404) {
            onExpired(tabKey)
            removeTabSession(tabKey)
          }
        },
      })

      useEffect(() => {
        // Construct Terminal + FitAddon + WebLinksAddon (theme verbatim from Phase 243)
        // ... (mirror Phase 243 PersistentTerminalPanel.tsx mount block exactly — theme, addons, fit, onData → send)
      }, [])

      return <div ref={containerRef} className={`h-full w-full bg-[#0b0b0c] p-2 ${isActive ? '' : 'hidden'}`} />
    }
    ```
