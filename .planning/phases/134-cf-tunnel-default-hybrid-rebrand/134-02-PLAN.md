# Plan 134-02 — Server5 wizard CF Tunnel auto-provision

**Wave:** 1 (parallel with 134-01)
**Autonomous:** true (on-server canonical state — Server5 has no git tree; edits applied via SSH + restart)
**Sacred SHA invariant:** unaffected (Server5 platform code; no `liv/` touch)

## What

The wizard at `https://livinity.io/dashboard/install` shall:
1. Accept ONLY a domain name (and optionally the operator's CF API token if not already cached).
2. Server-side: call CF API to create a CF Tunnel + bind the public hostname + create DNS CNAME.
3. Emit an install one-liner with the auto-generated `--cf-tunnel-token` pre-filled.

End result: the user pastes ONE command on any box and it works. No CF dashboard navigation needed.

## How

### 1. `cloudflare-api.ts` additions

Helper functions (pseudo-code; actual implementation matches existing style in the file):

```typescript
// Discover account ID for a CF API token (idempotent — cacheable per token).
export async function verifyTokenAccountId(token: string): Promise<string> {
    const r = await fetch('https://api.cloudflare.com/client/v4/user/tokens/verify', {
        headers: { Authorization: `Bearer ${token}` }
    });
    const body = await r.json();
    if (!body.success) throw new CfApiError(403, 'token verify failed', body);
    // /accounts endpoint returns the token's bound account; pick first.
    const accounts = await fetch('https://api.cloudflare.com/client/v4/accounts', {
        headers: { Authorization: `Bearer ${token}` }
    }).then(r => r.json());
    return accounts.result[0].id;
}

// Create CF Tunnel (or find existing by name).
export async function createCfTunnel(
    token: string, accountId: string, name: string
): Promise<{ id: string; token: string }> {
    // Idempotency: GET first
    const existing = await fetch(
        `https://api.cloudflare.com/client/v4/accounts/${accountId}/cfd_tunnel?name=${name}`,
        { headers: { Authorization: `Bearer ${token}` } }
    ).then(r => r.json());
    if (existing.success && existing.result?.length > 0) {
        const t = existing.result[0];
        // Fetch token for existing tunnel (different endpoint)
        const tk = await fetch(
            `https://api.cloudflare.com/client/v4/accounts/${accountId}/cfd_tunnel/${t.id}/token`,
            { headers: { Authorization: `Bearer ${token}` } }
        ).then(r => r.json());
        return { id: t.id, token: tk.result };
    }
    // Create
    const r = await fetch(
        `https://api.cloudflare.com/client/v4/accounts/${accountId}/cfd_tunnel`,
        {
            method: 'POST',
            headers: {
                Authorization: `Bearer ${token}`,
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({
                name,
                config_src: 'cloudflare'  // remote-managed config
            })
        }
    );
    const body = await r.json();
    if (!body.success) {
        // Surface scope errors clearly
        if (r.status === 403) {
            throw new CfApiError(403,
                'CF API token lacks Tunnel:Edit scope — add it at dash.cloudflare.com > Profile > API Tokens',
                body);
        }
        throw new CfApiError(r.status, 'cfd_tunnel create failed', body);
    }
    return { id: body.result.id, token: body.result.token };
}

// Bind public hostname → http://localhost:80. Idempotent (replaces config).
export async function configureTunnelPublicHostname(
    token: string, accountId: string, tunnelId: string, hostname: string
): Promise<void> {
    const r = await fetch(
        `https://api.cloudflare.com/client/v4/accounts/${accountId}/cfd_tunnel/${tunnelId}/configurations`,
        {
            method: 'PUT',
            headers: {
                Authorization: `Bearer ${token}`,
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({
                config: {
                    ingress: [
                        // Wildcard for any LivOS subdomain
                        { hostname: `*.${hostname}`, service: 'http://localhost:80' },
                        // Apex
                        { hostname, service: 'http://localhost:80' },
                        // Catch-all required by CF
                        { service: 'http_status:404' }
                    ]
                }
            })
        }
    );
    if (!r.ok) {
        const body = await r.json().catch(() => null);
        throw new CfApiError(r.status, 'cfd_tunnel configure failed', body);
    }
}

// Create CNAME for hostname → <tunnelId>.cfargotunnel.com. Idempotent.
export async function upsertDnsCname(
    token: string, zoneId: string, name: string, target: string
): Promise<void> {
    // GET first
    const existing = await fetch(
        `https://api.cloudflare.com/client/v4/zones/${zoneId}/dns_records?type=CNAME&name=${name}`,
        { headers: { Authorization: `Bearer ${token}` } }
    ).then(r => r.json());
    if (existing.success && existing.result?.length > 0) {
        const rec = existing.result[0];
        if (rec.content === target) return;  // already correct
        // Update existing
        await fetch(
            `https://api.cloudflare.com/client/v4/zones/${zoneId}/dns_records/${rec.id}`,
            {
                method: 'PUT',
                headers: { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' },
                body: JSON.stringify({ type: 'CNAME', name, content: target, proxied: true })
            }
        );
        return;
    }
    // Create
    await fetch(
        `https://api.cloudflare.com/client/v4/zones/${zoneId}/dns_records`,
        {
            method: 'POST',
            headers: { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' },
            body: JSON.stringify({ type: 'CNAME', name, content: target, proxied: true })
        }
    );
}
```

### 2. `route.ts` (api-keys POST handler) extension

Append AFTER existing API-key mint + `custom_domains` write logic:

```typescript
// Phase 134 — auto-provision CF Tunnel for the user's domain
import {
    createCfTunnel,
    configureTunnelPublicHostname,
    upsertDnsCname,
    verifyTokenAccountId
} from '@/lib/cloudflare-api';

const operatorCfToken = process.env.CF_TUNNEL_OPERATOR_TOKEN;
if (!operatorCfToken) {
    // Wizard explicitly told operator to configure this — surface clear error
    return Response.json({
        ok: false,
        error: 'CF_TUNNEL_OPERATOR_TOKEN not configured in Server5 env'
    }, { status: 500 });
}

let cfTunnelToken: string | null = null;
try {
    const accountId = await verifyTokenAccountId(operatorCfToken);
    // Tunnel name: livos-<user-id> (stable per user — idempotent on re-run)
    const tunnelName = `livos-${user.id}`;
    const { id: tunnelId, token: tunnelToken } = await createCfTunnel(
        operatorCfToken, accountId, tunnelName);
    await configureTunnelPublicHostname(
        operatorCfToken, accountId, tunnelId, domain /* from request body */);
    // Find zone ID for this domain (existing helper or new lookup)
    const zoneId = await findZoneIdForDomain(operatorCfToken, domain);
    await upsertDnsCname(operatorCfToken, zoneId, domain, `${tunnelId}.cfargotunnel.com`);
    await upsertDnsCname(operatorCfToken, zoneId, `*.${domain}`, `${tunnelId}.cfargotunnel.com`);
    cfTunnelToken = tunnelToken;
} catch (err) {
    console.error('[wizard] CF tunnel provision failed:', err);
    return Response.json({
        ok: false,
        error: err instanceof CfApiError ? err.message : String(err)
    }, { status: err instanceof CfApiError ? err.status : 500 });
}

// Existing response, plus the new tunnel token
return Response.json({
    ok: true,
    api_key: liv_k_...,
    domain: domain,
    cf_tunnel_token: cfTunnelToken,
    install_command: [
        'curl -fsSL https://livinity.io/install.sh | sudo bash -s -- \\',
        `  --mode hybrid \\`,
        `  --domain ${domain} \\`,
        `  --api-key ${liv_k_...} \\`,
        `  --cf-tunnel-token ${cfTunnelToken}`
    ].join('\n')
});
```
