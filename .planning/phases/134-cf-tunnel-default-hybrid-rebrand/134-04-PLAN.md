## Test cases (test-parse-cli-args.sh updates)

Add to existing fixture:

```bash
# Phase 134 — hybrid mode now accepts --cf-tunnel-token
(
    source ./_logging.sh
    source ./parse-cli.sh
    parse_cli --mode hybrid --domain test.example.com --cf-tunnel-token "fake-token-123"
    [[ "$MODE" == "hybrid" ]] || { echo "FAIL: MODE != hybrid"; exit 1; }
    [[ "$LIVOS_CF_TUNNEL_TOKEN" == "fake-token-123" ]] || { echo "FAIL: token not parsed"; exit 1; }
)

# Phase 134 — hybrid mode without --cf-tunnel-token fails
(
    set +e
    bash -c '
        source ./_logging.sh
        source ./parse-cli.sh
        parse_cli --mode hybrid --domain test.example.com
    ' 2>/dev/null
    rc=$?
    [[ $rc -eq 64 ]] || { echo "FAIL: expected exit 64, got $rc"; exit 1; }
)

# Phase 134 — --cf-zone-id no longer required in hybrid (back-compat: still accepted)
(
    source ./_logging.sh
    source ./parse-cli.sh
    parse_cli --mode hybrid --domain test.example.com --cf-tunnel-token fake \
        --cf-zone-id legacy-id
    [[ "$LIVOS_CF_ZONE_ID" == "legacy-id" ]] || { echo "FAIL: zone-id back-compat broken"; exit 1; }
)
```
