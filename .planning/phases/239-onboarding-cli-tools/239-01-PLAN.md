    ```typescript
    import {TRPCError} from '@trpc/server'
    import {z} from 'zod'
    import {adminProcedure, router} from './trpc.js'
    import {
      SUPPORTED_CLIS_SET,
      installCli,
      detectCli,
      type InstallResult,
      type DetectResult,
      type InstallerLogger,
    } from '../../cli-installer/index.js'

    export interface CliInstallerRouterDeps {
      logger: InstallerLogger
      // DI seams for tests; production wire-up passes nothing → real impl runs
      installFn?: (input: {name: string}, deps: {logger: InstallerLogger}) => Promise<InstallResult>
      detectFn?: (input: {name: string}, deps: {logger: InstallerLogger}) => Promise<DetectResult>
    }

    const NameInput = z.object({name: z.string().min(1).max(64)})

    function assertWhitelisted(name: string): void {
      if (!SUPPORTED_CLIS_SET.has(name as any)) {
        throw new TRPCError({
          code: 'BAD_REQUEST',
          message: `cli-installer: '${name}' is not in SUPPORTED_CLIS — install refused (D-239-07 RCE boundary)`,
          // Phase 240 + UI rely on this string code for error mapping
          cause: 'CLI_NOT_SUPPORTED',
        })
      }
    }

    export function createCliInstallerRouter(deps: CliInstallerRouterDeps) {
      const install = deps.installFn ?? installCli
      const detect = deps.detectFn ?? detectCli
      return router({
        install: adminProcedure.input(NameInput).mutation(async ({input}) => {
          assertWhitelisted(input.name)
          return install(input, {logger: deps.logger})
        }),
        detect: adminProcedure.input(NameInput).query(async ({input}) => {
          assertWhitelisted(input.name)
          return detect(input, {logger: deps.logger})
        }),
      })
    }

    // Empty-injection default — throws PRECONDITION_FAILED until production swap
    export const cliInstallerRouter = createCliInstallerRouter({
      logger: {info: () => {}, warn: () => {}, error: () => {}},
      installFn: async () => {
        throw new TRPCError({
          code: 'PRECONDITION_FAILED',
          message: 'cli-installer-router not yet injected — livinityd boot did not wire production deps',
        })
      },
      detectFn: async () => {
        throw new TRPCError({
          code: 'PRECONDITION_FAILED',
          message: 'cli-installer-router not yet injected — livinityd boot did not wire production deps',
        })
      },
    })
    ```
