       ```typescript
       // Phase 164-03 — Inbox writeback for autonomous agent results.
       //
       // After each autonomous run, scheduler.ts (164-02) calls writeInboxEntry
       // to materialise the result as an Obsidian-compatible markdown file at
       // <vaultPath>/inbox/<YYYY-MM-DD>_<HH-MM>_<agent>.md with locked
       // frontmatter (status/cost/duration/turns/model) + body + ## Backlinks
       // section.
       //
       // Pure file-IO. No Redis, no SDK, no network. Filesystem and clock are
       // injectable for deterministic vitest runs.

       import {createHash} from 'node:crypto'
       import {mkdir, readFile, writeFile, stat, chmod, access} from 'node:fs/promises'
       import path from 'node:path'

       export type AutonomousRunStatus = 'success' | 'error' | 'budget_exceeded' | 'skipped'

       export interface InboxEntryInput {
         vaultPath: string
         agent: string
         status: AutonomousRunStatus
         startedAt: Date
         durationMs: number
         costUsd: number
         turns: number
         model: string
         body: string
         agentSourceRelPath: string  // e.g. 'livos-agents/nightly-backup-audit'
         backlinks?: string[]        // optional extra wikilink targets (no [[ ]] needed)
       }

       export type WriteInboxResult =
         | {written: true; path: string}
         | {written: false; reason: 'duplicate'; path: string}
         | {written: false; reason: 'error'; err: string}

       function pad2(n: number): string {
         return n < 10 ? `0${n}` : String(n)
       }

       function filenameMinutePrefix(d: Date, agent: string): string {
         // UTC for filesystem determinism; frontmatter ALSO uses UTC ISO 8601
         const yyyy = d.getUTCFullYear()
         const mm = pad2(d.getUTCMonth() + 1)
         const dd = pad2(d.getUTCDate())
         const hh = pad2(d.getUTCHours())
         const min = pad2(d.getUTCMinutes())
         return `${yyyy}-${mm}-${dd}_${hh}-${min}_${agent}`
       }

       function buildFrontmatter(input: InboxEntryInput): string {
         return [
           '---',
           `agent: ${input.agent}`,
           `status: ${input.status}`,
           `started: ${input.startedAt.toISOString()}`,
           `duration_ms: ${Math.round(input.durationMs)}`,
           `cost_usd: ${input.costUsd.toFixed(4)}`,
           `turns: ${Math.round(input.turns)}`,
           `model: ${input.model}`,
           '---',
           '',
         ].join('\n')
       }

       function buildTitle(input: InboxEntryInput): string {
         const dateStr = input.startedAt.toISOString().slice(0, 10)
         return `# ${input.agent} — ${dateStr}`
       }

       function buildBacklinks(input: InboxEntryInput): string {
         const lines = ['## Backlinks', '', `- [[${input.agentSourceRelPath}]] — agent definition`]
         for (const b of input.backlinks ?? []) {
           lines.push(`- [[${b}]]`)
         }
         return lines.join('\n')
       }

       function buildContent(input: InboxEntryInput): string {
         return [
           buildFrontmatter(input),
           buildTitle(input),
           '',
           input.body.trim().length > 0 ? input.body : '_(no body)_',
           '',
           buildBacklinks(input),
           '',
         ].join('\n')
       }

       async function pathExists(p: string): Promise<boolean> {
         try { await access(p); return true } catch { return false }
       }

       async function sha256OfFile(p: string): Promise<string> {
         const buf = await readFile(p)
         return createHash('sha256').update(buf).digest('hex')
       }

       function sha256OfString(s: string): string {
         return createHash('sha256').update(s, 'utf8').digest('hex')
       }

       export async function writeInboxEntry(input: InboxEntryInput): Promise<WriteInboxResult> {
         try {
           const inboxDir = path.join(input.vaultPath, 'inbox')
           await mkdir(inboxDir, {recursive: true})

           const content = buildContent(input)
           const expectedHash = sha256OfString(content)
           const minutePrefix = filenameMinutePrefix(input.startedAt, input.agent)
           // Find the next available filename:
           //   <minutePrefix>.md
           //   <minutePrefix>_2.md
           //   <minutePrefix>_3.md
           //   ...
           // Cap at _99 to avoid runaway loops.
           let chosenPath: string | null = null
           for (let i = 1; i <= 100; i++) {
             const candidateName = i === 1 ? `${minutePrefix}.md` : `${minutePrefix}_${i}.md`
             const candidatePath = path.join(inboxDir, candidateName)
             if (!(await pathExists(candidatePath))) {
               chosenPath = candidatePath
               break
             }
             // Idempotency check: if any existing same-minute-prefix file has
             // identical content, treat as duplicate.
             try {
               const existingHash = await sha256OfFile(candidatePath)
               if (existingHash === expectedHash) {
                 return {written: false, reason: 'duplicate', path: candidatePath}
               }
             } catch {/* read failed — try next */}
           }
           if (chosenPath === null) {
             return {written: false, reason: 'error', err: 'inbox collision cap (100) reached'}
           }

           await writeFile(chosenPath, content, {encoding: 'utf8'})
           await chmod(chosenPath, 0o644).catch(() => {/* tolerate non-root */})
           return {written: true, path: chosenPath}
         } catch (err) {
           return {written: false, reason: 'error', err: (err as Error).message}
         }
       }
       ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-164-03-01 | Tampering | Agent body contains `---\n` sequences that break frontmatter parsing | accept | Obsidian's frontmatter parser stops at the FIRST `---` after the opening one. Our content puts the body AFTER the closing `---`, so embedded `---` in the body is downstream of the frontmatter and only affects Obsidian's content rendering, not metadata. Operator visual review is sufficient for v34 |
| T-164-03-02 | Tampering | Agent body contains wikilinks `[[evil/path]]` that fabricate backlinks | mitigate | The body is rendered verbatim and Obsidian's graph DOES follow body-level wikilinks. Our explicit `## Backlinks` section is authoritative; fabricated body links land in the graph too but are NOT presented as the agent's official trace. Phase 165 vault-doctor will flag suspicious backlinks |
| T-164-03-03 | Denial of Service | Sub-minute collision burst could exhaust the 100-sequence cap | mitigate | Cap at 100 returns a structured error instead of looping forever. Scheduler logs the error; operator can debug. In practice node-cron does not fire sub-second so 100 collisions per minute is impossible |
| T-164-03-04 | Information Disclosure | Inbox files written 0644 (world-readable on multi-user systems) | accept | LivOS Mini PC is single-user (bruce). Vault dir is bruce-owned (vault-scaffolder enforces this) and `/home/bruce/` perms restrict outside access |
