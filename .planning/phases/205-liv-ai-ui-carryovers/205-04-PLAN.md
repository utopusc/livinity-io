**Step A — Create `openclaw-config-store.ts`:**

```typescript
import { existsSync, readFileSync, writeFileSync, renameSync, chmodSync } from 'node:fs';

export interface OpenclawConfig {
  gateway?: {
    controlUi?: { allowedOrigins?: string[] };
    auth?: { token?: string; mode?: 'token' | 'master' };
  };
  plugins?: { entries?: unknown[] };
  [k: string]: unknown;
}

export class OpenclawConfigStore {
  constructor(private readonly path: string) {}

  read(): OpenclawConfig {
    if (!existsSync(this.path)) {
      throw new Error('OPENCLAW_CONFIG_MISSING');
    }
    return JSON.parse(readFileSync(this.path, 'utf8')) as OpenclawConfig;
  }

  patch(mut: (cfg: OpenclawConfig) => void): OpenclawConfig {
    const cfg = this.read();
    mut(cfg);
    const tmp = `${this.path}.tmp.${process.pid}.${Date.now()}`;
    writeFileSync(tmp, JSON.stringify(cfg, null, 2), { mode: 0o600 });
    try { chmodSync(tmp, 0o600); } catch { /* swallow */ }
    renameSync(tmp, this.path);
    return cfg;
  }
}
```

  // Per 205-01 A1 — apply whichever field was confirmed populated.
  // Try header first (most reliable for browser callers per 205-01 AUTH PATH):
  const headerId = ctx.request?.headers?.['x-claw-device-id'];
  if (typeof headerId === 'string' && headerId.length > 0) return headerId;

  // Fallback: cookie/Authorization JWT
  let token = ctx.request?.headers?.authorization?.toString().split(' ')[1];
  if (!token) token = ctx.request?.cookies?.LIVINITY_SESSION;
  if (!token) return null;

    if (typeof p.deviceId === 'string') return p.deviceId;
    if (typeof p.jti === 'string') return p.jti;

  return null;
}
