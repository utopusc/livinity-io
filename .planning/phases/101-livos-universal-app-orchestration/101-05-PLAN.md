<objective>
Finish Pillar B by giving native-app spawns a stream. After `spawnNativeApp()` (101-03) returns `{pid, child}`, `bindNativeAppWindow()` polls xdotool for a new WM_CLASS-matching window on the display, allocates a port from `PortAllocator` (101-02), and starts x11vnc against the matched wid. The route handler `apps.native.spawn(id)` becomes the single entry that the dock UI calls.

Purpose: A native app the user launches from the dock must auto-stream into a LivOS stream window in the same lifecycle as a WebApp. Wave 2 closes that gap with a baseline-and-poll matcher (PATTERNS.md "native-app-binder.ts" + RESEARCH.md Pattern 4 lines 555-608).

Output: A green binder test suite + an extended tRPC route that orchestrates spawn → bind → stream → response.

       ```typescript
       import {execFile} from 'node:child_process'
       import {promisify} from 'node:util'
       import {basename} from 'node:path'
       import type {PortAllocator} from '../streaming/port-allocator.js'

       const execFileP = promisify(execFile)
       type ExecFileFn = typeof execFileP

       export class NativeAppWindowNotFoundError extends Error {
         code = 'NATIVE_APP_WINDOW_NOT_FOUND'
         constructor(public wmClass: string) {
           super(`no new window matching WM_CLASS=${wmClass} appeared within timeout`)
         }
       }

       export async function snapshotWindowIds(display: string, fn?: ExecFileFn): Promise<Set<number>> {
         const f = fn ?? execFileP
         try {
           const {stdout} = await f('xdotool', ['search', '--onlyvisible', ''], {env: {DISPLAY: display}})
           const ids = stdout.trim().split('\n').filter(Boolean).map((s) => parseInt(s, 10)).filter((n) => Number.isFinite(n))
           return new Set(ids)
         } catch {
           return new Set()
         }
       }

       export function inferWmClass(binaryPath: string): string {
         return basename(binaryPath).toLowerCase().replace(/\.[^.]+$/, '')
       }

       export interface StreamStartFn {
         (opts: {wid: number; port: number; label?: string}): Promise<{streamId: string; wsUrl: string}>
       }

       export interface BindOpts {
         pid: number
         wmClass: string
         display?: string
         portAllocator: PortAllocator
         startStreamFn: StreamStartFn
         execFileFn?: ExecFileFn
         deadlineMs?: number
         pollIntervalMs?: number
         label?: string
         logger?: {info(m: string): void; warn(m: string): void; error(m: string): void; verbose?(m: string): void}
       }

       export async function bindNativeAppWindow(opts: BindOpts): Promise<{wid: number; port: number; streamId: string; wsUrl: string}> {
         const display = opts.display ?? ':1'
         const execFn = opts.execFileFn ?? execFileP
         const deadline = Date.now() + (opts.deadlineMs ?? 5_000)
         const poll = opts.pollIntervalMs ?? 100

         const baseline = await snapshotWindowIds(display, execFn)

         let wid: number | undefined
         while (Date.now() < deadline) {
           try {
             const {stdout} = await execFn('xdotool', ['search', '--onlyvisible', '--class', opts.wmClass], {env: {DISPLAY: display}})
             const wids = stdout.trim().split('\n').filter(Boolean).map((s) => parseInt(s, 10)).filter((n) => Number.isFinite(n))
             const cand = wids.find((w) => !baseline.has(w))
             if (cand !== undefined) { wid = cand; break }
           } catch {
             // xdotool exits non-zero when no match — keep polling
           }
           await new Promise((r) => setTimeout(r, poll))
         }
         if (wid === undefined) throw new NativeAppWindowNotFoundError(opts.wmClass)

         const port = opts.portAllocator.allocate()
         try {
           const {streamId, wsUrl} = await opts.startStreamFn({wid, port, label: opts.label})
           opts.logger?.info(`native-app bound wid=${wid} port=${port} streamId=${streamId}`)
           return {wid, port, streamId, wsUrl}
         } catch (err) {
           opts.portAllocator.release(port)
           throw err
         }
       }
       ```
