       ```typescript
       import {trpcReact} from '@/trpc/trpc'
       import {toast} from 'sonner' // (verify lib in package: 'sonner' is the standard per existing usage)

       export function useLaunchNativeApp() {
         const spawnMut = trpcReact.apps.native.spawn.useMutation()
         return async function launch(opts: {id: string; name: string}): Promise<{streamId: string; wsUrl: string} | null> {
           try {
             const r = await spawnMut.mutateAsync({id: opts.id})
             return {streamId: (r as any).streamId, wsUrl: (r as any).wsUrl}
           } catch (err) {
             toast.error(`Failed to launch ${opts.name}: ${(err as Error).message}`)
             return null
           }
         }
       }
       ```

       ```typescript
       import {useState} from 'react'
       import {randomUUID} from 'node:crypto' // Will use crypto.randomUUID() at runtime — verify it exists in the UI env (browser has it). If not available, use `crypto.randomUUID()` directly.
       import {Button} from '@/shadcn-components/ui/button'
       import {Dialog, DialogContent, DialogFooter, DialogHeader, DialogTitle} from '@/shadcn-components/ui/dialog'
       import {Input} from '@/shadcn-components/ui/input'
       import {trpcReact} from '@/trpc/trpc'

       const BINARY_PATH_RE = /^\/[a-zA-Z0-9_\-./]+$/
       const ARG_RE = /^[^;&|`$<>(){}\\]*$/

       function validateBinaryPath(s: string): string | null {
         if (!s.trim()) return 'Path is required'
         if (!BINARY_PATH_RE.test(s.trim())) return 'Path must be absolute with no shell metachars'
         return null
       }

       function validateArgsList(raw: string): {args: string[]; error: string | null} {
         const parts = raw.split(',').map((p) => p.trim()).filter(Boolean)
         for (const p of parts) {
           if (!ARG_RE.test(p)) return {args: parts, error: `Arg contains shell metachars: "${p}"`}
         }
         return {args: parts, error: null}
       }

       function validateEnv(entries: Array<[string, string]>): string | null {
         for (const [k] of entries) {
           if (/^(LD_|DYLD_)/.test(k)) return `Env key "${k}" is not allowed (LD_*/DYLD_*)`
         }
         return null
       }

       export interface NativeAppFormProps {
         open: boolean
         onOpenChange: (open: boolean) => void
         onCreated?: (id: string) => void
       }

       export function NativeAppForm({open, onOpenChange, onCreated}: NativeAppFormProps) {
         const utils = trpcReact.useUtils()
         const createMut = trpcReact.apps.native.create.useMutation()

         const [name, setName] = useState('')
         const [iconUrl, setIconUrl] = useState('')
         const [binaryPath, setBinaryPath] = useState('')
         const [argsRaw, setArgsRaw] = useState('')
         const [envEntries, setEnvEntries] = useState<Array<[string, string]>>([])
         const [wmClassHint, setWmClassHint] = useState('')

         const binaryErr = binaryPath ? validateBinaryPath(binaryPath) : null
         const {args, error: argsErr} = validateArgsList(argsRaw)
         const envErr = validateEnv(envEntries)
         const canSave = !!name.trim() && !!binaryPath.trim() && !binaryErr && !argsErr && !envErr

         async function handleSave() {
           if (!canSave) return
           const id = (globalThis.crypto && 'randomUUID' in globalThis.crypto)
             ? (globalThis.crypto as Crypto).randomUUID()
             : `${Date.now()}-${Math.random().toString(16).slice(2)}`
           await createMut.mutateAsync({
             id,
             name: name.trim(),
             iconUrl: iconUrl.trim() || undefined,
             binaryPath: binaryPath.trim(),
             args: args.length > 0 ? args : undefined,
             env: envEntries.length > 0 ? Object.fromEntries(envEntries) : undefined,
             wmClassHint: wmClassHint.trim() || undefined,
           })
           await utils.apps.native.list.invalidate()
           onCreated?.(id)
           onOpenChange(false)
         }

         return (
           <Dialog open={open} onOpenChange={onOpenChange}>
             <DialogContent>
               <DialogHeader><DialogTitle>Add Ubuntu app</DialogTitle></DialogHeader>
               <Input placeholder='Name' value={name} onChange={(e) => setName(e.target.value)} />
               <Input placeholder='Icon URL (optional)' value={iconUrl} onChange={(e) => setIconUrl(e.target.value)} />
               <Input placeholder='/absolute/path/to/binary' value={binaryPath} onChange={(e) => setBinaryPath(e.target.value)} />
               {binaryErr ? <div className='text-destructive text-xs'>{binaryErr}</div> : null}
               <Input placeholder='Args (comma-separated)' value={argsRaw} onChange={(e) => setArgsRaw(e.target.value)} />
               {argsErr ? <div className='text-destructive text-xs'>{argsErr}</div> : null}
               {/* envEntries UI: simple key/value rows + add button — minimal MVP */}
               {envErr ? <div className='text-destructive text-xs'>{envErr}</div> : null}
               <Input placeholder='WM_CLASS hint (optional)' value={wmClassHint} onChange={(e) => setWmClassHint(e.target.value)} />
               <DialogFooter>
                 <Button variant='ghost' onClick={() => onOpenChange(false)}>Cancel</Button>
                 <Button disabled={!canSave || createMut.isPending} onClick={handleSave}>Save</Button>
               </DialogFooter>
             </DialogContent>
           </Dialog>
         )
       }
       ```

       ```typescript
       import {useState} from 'react'
       import {AppIcon} from '@/modules/desktop/app-icon' // adapt to actual path
       import {ContextMenu, ContextMenuContent, ContextMenuItem, ContextMenuTrigger} from '@/shadcn-components/ui/context-menu'
       import {AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle} from '@/shadcn-components/ui/alert-dialog'
       import {contextMenuClasses} from '@/modules/shared/context-menu-classes' // adapt path

         id: string

       }

       export function NativeAppIcon({id, name, iconUrl}: NativeAppIconProps) {

                 </AlertDialogHeader>
                 <AlertDialogFooter>
                   <AlertDialogCancel>Cancel</AlertDialogCancel>
       ```
