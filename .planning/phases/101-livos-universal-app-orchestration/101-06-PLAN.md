       ```typescript
       export interface ActiveAppMeta {
         appId: string
         kind: 'webapp' | 'native'
         url?: string
         binary?: string
         title: string
       }
       export interface ActiveWindowContext {
         activeWid: number
         appMeta: ActiveAppMeta
       }

       const MAX_TITLE_LEN = 256
       const MAX_URL_LEN = 512
       const CONTROL_CHARS_RE = /[\x00-\x1f\x7f]/g

       function clean(s: string, maxLen: number): string {
         const stripped = s.replace(CONTROL_CHARS_RE, '')
         if (stripped.length <= maxLen) return stripped
         return stripped.slice(0, maxLen - 1) + '…'
       }

       export function sanitizeActiveAppMeta(meta: ActiveAppMeta): ActiveAppMeta {
         const kind: 'webapp' | 'native' = meta.kind === 'native' ? 'native' : 'webapp'
         return {
           appId: clean(meta.appId ?? '', 128),
           kind,
           url: meta.url ? clean(meta.url, MAX_URL_LEN) : undefined,
           binary: meta.binary ? clean(meta.binary, MAX_URL_LEN) : undefined,
           title: clean(meta.title ?? '', MAX_TITLE_LEN),
         }
       }

       export function buildActiveWindowSnippet(input: ActiveWindowContext): string {
         if (!Number.isInteger(input.activeWid)) return ''
         const safe = sanitizeActiveAppMeta(input.appMeta)
         const target = safe.url ?? safe.binary ?? '(unknown)'
         return `## Active Window Context
You are operating in the context of the LivOS app: ${safe.title} (${safe.kind}).
Window ID: ${input.activeWid}
URL/Binary: ${target}
Default LUSE_TARGET_WINDOW_ID for all your tool calls is ${input.activeWid} unless you override explicitly.`
       }
       ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| UI WS payload → backend → system prompt | Untrusted: `activeAppMeta.title` is set client-side and interpolated verbatim into the LLM system prompt |
| Backend → SDK runner contextPrefix | Trusted at the broker layer; content is part of the agent's input |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-101-03 | Tampering / Information Disclosure (prompt injection) | agent-prompt-builder.ts string interpolation of `appMeta.title` | mitigate | `sanitizeActiveAppMeta()` strips control chars (`\x00-\x1f\x7f`) and length-caps at 256/512 chars (Task 1). The snippet has fixed structural surrounding text, so a one-line title cannot easily break out into "new instructions" without the agent recognizing the injection attempt. The Sacred SDK runner's existing identity preservation (BROKER-CARRY-05) is a second line of defense. |
| T-101-03b | Authorization | `activeWid` could reference a window the user does not own | accept (defer to 102) | Today the UI only knows wids it spawned. Cross-user wid forgery would require a hand-crafted WS client. Listed in 101-RESEARCH.md §Security as a v2+ hardening item. |
</threat_model>
