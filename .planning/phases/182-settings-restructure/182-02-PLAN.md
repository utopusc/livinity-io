```typescript
// Group items by group, excluding footer items from main list
const groupOrder: SettingsGroup[] = ['personal', 'workspace', 'ai', 'system']
const groupLabels: Record<SettingsGroup, string> = {
  personal: 'PERSONAL',
  workspace: 'WORKSPACE',
  ai: 'AI',
  system: 'SYSTEM',
}

const mainItems = MENU_ITEMS.filter((item) => !item.footer && (isAdmin || !item.adminOnly))
const footerItems = MENU_ITEMS.filter((item) => item.footer && (isAdmin || !item.adminOnly))

// Render: for each group, show header then items
const grouped = groupOrder
  .map((g) => ({
    group: g,
    items: mainItems.filter((item) => item.group === g),
  }))
  .filter((g) => g.items.length > 0)
```
