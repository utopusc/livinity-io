import {useState} from 'react'
import {trpcReact} from '@/trpc/trpc'
import {McpServerList} from '@/components/mcp/McpServerList'

import {FEATURED_MCPS} from '@/components/mcp/featured-mcps'
import {SettingsPageLayout} from './_components/settings-page-layout'
import {SettingsPageHeader} from '@/components/settings-page-header'

export default function McpServersPage() {
  const [selectedName, setSelectedName] = useState<string | null>(null)
  const utils = trpcReact.useUtils()

  // Use existing mcp tRPC queries (from mcpRouter in index.ts)
  const serversQ = trpcReact.mcp.listServers.useQuery()
  const statusesQ = trpcReact.mcp.getStatuses.useQuery()
  const toggleMutation = trpcReact.mcp.toggleServer.useMutation({
    onSuccess: () => {
      utils.mcp.listServers.invalidate()
      utils.mcp.getStatuses.invalidate()
    },
  })
  const removeMutation = trpcReact.mcp.removeServer.useMutation({
    onSuccess: () => {
      utils.mcp.listServers.invalidate()
      setSelectedName(null)
    },
  })

  // Adapt tRPC data to component props shape
  const servers = (serversQ.data ?? []).map((s) => ({
    name: s.name,
    config: s,
    status: statusesQ.data?.[s.name],
  }))

  const selectedServer = selectedName
    ? (servers.find((s) => s.name === selectedName) ?? null)
    : null

  const installedNames = new Set(servers.map((s) => s.name))

  const handleInstallFeatured = async (mcp: (typeof FEATURED_MCPS)[number]) => {
    // Use the existing mcpRouter install path
    // (identical to chat sidebar install — no new tRPC routes needed)
    // Navigate to featured install flow — this is already handled inside
    // the install mutation path from the mcpRouter; placeholder here
    // until mcp.installFromRegistry is available (same as mcp-panel.tsx).
  }

  return (
    <SettingsPageLayout title='MCP Servers' description='Manage Model Context Protocol servers' hideHeader>
      <SettingsPageHeader
        eyebrow='10 · MCP'
        title='Model Context Protocol'
        titleAccent='servers.'
        sub='Add, configure, and monitor MCP servers that extend Claude with tools.'
      />
      <div className='h-4' />
      <div className='flex gap-4 px-1 h-full min-h-0'>
        {/* Left: server list */}
        <div className='w-72 shrink-0'>
          <McpServerList
            servers={servers}
            selectedName={selectedName}
            onSelect={setSelectedName}
            onToggleEnabled={(name, enabled) => toggleMutation.mutate({name, enabled})}
            onRemove={(name) => removeMutation.mutate({name})}
            isLoading={serversQ.isLoading}
          />
        </div>

        {/* Right: detail or featured */}
        <div className='flex-1 min-w-0'>
          {selectedServer ? (
            <McpServerDetail
              server={selectedServer}
              onClose={() => setSelectedName(null)}
              onToggleEnabled={(name, enabled) => toggleMutation.mutate({name, enabled})}
            />
          ) : (
            <div className='space-y-3'>
              <p className='text-sm text-white/50 mb-4'>Featured MCP servers — one-click install:</p>
              <div className='grid grid-cols-2 gap-3'>
                {FEATURED_MCPS.slice(0, 6).map((mcp) => (
                  <div
                    key={mcp.name}
                    data-testid={`featured-mcp-${mcp.name}`}
                    className={`rounded-lg p-3 bg-gradient-to-br ${mcp.gradient} border border-white/10 cursor-pointer hover:border-white/20 transition-colors`}
                    onClick={() => !installedNames.has(mcp.name) && handleInstallFeatured(mcp)}
                  >
                    <p className='text-sm font-medium text-white'>{mcp.displayName}</p>
                    <p className='text-xs text-white/60 mt-0.5 line-clamp-2'>{mcp.description}</p>
                    {installedNames.has(mcp.name) && (
                      <span className='text-[10px] text-green-400 mt-1 block'>Installed</span>
                    )}
                  </div>
                ))}
              </div>
            </div>
          )}
        </div>
      </div>
    </SettingsPageLayout>
  )
}
