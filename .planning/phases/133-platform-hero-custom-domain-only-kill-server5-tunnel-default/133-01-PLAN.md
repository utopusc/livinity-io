<objective>
Fix Bug A: wizard's Generate API Key flow does not register the user's
chosen hybrid domain in `custom_domains`. Result: even after a successful
hybrid install, platform UI surfaces (dashboard hero, profile, etc.)
have no record of the user's actual computer URL.

Fix: extend `POST /api/account/api-keys` to accept an optional `domain`
JSON field; when present, INSERT a row into `custom_domains` with
status='pending_dns' alongside the API key mint. Update the wizard's
fetch call in `dashboard-install.html` to send `domain` when in
hybrid mode.
</objective>

```ts
export async function POST(req: NextRequest) {
  const user = await getUser(req);
  if (!user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  // Phase 132-02: emailVerified gate removed (already shipped)

  // ── Phase 133-01: optional hybrid-domain registration ─────────
  let body: any = {};
  try { body = await req.json(); } catch {} // tolerate missing body
  const domain = (body?.domain ?? "").trim();

  // Generate fresh key (unchanged)
  const rawKey = `liv_k_${nanoid(20)}`;
  // ... existing key persistence ...

  if (domain) {
    // Duplicate-safe insert: ignore if (user_id, domain) already exists.
    // Schema has UNIQUE on `domain` globally, so we use INSERT ... ON CONFLICT (domain) DO NOTHING.
    try {
      await db.execute(sql`
        INSERT INTO custom_domains (user_id, domain, verification_token)
        VALUES (${user.id}, ${domain}, ${nanoid(32)})
        ON CONFLICT (domain) DO NOTHING
      `);
    } catch (e) {
      // Non-fatal — key mint still returns; domain registration is best-effort
      console.error("[133-01] custom_domains INSERT failed:", e);
    }
  }

  return NextResponse.json({ key: rawKey, /* ... */ });
}
```
