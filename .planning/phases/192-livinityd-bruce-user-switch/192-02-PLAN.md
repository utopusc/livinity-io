    ```bash
    #!/usr/bin/env bash
    # Phase 192-02 — Migrate livinityd from User=root to User=bruce.
    # Idempotent: marker /opt/livos/data/.bruce-migrated guards re-runs.
    # See .planning/phases/192-livinityd-bruce-user-switch/192-02-PLAN.md
    set -uo pipefail

    # ── Test affordance ─────────────────────────────────────────────────
    # TEST_ROOT lets CI run this script against a tmpdir without touching
    # the real filesystem. DRY_RUN=1 prints actions without side-effects.
    LIVOS_ROOT="${TEST_ROOT:-/opt/livos}"
    LIVOS_DATA_DIR="$LIVOS_ROOT/data"
    LIVOS_ENV_FILE="$LIVOS_ROOT/.env"
    MARKER="$LIVOS_DATA_DIR/.bruce-migrated"
    REPO_ROOT="${REPO_ROOT:-$(cd "$(dirname "$0")/.." && pwd)}"
    SUDOERS_SRC="$REPO_ROOT/scripts/install/sudoers.d/livinityd"
    SUDOERS_DEST="${SUDOERS_DEST:-/etc/sudoers.d/livinityd}"
    DRY_RUN="${DRY_RUN:-0}"

    log() { echo "[migrate-to-bruce] $*"; }
    do_cmd() {
      if [[ "$DRY_RUN" == "1" ]]; then
        log "DRY_RUN: $*"
      else
        eval "$@"
      fi
    }

    # ── 1. Pre-flight: must run as root ─────────────────────────────────
    if [[ -z "${TEST_ROOT:-}" && "$(id -u)" -ne 0 ]]; then
      log "ERROR: must run as root (or set TEST_ROOT for CI)"
      exit 1
    fi

    # ── 2. Idempotency check ────────────────────────────────────────────
    if [[ -f "$MARKER" ]]; then
      log "marker $MARKER exists — already migrated, exit 0"
      exit 0
    fi

    # ── 3. Ensure bruce user exists ─────────────────────────────────────
    if ! id bruce >/dev/null 2>&1; then
      log "creating bruce user"
      do_cmd "useradd -m -s /bin/bash bruce"
    fi

    # ── 4. Chown /opt/livos/data → bruce:bruce ──────────────────────────
    if [[ -d "$LIVOS_DATA_DIR" ]]; then
      log "chown -R bruce:bruce $LIVOS_DATA_DIR"
      do_cmd "chown -R bruce:bruce $LIVOS_DATA_DIR"
    else
      log "WARN: $LIVOS_DATA_DIR does not exist — skipping data chown"
      do_cmd "mkdir -p $LIVOS_DATA_DIR"
      do_cmd "chown bruce:bruce $LIVOS_DATA_DIR"
    fi

    # ── 5. Chown .env* files ────────────────────────────────────────────
    for f in "$LIVOS_ENV_FILE" "$LIVOS_ENV_FILE.local"; do
      if [[ -f "$f" ]]; then
        log "chown bruce:bruce $f + chmod 0640"
        do_cmd "chown bruce:bruce $f"
        do_cmd "chmod 0640 $f"
      fi
    done

    # ── 6. Add bruce to docker group (if docker installed) ──────────────
    if [[ -S /var/run/docker.sock ]] || command -v docker >/dev/null 2>&1; then
      if ! id -nG bruce 2>/dev/null | grep -qw docker; then
        log "adding bruce to docker group"
        do_cmd "usermod -aG docker bruce"
      else
        log "bruce already in docker group"
      fi
    fi

    # ── 7. Install sudoers fragment ─────────────────────────────────────
    if [[ -f "$SUDOERS_SRC" ]]; then
      log "installing $SUDOERS_SRC → $SUDOERS_DEST (0440 root:root)"
      do_cmd "install -m 0440 -o root -g root $SUDOERS_SRC $SUDOERS_DEST"
      # ── 7b. visudo syntax check (skipped in TEST_ROOT mode) ────────
      if [[ -z "${TEST_ROOT:-}" ]] && command -v visudo >/dev/null 2>&1; then
        if ! visudo -cf "$SUDOERS_DEST" >/dev/null; then
          log "ERROR: visudo syntax check FAILED on $SUDOERS_DEST"
          do_cmd "rm -f $SUDOERS_DEST"
          exit 2
        fi
      fi
    else
      log "WARN: $SUDOERS_SRC missing — skipping sudoers install (192-01 not deployed?)"
    fi

    # ── 8. Write idempotency marker ─────────────────────────────────────
    do_cmd "touch $MARKER"
    do_cmd "chown bruce:bruce $MARKER"

    log "MIGRATION COMPLETE — restart livos.service to apply User=bruce"
    log "  systemctl daemon-reload && systemctl restart livos.service"
    exit 0
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| installer (root) → migration script | deploy-livinityd.sh runs as root; migration script runs ON SAME UID and writes to /etc/sudoers.d/ |
| migration script → /etc/sudoers.d/livinityd | mode 0440 root:root mandatory or sudo refuses to read the file |
| systemd → livinityd process | After cutover, systemd spawns process as bruce (uid 1000); kernel enforces from this point |
| livinityd (bruce) → /opt/livos/data | bruce-owned, full rw |
| livinityd (bruce) → /opt/livos/packages source | root-owned, world-readable (0755/0644), bruce reads as "other" |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-192-02-1 | T (Tampering) | sudoers fragment installed with wrong mode | mitigate | `install -m 0440 -o root -g root` explicit; `visudo -cf` syntax-check before commit; remove file on syntax fail |
| T-192-02-2 | D (DoS) | Migration script run twice corrupts state | mitigate | Idempotency marker `/opt/livos/data/.bruce-migrated` guards re-runs; second invocation exits 0 silently |
