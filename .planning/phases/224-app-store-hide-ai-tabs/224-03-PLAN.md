**File 1: `livos/packages/ui/src/components/banners/v42-migration-banner.tsx`**

```typescript
/**
 * Phase 224 — Liv Assistant migration banner.
 *
 * Shown at the top of App Store + Settings while the v42 migration is
 * active. Tells operators WHY the AI-shaped surfaces are hidden and
 * points them to Liv Assistant (dock icon, installed by Phase 223).
 *
 * Dismissible per-session via `useState` only (NOT localStorage / sessionStorage
 * / Redis). Spec: re-appears on next session so operators don't forget the
 * legacy surfaces are temporarily hidden.
 *
 * Sacred SHA `f3538e1d811992b782a9bb057d1b7f0a0189f95f` UNCHANGED — UI only.
 */
import {useState} from 'react'
import {TbX} from 'react-icons/tb'

export interface V42MigrationBannerProps {
  context: 'app-store' | 'settings'
}

export const V42_MIGRATION_BANNER_TEXT =
  'AI integrations temporarily disabled during Liv Assistant migration. Open Liv Assistant from the dock to use AI features.'

export function V42MigrationBanner({context}: V42MigrationBannerProps) {
  const [dismissed, setDismissed] = useState(false)
  if (dismissed) return null
  return (
    <div
      role='status'
      data-context={context}
      data-testid='v42-migration-banner'
      className='mb-4 flex items-start gap-3 rounded-radius-md border border-line bg-[color:var(--bg-2)] px-4 py-3 text-body-sm text-[color:var(--fg-mute)]'
    >
      <span className='flex-1 leading-relaxed'>{V42_MIGRATION_BANNER_TEXT}</span>
      <button
        type='button'
        onClick={() => setDismissed(true)}
        aria-label='Dismiss banner'
        className='flex h-6 w-6 shrink-0 items-center justify-center rounded-radius-sm text-[color:var(--fg-faint)] transition-colors hover:bg-[color:var(--bg)] hover:text-[color:var(--fg)]'
      >
        <TbX className='h-3.5 w-3.5' />
      </button>
    </div>
  )
}
```

```typescript
import {describe, it, expect} from 'vitest'
import {render, screen, fireEvent} from '@testing-library/react'
import {V42MigrationBanner, V42_MIGRATION_BANNER_TEXT} from './v42-migration-banner'

describe('V42MigrationBanner', () => {
  it('renders the migration text in app-store context', () => {
    render(<V42MigrationBanner context='app-store' />)
    expect(screen.getByText(V42_MIGRATION_BANNER_TEXT)).toBeTruthy()
  })

  it('renders the migration text in settings context', () => {
    render(<V42MigrationBanner context='settings' />)
    expect(screen.getByText(V42_MIGRATION_BANNER_TEXT)).toBeTruthy()
  })

  it('hides itself when the dismiss button is clicked', () => {
    render(<V42MigrationBanner context='app-store' />)
    expect(screen.queryByText(V42_MIGRATION_BANNER_TEXT)).toBeTruthy()
    fireEvent.click(screen.getByLabelText('Dismiss banner'))
    expect(screen.queryByText(V42_MIGRATION_BANNER_TEXT)).toBeNull()
  })

  it('dismissal is per-instance (no persistence across re-mounts)', () => {
    const {unmount} = render(<V42MigrationBanner context='settings' />)
    fireEvent.click(screen.getByLabelText('Dismiss banner'))
    expect(screen.queryByText(V42_MIGRATION_BANNER_TEXT)).toBeNull()
    unmount()
    // Fresh mount → banner re-appears (no localStorage / sessionStorage).
    render(<V42MigrationBanner context='settings' />)
    expect(screen.getByText(V42_MIGRATION_BANNER_TEXT)).toBeTruthy()
  })
})
```
