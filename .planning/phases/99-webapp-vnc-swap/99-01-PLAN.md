```
"/c/Windows/System32/OpenSSH/ssh.exe" -i "C:/Users/hello/Desktop/Projects/contabo/pem/minipc" -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null bruce@10.69.31.68 'set -e
   echo "=== STEP 1: confirm session_type ==="
   echo "session_type=$XDG_SESSION_TYPE"
   if [ "$XDG_SESSION_TYPE" != "x11" ]; then
     echo "FAIL: session is $XDG_SESSION_TYPE, expected x11"
     exit 1
   fi

   echo "=== STEP 2: spawn fresh Chrome window ==="
   sudo -n -u bruce DISPLAY=:0 XAUTHORITY=/run/user/1000/gdm/Xauthority \
     google-chrome --new-window about:blank >/dev/null 2>&1 &
   CHROME_PARENT_PID=$!
   sleep 2

   echo "=== STEP 3: discover wid via xdotool ==="
   WID=$(sudo -n -u bruce DISPLAY=:0 XAUTHORITY=/run/user/1000/gdm/Xauthority \
     xdotool search --name "New Tab" 2>/dev/null | tail -1)
   if [ -z "$WID" ]; then
     # Fallback — match on "about:blank"
     WID=$(sudo -n -u bruce DISPLAY=:0 XAUTHORITY=/run/user/1000/gdm/Xauthority \
       xdotool search --name "about:blank" 2>/dev/null | tail -1)
   fi
   if [ -z "$WID" ]; then
     echo "FAIL: no Chrome window discovered"
     pkill -f "google-chrome.*about:blank" 2>/dev/null || true
     exit 1
   fi
   WID_HEX=$(printf "0x%x" "$WID")
   echo "wid_dec=$WID wid_hex=$WID_HEX"

   echo "=== STEP 4: spawn x11vnc on port 15999 ==="
   sudo -n -u bruce DISPLAY=:0 XAUTHORITY=/run/user/1000/gdm/Xauthority \
     /usr/bin/x11vnc -id "$WID_HEX" -rfbport 15999 -localhost -shared -forever -noxdamage -nopw \
     >/tmp/p99-x11vnc.log 2>&1 &
   X11VNC_PID=$!
   sleep 1.5

   echo "=== STEP 5: verify RFB handshake (expect: RFB 003.008\\n) ==="
   HANDSHAKE_HEX=$(timeout 1 nc 127.0.0.1 15999 | head -c 12 | xxd -p | head -1 || echo "FAIL")
   echo "handshake_hex=$HANDSHAKE_HEX"
   # 524642 = "RFB"; 30 30 33 = "003"; 2e = "."; 30 30 38 = "008"; 0a = newline
   # Full expected: 524642 20 303033 2e 303038 0a -> "524642203030332e3030380a"

   echo "=== STEP 6: verify x11vnc still alive ==="
   if kill -0 "$X11VNC_PID" 2>/dev/null; then
     echo "x11vnc=alive pid=$X11VNC_PID"
   else
     echo "x11vnc=dead pid=$X11VNC_PID"
   fi

   echo "=== STEP 7: stderr tail ==="
   echo "--- /tmp/p99-x11vnc.log (last 30 lines) ---"
   tail -30 /tmp/p99-x11vnc.log || true

   echo "=== STEP 8: cleanup ==="
   kill -TERM "$X11VNC_PID" 2>/dev/null || true
   pkill -f "google-chrome.*about:blank" 2>/dev/null || true
   echo "=== DONE ==="
'
```
