    ```typescript
    // Phase 177-01 — per-Agent cron schedule registry.
    // ADDITIVE to Phase 164 autonomous-scheduler — scheduler.ts NOT touched.
    // Reads AgentItem.schedule (Phase 171 types) and registers node-cron tasks.
    // Sacred SHA f3538e1d... UNCHANGED. vault-items/index.ts barrel is sacred —
    // exports flow through autonomous-scheduler/index.ts instead.

    import * as cron from 'node-cron'
    import type {ItemStore} from './item-store.js'

    export interface BootSweepResult {
      registered: number
      skipped: number
    }

    export class AgentScheduleRegistry {
      readonly tasks = new Map<string, cron.ScheduledTask>()
    }

    export function scheduleAgent(
      agentId: string,
      cronExpr: string,
      callback: () => void,
      registry: AgentScheduleRegistry,
    ): {ok: true} | {ok: false; reason: 'invalid_cron'} {
      if (!cron.validate(cronExpr)) {
        return {ok: false, reason: 'invalid_cron'}
      }
      // Idempotent: stop existing task before re-registering
      const existing = registry.tasks.get(agentId)
      if (existing) {
        try { existing.stop() } catch { /* swallow */ }
      }
      const task = cron.schedule(cronExpr, callback)
      registry.tasks.set(agentId, task)
      return {ok: true}
    }

    export function unscheduleAgent(
      agentId: string,
      registry: AgentScheduleRegistry,
    ): {ok: true} | {ok: false; reason: 'not_found'} {
      const task = registry.tasks.get(agentId)
      if (!task) return {ok: false, reason: 'not_found'}
      try { task.stop() } catch { /* swallow */ }
      registry.tasks.delete(agentId)
      return {ok: true}
    }

    export async function bootSweepAgentSchedules(
      store: ItemStore,
      registry: AgentScheduleRegistry,
      runFn: (agentId: string) => void,
    ): Promise<BootSweepResult> {
      const items = await store.list({archived: false})
      let registered = 0
      let skipped = 0
      for (const item of items) {
        if (item.type !== 'agent') continue
        const schedule = item.schedule
        if (!schedule || schedule.trim() === '') {
          skipped++
          continue
        }
        const result = scheduleAgent(item.id, schedule, () => runFn(item.id), registry)
        if (result.ok) {
          registered++
        } else {
          skipped++
        }
      }
      return {registered, skipped}
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| ItemStore → AgentScheduleRegistry | `item.json.schedule` is user-written cron string — must be validated before passing to node-cron |
| Registry → node-cron | cron.schedule() accepts any string; invalid expressions can cause silent NaN timing or throw |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-177-01-01 | Tampering | AgentItem.schedule | mitigate | `cron.validate()` call in `scheduleAgent` before registering; returns `{ok: false, reason: 'invalid_cron'}` — bad strings never reach cron.schedule() |
| T-177-01-02 | Denial of Service | bootSweepAgentSchedules | mitigate | Skip (count as `skipped`) any agent with invalid cron string — one bad agent cannot block the rest from scheduling |
| T-177-01-03 | Elevation of Privilege | node-cron callback | accept | Callbacks run inside livinityd process which already has vault write access; no additional privilege gained by cron registration |
</threat_model>
