<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| UI → tRPC chromeMaster.* | Authenticated session must be admin role (RBAC). |
| tRPC → spawn chrome on :0 | Chrome runs under `bruce` with privileged user-data-dir at /opt/livos/data/. |
| Chrome session → Google OAuth | TLS to accounts.google.com; OAuth state persisted to master dir. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-102-07 | Elevation of Privilege | `chromeMaster.startLogin` (spawn chrome with privileged --user-data-dir) | mitigate | adminProcedure gate at the tRPC level. The procedure throws UNAUTHORIZED for non-admin sessions BEFORE spawn(). Test asserts the gate via a non-admin ctx fixture. |
| T-102-07b | Tampering (concurrent master spawns) | currentMaster singleton state | mitigate | startLogin throws if `currentMaster` non-null. Exit watcher resets `currentMaster` on Chrome exit so retry works. |
| T-102-07c | Data Loss (accidental reset) | chromeMaster.reset | mitigate | Default backup=true renames master → master.backup before delete. UI confirms via AlertDialog before invoking. restoreBackup is also adminProcedure. |
| T-102-07d | Information Disclosure (Cookies file content) | status query | accept | Status checks file existence only, not content. Cookies file is bruce-readable; livinityd runs as bruce so access does not escalate privilege. |
</threat_model>

       /**
        * Phase 102-07 — Chrome Master Login tRPC routes.
        *
        * adminProcedure-gated (T-102-07). startLogin spawns google-chrome on
        * bruce's physical :0 display with --user-data-dir=/opt/livos/data/chrome-master.
        * User logs into Google + closes Chrome; master profile inherits the login
        * state. Subsequent WebApp spawns seed from this master.
        *
        * Sacred SHA: f3538e1d811992b782a9bb057d1b7f0a0189f95f — never touched.
        */
       import {z} from 'zod'
       import {TRPCError} from '@trpc/server'
       import {router, adminProcedure, privateProcedure} from '../server/trpc/trpc.js'
       import {spawn as nodeSpawn, type ChildProcess} from 'node:child_process'
       import {access as nodeAccess, rm as nodeRm, rename as nodeRename, mkdir as nodeMkdir} from 'node:fs/promises'
       import {constants as fsConstants} from 'node:fs'

       export const MASTER_PROFILE_DIR = '/opt/livos/data/chrome-master'
       export const MASTER_BACKUP_DIR = '/opt/livos/data/chrome-master.backup'
       const COOKIES_PATH = `${MASTER_PROFILE_DIR}/Default/Cookies`

       export interface MasterLoginInjectables {
         spawnFn?: typeof nodeSpawn
         accessFn?: typeof nodeAccess
         rmFn?: typeof nodeRm
         renameFn?: typeof nodeRename
         mkdirFn?: typeof nodeMkdir
       }

       // Module-singleton state (per livinityd boot)
       let currentMaster: {pid: number; child: ChildProcess; startedAt: number} | null = null

       export function createChromeMasterRouter(injectables: MasterLoginInjectables = {}) {
         const spawnFn = injectables.spawnFn ?? nodeSpawn
         const accessFn = injectables.accessFn ?? nodeAccess
         const rmFn = injectables.rmFn ?? nodeRm
         const renameFn = injectables.renameFn ?? nodeRename
         const mkdirFn = injectables.mkdirFn ?? nodeMkdir

         return router({
           status: privateProcedure.query(async () => {
             let hasCookies = false
             try {
               await accessFn(COOKIES_PATH, fsConstants.R_OK)
               hasCookies = true
             } catch { /* not logged in */ }
             return {
               hasCookies,
               dir: MASTER_PROFILE_DIR,
               running: !!currentMaster,
               pid: currentMaster?.pid,
               startedAt: currentMaster?.startedAt,
             }
           }),

           startLogin: adminProcedure.mutation(async ({ctx}) => {
             if (currentMaster) {
               throw new TRPCError({code: 'CONFLICT', message: 'master chrome already running; close existing first'})
             }
             const args = [
               '-n', '-u', 'bruce',
               'DISPLAY=:0',
               'google-chrome',
               `--user-data-dir=${MASTER_PROFILE_DIR}`,
               '--no-first-run',
               '--no-default-browser-check',
             ]
             const child = spawnFn('sudo', args, {detached: false, stdio: ['ignore', 'ignore', 'pipe']})
             if (!child.pid) {
               throw new TRPCError({code: 'INTERNAL_SERVER_ERROR', message: 'chrome failed to spawn'})
             }
             currentMaster = {pid: child.pid, child, startedAt: Date.now()}
             child.on('exit', () => { currentMaster = null })
             return {pid: child.pid, startedAt: currentMaster.startedAt}
           }),

           reset: adminProcedure
             .input(z.object({backup: z.boolean().default(true)}))
             .mutation(async ({input}) => {
               if (currentMaster) {
                 throw new TRPCError({code: 'CONFLICT', message: 'master chrome is still running; close it first'})
               }
               if (input.backup) {
                 // Try to backup; if master doesn't exist, this is a no-op
                 try {
                   await accessFn(MASTER_PROFILE_DIR, fsConstants.R_OK)
                   try { await rmFn(MASTER_BACKUP_DIR, {recursive: true, force: true}) } catch {}
                   await renameFn(MASTER_PROFILE_DIR, MASTER_BACKUP_DIR)
                 } catch { /* nothing to back up */ }
               } else {
                 await rmFn(MASTER_PROFILE_DIR, {recursive: true, force: true})
               }
               await mkdirFn(MASTER_PROFILE_DIR, {recursive: true})
               return {ok: true}
             }),

           restoreBackup: adminProcedure.mutation(async () => {
             if (currentMaster) {
               throw new TRPCError({code: 'CONFLICT', message: 'master chrome is still running; close it first'})
             }
             try {
               await accessFn(MASTER_BACKUP_DIR, fsConstants.R_OK)
             } catch {
               throw new TRPCError({code: 'NOT_FOUND', message: 'no backup available'})
             }
             try { await rmFn(MASTER_PROFILE_DIR, {recursive: true, force: true}) } catch {}
             await renameFn(MASTER_BACKUP_DIR, MASTER_PROFILE_DIR)
             return {ok: true}
           }),
         })
       }

       // Default export — used by app-level router composition
       export const chromeMasterRouter = createChromeMasterRouter()
       ```

       /**
        * Phase 102-07 — Master Chrome Login UI affordance.
        *
        * Settings page component. Status indicator + Open Master Chrome + Reset.
        * tRPC routes adminProcedure-gated (T-102-07).
        */
       import {useState} from 'react'
       import {Button} from '@/shadcn-components/ui/button'
       import {Card, CardContent, CardFooter, CardHeader, CardTitle, CardDescription} from '@/shadcn-components/ui/card'
       import {
         AlertDialog,
         AlertDialogAction,
         AlertDialogCancel,
         AlertDialogContent,
         AlertDialogDescription,
         AlertDialogFooter,
         AlertDialogHeader,
         AlertDialogTitle,
         AlertDialogTrigger,
       } from '@/shadcn-components/ui/alert-dialog'
       import {trpcReact} from '@/trpc/trpc'

       export function MasterChromeLogin() {
         const utils = trpcReact.useUtils()
         const status = trpcReact.chromeMaster.status.useQuery(undefined, {refetchInterval: 2000})
         const startMut = trpcReact.chromeMaster.startLogin.useMutation({
           onSuccess: () => utils.chromeMaster.status.invalidate(),
         })
         const resetMut = trpcReact.chromeMaster.reset.useMutation({
           onSuccess: () => utils.chromeMaster.status.invalidate(),
         })
         const [confirmOpen, setConfirmOpen] = useState(false)

         const loggedIn = status.data?.hasCookies ?? false
         const running = status.data?.running ?? false

         return (
           <Card>
             <CardHeader>
               <CardTitle>Chrome Master Login</CardTitle>
               <CardDescription>
                 Log into Google once. All LivOS apps inherit this login.
               </CardDescription>
             </CardHeader>
             <CardContent>
               <div className='space-y-2 text-sm'>
                 <div>
                   Status: <span className={loggedIn ? 'text-green-500' : 'text-muted-foreground'}>
                     {loggedIn ? 'Logged in' : 'Not logged in'}
                   </span>
                 </div>
                 <div>
                   Master Chrome running: <span>{running ? 'yes' : 'no'}</span>
                 </div>
               </div>
             </CardContent>
             <CardFooter className='gap-2'>
               <Button onClick={() => startMut.mutate()} disabled={running || startMut.isPending}>
                 {running ? 'Master Chrome running' : 'Open Master Chrome'}
               </Button>
               <AlertDialog open={confirmOpen} onOpenChange={setConfirmOpen}>
                 <AlertDialogTrigger asChild>
                   <Button variant='destructive' disabled={running}>Reset Master Profile</Button>
                 </AlertDialogTrigger>
                 <AlertDialogContent>
                   <AlertDialogHeader>
                     <AlertDialogTitle>Reset Master Profile?</AlertDialogTitle>
                     <AlertDialogDescription>
                       Your current master profile will be backed up to /opt/livos/data/chrome-master.backup.
                       After reset you must run Master Login again.
                     </AlertDialogDescription>
                   </AlertDialogHeader>
                   <AlertDialogFooter>
                     <AlertDialogCancel>Cancel</AlertDialogCancel>
                     <AlertDialogAction
                       onClick={() => { resetMut.mutate({backup: true}); setConfirmOpen(false) }}
                     >
                       Reset
                     </AlertDialogAction>
                   </AlertDialogFooter>
                 </AlertDialogContent>
               </AlertDialog>
             </CardFooter>
           </Card>
         )
       }
       ```
