<objective>
Build the per-app Chrome subprocess spawner (D-102-PER-APP-CHROME). This is the SelfClaude pattern verified working by the user.

    ```typescript
    /**
     * Phase 102-02 — ChromeProcessSpawner (D-102-PER-APP-CHROME).
     *
     * Spawns a per-app google-chrome subprocess scoped to its own --user-data-dir
     * (singleton-lock isolation) and Xvfb display (DISPLAY=:N). --start-fullscreen
     * + --app=<URL> yields chromeless full-display rendering.
     *
     * Threat T-102-02: url/userDataDir/display are all validated at the gate BEFORE
     * argv interpolation. Invalid input throws ChromeProcessSpawnError; spawn() is
     * NEVER called with attacker-controlled values.
     *
     * Sacred SHA: f3538e1d811992b782a9bb057d1b7f0a0189f95f — never touched.
     */
    import {spawn as nodeSpawn, type ChildProcess, type SpawnOptions} from 'node:child_process'

    const USER_DATA_DIR_RE = /^\/tmp\/livos-chrome-app-[a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12}$/
    const DISPLAY_RE = /^:[1-9][0-9]?$/  // :1..:99

    export class ChromeProcessSpawnError extends Error {
      constructor(public code: 'CHROME_INVALID_URL' | 'CHROME_INVALID_USERDATADIR' | 'CHROME_INVALID_DISPLAY' | 'CHROME_SPAWN_FAILED', msg: string) {
        super(msg)
        this.name = 'ChromeProcessSpawnError'
      }
    }

    export interface ChromeSpawnOpts {
      display: string                  // ':10', ':11', ... (regex validated)
      userDataDir: string              // /tmp/livos-chrome-app-<uuid-v4> (regex validated)
      url: string                      // https://... (URL-validated)
      user?: string                    // default 'bruce'
      chromeBinary?: string            // default 'google-chrome'
      spawnFn?: typeof nodeSpawn
      logger?: {info(m: string): void; warn(m: string, e?: unknown): void; error(m: string, e?: unknown): void}
    }

    export interface ChromeProcessHandle {
      pid: number
      child: ChildProcess
      display: string
      userDataDir: string
      stop(): Promise<void>
    }

    const STATIC_ARGS = [
      '--no-first-run',
      '--no-default-browser-check',
      '--no-sandbox',
      '--start-fullscreen',
      '--disable-features=ChromeWhatsNewUI,TranslateUI,InfoBars',
      '--disable-infobars',
      '--test-type',
    ] as const

    function validateInputs(opts: ChromeSpawnOpts): void {
      // T-102-02 — URL validation
      try {
        const u = new URL(opts.url)
        if (!['http:', 'https:', 'file:'].includes(u.protocol)) {
          throw new ChromeProcessSpawnError('CHROME_INVALID_URL', `unsupported url protocol: ${u.protocol}`)
        }
      } catch (err) {
        if (err instanceof ChromeProcessSpawnError) throw err
        throw new ChromeProcessSpawnError('CHROME_INVALID_URL', `invalid url: ${opts.url}`)
      }
      // T-102-02 — userDataDir regex (must be /tmp/livos-chrome-app-<uuid v4-like>)
      if (!USER_DATA_DIR_RE.test(opts.userDataDir)) {
        throw new ChromeProcessSpawnError('CHROME_INVALID_USERDATADIR', `userDataDir must match ${USER_DATA_DIR_RE}, got: ${opts.userDataDir}`)
      }
      // T-102-02 — display regex (:1..:99)
      if (!DISPLAY_RE.test(opts.display)) {
        throw new ChromeProcessSpawnError('CHROME_INVALID_DISPLAY', `display must match :1..:99, got: ${opts.display}`)
      }
    }

    export async function spawnChromeProcess(opts: ChromeSpawnOpts): Promise<ChromeProcessHandle> {
      validateInputs(opts)
      const user = opts.user ?? 'bruce'
      const bin = opts.chromeBinary ?? 'google-chrome'
      const spawnFn = opts.spawnFn ?? nodeSpawn

      const args = [
        '-n', '-u', user,
        `DISPLAY=${opts.display}`,
        bin,
        `--user-data-dir=${opts.userDataDir}`,
        ...STATIC_ARGS,
        `--app=${opts.url}`,
      ]
      const child = spawnFn('sudo', args, {
        detached: true,
        stdio: ['ignore', 'ignore', 'pipe'],
        env: {...process.env, DISPLAY: opts.display},
      })
      if (!child.pid) {
        throw new ChromeProcessSpawnError('CHROME_SPAWN_FAILED', `spawn returned no pid`)
      }

      // Stderr tail (50 lines) — D-99-07 pattern
      const stderrTail: string[] = []
      child.stderr?.on('data', (chunk: Buffer) => {
        const line = chunk.toString('utf-8').trim()
        if (!line) return
        stderrTail.push(line)
        if (stderrTail.length > 50) stderrTail.shift()
      })
      child.on('exit', (code, signal) => {
        if (code !== 0 && code !== null) {
          const tail = stderrTail.length > 0
            ? `\n--- chrome[${opts.display}] stderr (last ${stderrTail.length}) ---\n${stderrTail.join('\n')}`
            : ''
          opts.logger?.error?.(`chrome[${opts.display}] exited code=${code} signal=${signal}${tail}`)
        }
      })
      try { child.unref?.() } catch {}
      opts.logger?.info?.(`chrome[${opts.display}] spawned pid=${child.pid} userDataDir=${opts.userDataDir} url=${opts.url}`)

      return {
        pid: child.pid,
        child,
        display: opts.display,
        userDataDir: opts.userDataDir,
        stop: () => stopChrome(child, 2000, opts.logger),
      }
    }

    async function stopChrome(child: ChildProcess, graceMs: number, logger?: ChromeSpawnOpts['logger']): Promise<void> {
      try { child.kill('SIGTERM') } catch {}
      const killer = setTimeout(() => {
        try { child.kill('SIGKILL') } catch {}
      }, graceMs)
      await new Promise<void>((resolve) => child.once('exit', () => resolve()))
      clearTimeout(killer)
    }
    ```
