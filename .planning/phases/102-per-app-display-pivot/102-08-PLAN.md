    ```typescript
    async close(webappId: string): Promise<void> {
      const entry = this.active.get(webappId)
      if (!entry) return  // idempotent — no-op for missing entries
      // Eagerly remove from active so concurrent close attempts return early
      this.active.delete(webappId)

      // D-102-CLOSE-LIFECYCLE ordered teardown. All steps wrapped — failures logged
      // but never block subsequent steps.

      // 1. Chrome SIGTERM → 2s grace → SIGKILL
      try { await entry.chromeHandle?.stop() }
      catch (err) { this.logger?.warn?.(`close(${webappId}): chrome stop failed`, err) }

      // 2. x11vnc (via stream-manager — also releases port internally)
      try { await this.streamManager.stopStream(entry.streamId) }
      catch (err) { this.logger?.warn?.(`close(${webappId}): stream stop failed`, err) }

      // 3. Xvfb
      try { await entry.xvfbHandle?.stop() }
      catch (err) { this.logger?.warn?.(`close(${webappId}): xvfb stop failed`, err) }

      // 4. /tmp profile cleanup
      if (entry.profileUuid) {
        try { await this.profileSeeder.cleanup(entry.profileUuid) }
        catch (err) { this.logger?.warn?.(`close(${webappId}): profile cleanup failed`, err) }
      }

      // 5. release display
      try { this.displayAllocator.release(entry.displayN) }
      catch (err) { this.logger?.warn?.(`close(${webappId}): display release failed`, err) }

      // 6. release port (stream-manager already does this; second call is no-op per port-allocator contract)
      try { if (typeof entry.port === 'number') this.portAllocator.release(entry.port) }
      catch (err) { this.logger?.warn?.(`close(${webappId}): port release failed`, err) }

      // 7. unregister Luse MCP
      try { await this.unregisterWebAppMcp(webappId) }
      catch (err) { this.logger?.warn?.(`close(${webappId}): mcp unregister failed`, err) }

      this.logger?.info?.(`close(${webappId}): complete`)
    }
    ```

       export interface CloseNativeOpts {
         id: string
         active: Map<string, {display: string; displayN: number; port: number; streamId: string; xvfb: XvfbHandle; child: ChildProcess}>
         displayAllocator: DisplayAllocator
         portAllocator: PortAllocator
         streamManager: {stopStream(id: string): Promise<void>}
         logger?: any
       }
       export async function closeNativeApp(opts: CloseNativeOpts): Promise<void> {
         const entry = opts.active.get(opts.id)
         if (!entry) return  // idempotent
         opts.active.delete(opts.id)

         // SIGTERM child binary → 2s grace → SIGKILL
         try { entry.child.kill('SIGTERM') } catch {}
         const killer = setTimeout(() => { try { entry.child.kill('SIGKILL') } catch {} }, 2000)
         try { await new Promise<void>((r) => entry.child.once('exit', () => r())) } catch {}
         clearTimeout(killer)

         try { await opts.streamManager.stopStream(entry.streamId) }
         catch (err) { opts.logger?.warn?.(`closeNative(${opts.id}): stream stop failed`, err) }

         try { await entry.xvfb.stop() }
         catch (err) { opts.logger?.warn?.(`closeNative(${opts.id}): xvfb stop failed`, err) }

         try { opts.displayAllocator.release(entry.displayN) } catch {}
         try { opts.portAllocator.release(entry.port) } catch {}

         opts.logger?.info?.(`closeNative(${opts.id}): complete`)
       }
