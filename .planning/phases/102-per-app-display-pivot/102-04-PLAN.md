```typescript
async spawn(opts: SpawnOpts): Promise<SpawnResult> {
  // 1-2. Idempotency + per-user cap (unchanged)
  // 3. display = displayAllocator.allocate() // returns NUMBER post-102-01
  let xvfb, chrome, port, seed = null
  try {
    xvfb = await spawnXvfb({display: `:${displayN}`, width: 1280, height: 720, logger})
    seed = await profileSeeder.seed({uuid: opts.webappId})  // or randomUUID
    chrome = await spawnChromeProcess({display: `:${displayN}`, userDataDir: seed.appDir, url: opts.url, logger})
    port = portAllocator.allocate()
    const stream = await streamManager.startStream({mode: 'vnc-window', target: {display: `:${displayN}`}, port, ...})
    await registerWebAppMcp(webappId, `:${displayN}`)
    active.set(webappId, {webappId, displayN, display: `:${displayN}`, xvfb, chrome, port, profileUuid: seed.uuid, streamId})
    return {streamId, wsUrl, display: `:${displayN}`, port}
  } catch (err) {
    if (chrome) await chrome.stop().catch(() => {})
    if (seed) await profileSeeder.cleanup(seed.uuid).catch(() => {})
    if (xvfb) await xvfb.stop().catch(() => {})
    if (port !== null) portAllocator.release(port)
    displayAllocator.release(displayN)
    throw err
  }
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| caller (tRPC) → WebAppWindowManager.spawn() | Caller (UI dock icon click) supplies `url` from webapp config (DB-stored). |
| spawn() → child primitives | Inputs forwarded to validated spawners (102-02 ChromeProcessSpawner enforces argv validation; 102-03 ProfileSeeder enforces uuid validation). |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-102-04 | Tampering (downstream) | WebAppWindowManager.spawn(opts.url) | mitigate (delegated) | URL is forwarded to `spawnChromeProcess({url})` which validates via `new URL(url)` (T-102-02). webappId is forwarded to `profileSeeder.seed({uuid})` which validates against UUID_RE (T-102-03). No additional gate needed at this layer. |
| T-102-04b | Resource exhaustion (display + port pool) | DisplayAllocator + PortAllocator | mitigate | Both allocators throw on exhaustion. spawn() compensating cleanup releases on partial failure. Per-user cap (existing in window-manager) limits concurrent spawns. |

    async spawn(opts: SpawnOpts): Promise<SpawnResult> {
      // 1. Idempotency check (preserve existing logic)
      const existing = this.active.get(opts.webappId)
      if (existing) {
        return {webappId: opts.webappId, streamId: existing.streamId, wsUrl: existing.wsUrl, ...}
      }
      // 2. Per-user cap check (preserve existing logic)
      this.enforceCap(opts.userId)

      // 3. Allocate display
      const displayN = this.displayAllocator.allocate()
      const display = `:${displayN}`

      let xvfb: XvfbHandle | null = null
      let chrome: ChromeProcessHandle | null = null
      let port: number | null = null
      let seed: {uuid: string; appDir: string} | null = null
      let streamId: string | null = null
      let wsUrl: string | null = null

      try {
        // 4. Spawn Xvfb with readiness poll
        xvfb = await this.xvfbSpawnFn({display, width: 1280, height: 720, logger: this.logger})

        // 5. (A2) Optional fluxbox per-display
        if (this.withWindowManager) {
          // Reuse existing fluxbox-wm.ts spawn pattern; non-fatal if it fails
          try {
            const {spawnFluxbox} = await import('./fluxbox-wm.js')
            await spawnFluxbox({display, logger: this.logger})
          } catch (err) {
            this.logger?.warn?.(`fluxbox spawn on ${display} failed (non-fatal)`, err)
          }
        }

        // 6. Seed master profile (uuid = webappId for traceability)
        seed = await this.profileSeeder.seed({uuid: opts.webappId})

        // 7. Spawn Chrome on :N with seeded profile
        chrome = await this.chromeSpawnFn({
          display,
          userDataDir: seed.appDir,
          url: opts.url,
          logger: this.logger,
        })

        // 8. Allocate port
        port = this.portAllocator.allocate()

        // 9. Start x11vnc whole-display capture (stream-manager already supports {display})
        const stream = await this.streamManager.startStream({
          userId: opts.userId,
          mode: 'vnc-window',
          target: {display},
          port,
        })
        streamId = stream.streamId
        wsUrl = stream.wsUrl

        // 10. Register per-WebApp Luse MCP (env LUSE_TARGET_DISPLAY=:N) — uses existing helper
        await this.registerWebAppMcp(opts.webappId, display).catch((err) => {
          this.logger?.warn?.('registerWebAppMcp failed (non-fatal)', err)
        })

        this.active.set(opts.webappId, {
          webappId: opts.webappId,
          userId: opts.userId,
          url: opts.url,
          displayN,
          display,
          xvfbHandle: xvfb,
          chromeHandle: chrome,
          profileUuid: seed.uuid,

          wsUrl,
          mode: 'vnc-window',
          // legacy fields preserved as 0/'' for v33 idle-cleanup poller compat:
          wid: 0,
          targetId: '',
          portalSession: null,
          geometryTracker: null,

        if (port !== null) this.portAllocator.release(port)
        this.displayAllocator.release(displayN)
        throw err
      }
    }
