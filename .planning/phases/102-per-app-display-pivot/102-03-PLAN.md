<objective>
Build the master Chrome profile seeder (D-102-MASTER-PROFILE-SEED). At every WebApp spawn, `cp -r /opt/livos/data/chrome-master /tmp/livos-chrome-app-<uuid>` produces an isolated per-app profile that inherits the master's Google login state. After cp, explicitly delete `SingletonLock`/`SingletonCookie`/`SingletonSocket` from the new profile (A7 empirical-risk mitigation) so the per-app Chrome doesn't refuse-to-start citing the master's stale PID.

Purpose: Solve the "same Google login across apps" problem without sharing `--user-data-dir` (which would force Chrome's singleton-lock IPC merge — the bug that motivated this whole phase pivot).

Output: NEW `chrome-master/profile-seeder.ts` + tests + livinityd boot wire (ensure master dir + boot-time orphan sweep).
</objective>

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| caller → seedProfile | Caller supplies UUID (or seeder generates one via randomUUID). UUID is interpolated into a filesystem path. |
| livinityd → execFile('cp', ...) | Filesystem writes under /tmp/. Master dir read /opt/livos/data/chrome-master/. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-102-03 | Tampering (path traversal) | `seedProfile(uuid)` interpolating into `/tmp/livos-chrome-app-<uuid>` | mitigate | UUID generation uses Node's `randomUUID()` (cryptographically random, RFC 4122 v4). If caller passes a custom `uuid` opt, validate against `/^[a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12}$/` BEFORE interpolation. `masterDir` is constant `/opt/livos/data/chrome-master` (no caller input). Reject non-UUID with `ProfileSeederInputError`. |
| T-102-03b | Information Disclosure (master profile readable by per-app Chrome) | seeded /tmp dir contains Cookies + Login Data from master | accept | This is the design (D-102-MASTER-PROFILE-SEED — inherit login state). Master profile auth tokens are scoped to bruce's Google account; same account is shared across all WebApps under the same logical user. |
| T-102-03c | Denial of Service (disk fill from orphan /tmp dirs) | crashed livinityd leaves /tmp/livos-chrome-app-* dirs | mitigate | Boot-time `sweepOrphans()` removes all leftover dirs. Per-app close `cleanup(uuid)` removes on graceful exit. |

    ```typescript
    /**
     * Phase 102-03 — MasterProfileSeeder (D-102-MASTER-PROFILE-SEED).
     *
     * cp -r /opt/livos/data/chrome-master → /tmp/livos-chrome-app-<uuid> at every
     * WebApp spawn. One-way inheritance (master writes propagate to new apps; app
     * writes don't propagate back). Post-seed cleanup of SingletonLock/Cookie/
     * Socket (A7 empirical risk) so per-app Chrome doesn't refuse-to-start.
     *
     * A1 risk: `--reflink=auto` requires CoW filesystem (btrfs/xfs). On ext4 it
     * either silently falls back to plain copy (newer coreutils) or errors. We
     * try `--reflink=auto` first; on error, retry plain `cp -r`.
     *
     * T-102-03: Custom `uuid` opts are validated against RFC 4122 v4 regex BEFORE
     * filesystem ops to prevent path traversal.
     *
     * Sacred SHA: f3538e1d811992b782a9bb057d1b7f0a0189f95f — never touched.
     */
    import {execFile} from 'node:child_process'
    import {access, mkdir, rm} from 'node:fs/promises'
    import {constants as fsConstants} from 'node:fs'
    import {randomUUID} from 'node:crypto'
    import {promisify} from 'node:util'

    const execFileP = promisify(execFile)

    export const MASTER_PROFILE_DIR = '/opt/livos/data/chrome-master'
    export const APP_PROFILE_PREFIX = '/tmp/livos-chrome-app-'

    // RFC 4122 v4 UUID
    const UUID_RE = /^[a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12}$/

    export class MasterProfileMissingError extends Error {
      code = 'MASTER_PROFILE_MISSING' as const
      constructor(public masterDir: string) {
        super(`master profile directory does not exist: ${masterDir} — run Settings → Chrome Master Login`)
        this.name = 'MasterProfileMissingError'
      }
    }

    export class ProfileSeederInputError extends Error {
      constructor(public code: 'PROFILE_INVALID_UUID', msg: string) {
        super(msg)
        this.name = 'ProfileSeederInputError'
      }
    }

    type ExecFileFn = typeof execFile
    type AccessFn = typeof access
    type MkdirFn = typeof mkdir

    export interface ProfileSeederOpts {
      masterDir?: string
      appPrefix?: string
      execFileFn?: ExecFileFn
      accessFn?: AccessFn
      mkdirFn?: MkdirFn
      uuidFn?: () => string
      logger?: {info(m: string): void; warn(m: string, e?: unknown): void; error(m: string, e?: unknown): void}
    }

    export interface SeedOpts {
      uuid?: string  // when supplied, MUST match UUID_RE
    }

    export interface ProfileSeederHandle {
      seed(opts?: SeedOpts): Promise<{uuid: string; appDir: string}>
      ensureMasterExists(): Promise<void>
      cleanup(uuid: string): Promise<void>
      sweepOrphans(): Promise<number>
    }

    export function createProfileSeeder(opts: ProfileSeederOpts = {}): ProfileSeederHandle {
      const masterDir = opts.masterDir ?? MASTER_PROFILE_DIR
      const appPrefix = opts.appPrefix ?? APP_PROFILE_PREFIX
      const execFn = opts.execFileFn ?? execFile
      const accFn = opts.accessFn ?? access
      const mkFn = opts.mkdirFn ?? mkdir
      const uuidFn = opts.uuidFn ?? randomUUID
      const execP = promisify(execFn) as (cmd: string, args: string[]) => Promise<{stdout: string; stderr: string}>

      return {
        async ensureMasterExists(): Promise<void> {
          try {
            await accFn(masterDir, fsConstants.R_OK | fsConstants.X_OK)
          } catch {
            await mkFn(masterDir, {recursive: true})
            opts.logger?.info?.(`profile-seeder: created empty master at ${masterDir} (user must run Master Login)`)
          }
        },

        async seed(seedOpts: SeedOpts = {}): Promise<{uuid: string; appDir: string}> {
          // T-102-03 — validate uuid if caller-supplied
          const uuid = seedOpts.uuid ?? uuidFn()
          if (!UUID_RE.test(uuid)) {
            throw new ProfileSeederInputError('PROFILE_INVALID_UUID', `uuid must match RFC 4122 v4: ${uuid}`)
          }
          // Master must exist
          try {
            await accFn(masterDir, fsConstants.R_OK | fsConstants.X_OK)
          } catch {
            throw new MasterProfileMissingError(masterDir)
          }

          const appDir = `${appPrefix}${uuid}`
          const t0 = Date.now()
          // A1 — try CoW first
          try {
            await execP('cp', ['-r', '--reflink=auto', masterDir, appDir])
          } catch (err) {
            opts.logger?.warn?.(`profile-seeder: --reflink=auto failed, falling back to plain cp -r`, err)
            await execP('cp', ['-r', masterDir, appDir])
          }
          // A7 — delete master's Singleton files so per-app Chrome doesn't see master's PID
          try {
            await execP('rm', ['-f',
              `${appDir}/SingletonLock`,
              `${appDir}/SingletonCookie`,
              `${appDir}/SingletonSocket`,
            ])
          } catch (err) {
            opts.logger?.warn?.(`profile-seeder: Singleton cleanup failed (non-fatal)`, err)
          }
          opts.logger?.info?.(`profile-seeder: seeded ${appDir} from ${masterDir} in ${Date.now() - t0}ms`)
          return {uuid, appDir}
        },

        async cleanup(uuid: string): Promise<void> {
          if (!UUID_RE.test(uuid)) {
            // Defensive — caller passed something other than what seed() returned
            opts.logger?.warn?.(`profile-seeder: cleanup got invalid uuid ${uuid} — no-op`)
            return
          }
          try {
            await execP('rm', ['-rf', `${appPrefix}${uuid}`])
            opts.logger?.info?.(`profile-seeder: cleaned ${appPrefix}${uuid}`)
          } catch (err) {
            opts.logger?.warn?.(`profile-seeder: cleanup ${appPrefix}${uuid} failed (non-fatal)`, err)
          }
        },

        async sweepOrphans(): Promise<number> {
          try {
            await execP('sh', ['-c', `rm -rf ${appPrefix}*`])
            opts.logger?.info?.(`profile-seeder: swept orphan profiles matching ${appPrefix}*`)
            return 1
          } catch (err) {
            opts.logger?.warn?.(`profile-seeder: sweep failed`, err)
            return 0
          }
        },
      }
    }
    ```
