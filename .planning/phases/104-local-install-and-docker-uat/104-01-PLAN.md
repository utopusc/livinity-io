```dockerfile
# syntax=docker/dockerfile:1
FROM trfore/docker-ubuntu2404-systemd:latest

ENV DEBIAN_FRONTEND=noninteractive \
    DISPLAY=:0 \
    TZ=Europe/Istanbul

# Base deps for install.sh prereqs + GUI stack + Chrome
# (Caddy/dnsmasq/Node/Postgres/Redis are installed by install.sh at runtime to
#  exercise the real install path — keep the Dockerfile thin.)
RUN apt-get update && apt-get install -y --no-install-recommends \
    curl ca-certificates gnupg2 wget jq dig openssl sudo \
    xvfb fluxbox x11vnc websockify novnc \
    xfonts-base xfonts-100dpi xfonts-75dpi fonts-liberation fonts-noto-color-emoji \
    dbus-x11 procps net-tools iproute2 systemd \
 && wget -q -O - https://dl.google.com/linux/linux_signing_key.pub | gpg --dearmor -o /usr/share/keyrings/google-chrome.gpg \
 && echo 'deb [arch=amd64 signed-by=/usr/share/keyrings/google-chrome.gpg] http://dl.google.com/linux/chrome/deb/ stable main' > /etc/apt/sources.list.d/google-chrome.list \
 && apt-get update && apt-get install -y --no-install-recommends google-chrome-stable \
 && rm -rf /var/lib/apt/lists/*

COPY docker/local-uat/entrypoint.sh /usr/local/bin/livos-uat-entrypoint.sh
RUN chmod +x /usr/local/bin/livos-uat-entrypoint.sh

# Systemd unit that runs the UAT entrypoint after systemd has booted
RUN cat > /etc/systemd/system/livos-uat.service <<'UNIT' \
 && systemctl enable livos-uat.service
[Unit]
Description=LivOS UAT bootstrap (Xvfb + noVNC + Chrome + install.sh)
After=multi-user.target
Wants=multi-user.target

[Service]
Type=simple
ExecStart=/usr/local/bin/livos-uat-entrypoint.sh
Restart=no
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target
UNIT

EXPOSE 80 443 53/udp 6080 9223

# systemd is PID 1 (provided by trfore base image's ENTRYPOINT/CMD).
# Do NOT override ENTRYPOINT — trfore image already sets it to /sbin/init.
```

```bash
#!/usr/bin/env bash
# docker/local-uat/entrypoint.sh
# Runs inside the UAT container as a systemd unit AFTER multi-user.target.
# Source: 104-RESEARCH.md §Example 3 (lines 781-826).

set -euo pipefail

LOG_PREFIX="[livos-uat]"
log()  { echo "${LOG_PREFIX} $*"; }
fail() { echo "${LOG_PREFIX} FATAL: $*" >&2; exit 1; }

# ── Pre-flight: verify cgroup v2 (Pitfall 7) ──
if ! grep -q '^0::' /proc/self/cgroup; then
    fail "container is on cgroup v1 — systemd will fail. Upgrade WSL to >= 2.5.1, or add 'kernelCommandLine=cgroup_no_v1=all systemd.unified_cgroup_hierarchy=1' to .wslconfig."
fi
log "cgroup v2 OK"

# ── Start X stack ──
log "starting Xvfb :0 1280x720x24"
Xvfb :0 -screen 0 1280x720x24 &
sleep 1
log "starting fluxbox"
DISPLAY=:0 fluxbox &
sleep 1
log "starting x11vnc on :5900"
x11vnc -display :0 -nopw -shared -forever -bg -rfbport 5900
log "starting websockify (noVNC) on :6080 -> :5900"
websockify --web=/usr/share/novnc 6080 localhost:5900 &
sleep 1

# ── Run install.sh (or scaffold-only fallback if it doesn't exist yet) ──
INSTALL_SH=/livinity-io/scripts/install.sh
MODE="${LIVOS_UAT_MODE:-local-lan}"
if [[ -f "$INSTALL_SH" ]]; then
    log "running install.sh --mode ${MODE}"
    bash "$INSTALL_SH" --mode "$MODE" || log "install.sh exited non-zero (continuing scaffold smoke test)"
else
    log "SCAFFOLD-ONLY MODE: $INSTALL_SH not present yet (plan 104-02 creates it)"
    log "proceeding to launch Chrome so AC-104-1 + AC-104-13 + AC-104-14 can be smoke-tested"
fi

# ── Launch Chrome with CDP on 0.0.0.0 (D-104-UAT-CDP-BIND) ──
log "launching Chrome on CDP :9223 (--remote-debugging-address=0.0.0.0)"
google-chrome \
    --remote-debugging-port=9223 \
    --remote-debugging-address=0.0.0.0 \
    --user-data-dir=/tmp/uat-chrome \
    --no-sandbox \
    --disable-dev-shm-usage \
    --display=:0 \
    "about:blank" &

# ── Readiness sentinel: write a file the test harness can wait on ──
# walk.mjs polls for this file to know the entrypoint is past Chrome launch.
sleep 3
echo "ready=$(date -u +%FT%TZ)" > /tmp/livos-uat-ready
log "READY: noVNC http://<host>:6080/vnc.html, CDP http://<host>:9223"

# ── Block forever; systemd manages process tree ──
wait
```

```javascript
// docker/local-uat/uat-driver/walk.mjs
// Phase 104 — UAT walk driver. This stub covers AC-104-13 + AC-104-14 only.
// Plan 104-07 extends this with full AC-104-{1,2,4,5,6,7,9,10,11,15} walk.
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { setTimeout as sleep } from 'node:timers/promises';

const CDP_URL = process.env.LIVOS_UAT_CDP_URL ?? 'http://localhost:9223';
const NOVNC_URL = process.env.LIVOS_UAT_NOVNC_URL ?? 'http://localhost:6080/vnc.html';
const READY_TIMEOUT_MS = 60_000;

async function waitForReady(url, label) {
    const start = Date.now();
    while (Date.now() - start < READY_TIMEOUT_MS) {
        try {
            const r = await fetch(url);
            if (r.ok) return r;
        } catch { /* container not up yet */ }
        await sleep(2000);
    }
    throw new Error(`${label} did not become ready within ${READY_TIMEOUT_MS}ms (url=${url})`);
}

test('AC-104-13: Chrome DevTools MCP can reach UAT container CDP on :9223', async () => {
    const r = await waitForReady(`${CDP_URL}/json/version`, 'CDP');
    const body = await r.json();
    assert.match(body.Browser, /^Chrome/, `CDP /json/version Browser=${body.Browser}`);
    assert.ok(body.webSocketDebuggerUrl, 'CDP must expose webSocketDebuggerUrl');
});

test('AC-104-14: noVNC HTML5 client reachable on :6080', async () => {
    const r = await waitForReady(NOVNC_URL, 'noVNC');
    assert.equal(r.status, 200);
    const text = await r.text();
    assert.match(text, /noVNC|<title>/i, 'noVNC index must render');
});
```

| Boundary | Description |
|----------|-------------|
| Host ↔ UAT container | Developer's Windows/Linux host runs Docker; UAT container is `--privileged`. Host should NEVER publish this image to any registry — privileged + bind /sys/fs/cgroup is a local-dev-only stance. |
| UAT container ↔ external network | The container's port 9223 is exposed to the host's loopback (and potentially LAN if Docker default network leaks). Chrome on :9223 has NO authentication. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-104-01-S1 | Spoofing | UAT container shipped to a registry / publicly run | accept | README.md explicitly states "do NOT publish image to any registry". UAT container is `--rm`-style ephemeral. |
| T-104-01-I1 | Information disclosure | Chrome user-data leaking via Docker image layers | mitigate | `--user-data-dir=/tmp/uat-chrome` lives on tmpfs (per compose), not in image layers. README documents. |
| T-104-01-D1 | Denial of service | Container fails to boot on cgroup v1 host | mitigate | Pre-flight `grep -q '^0::' /proc/self/cgroup` in entrypoint.sh; fails loudly with WSL upgrade instructions. |
| T-104-01-E1 | Elevation of privilege | `--privileged` container access to host /sys/fs/cgroup | accept | Local dev environment; sacred-SHA pre-commit hook + git push hooks prevent committing privileged images. Threat is scoped to developer's own machine. |
