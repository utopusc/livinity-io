**Step A — Create `docker/local-uat/uat-driver/lib/chrome-cdp.mjs`:** Thin helpers — no new npm dep. Uses only `fetch` (Node 22+ global) and `child_process.exec`.

```javascript
// docker/local-uat/uat-driver/lib/chrome-cdp.mjs
// Phase 104 plan 104-07 — Chrome DevTools Protocol helpers.
// No third-party deps (D-NO-NEW-DEPS); uses Node 22 globals only.
import {exec} from 'node:child_process';
import {promisify} from 'node:util';
import {setTimeout as sleep} from 'node:timers/promises';

const execAsync = promisify(exec);

/**
 * AC-104-13: probe CDP /json/version endpoint.
 * Returns the Chrome version metadata, or throws.
 */
export async function probeCdpVersion(cdpUrl = 'http://localhost:9223') {
    const r = await fetch(`${cdpUrl}/json/version`);
    if (!r.ok) throw new Error(`CDP /json/version: HTTP ${r.status}`);
    return await r.json();
}

/**
 * AC-104-9 + AC-104-10: navigate Chrome (inside the container) to a URL and
 * capture page errors. Uses container-side `google-chrome --headless --screenshot`
 * via `docker exec` to avoid implementing a full WS-based CDP client.
 * Returns {exitCode, stdout, stderr, screenshotPath}.
 */
export async function navigateAndScreenshot(
    {containerName, url, outputPath},
) {
    // --headless=new for clean capture; --user-data-dir different from the
    // entrypoint's Chrome so we don't fight over the singleton lock.
    const cmd = [
        'docker', 'exec', containerName,
        'google-chrome',
        '--headless=new',
        '--no-sandbox',
        '--disable-dev-shm-usage',
        '--user-data-dir=/tmp/uat-walk-chrome',
        '--ignore-certificate-errors',                // for self-signed CA case
        `--screenshot=${outputPath}`,
        '--window-size=1280,720',
        '--virtual-time-budget=10000',                // wait 10s for page load
        url,
    ].map(a => `'${a.replace(/'/g, "'\\''")}'`).join(' ');

    try {
        const {stdout, stderr} = await execAsync(cmd, {timeout: 30000});
        return {exitCode: 0, stdout, stderr, screenshotPath: outputPath};
    } catch (err) {
        return {
            exitCode: err.code ?? 1,
            stdout: err.stdout ?? '',
            stderr: err.stderr ?? String(err),
            screenshotPath: null,
        };
    }
}

/**
 * AC-104-10: probe a URL via curl from inside the container and capture
 * the HTTP response code + any TLS error.
 */
export async function curlInContainer(
    {containerName, url, cacertPath = null, extraArgs = []},
) {
    const args = [
        'docker', 'exec', containerName,
        'curl', '-sSL', '-o', '/dev/null',
        '-w', '%{http_code}\\n%{ssl_verify_result}\\n%{errormsg}',
        '--max-time', '15',
    ];
    if (cacertPath) args.push('--cacert', cacertPath);
    for (const a of extraArgs) args.push(a);
    args.push(url);
    const cmd = args.map(a => `'${a.replace(/'/g, "'\\''")}'`).join(' ');
    try {
        const {stdout} = await execAsync(cmd, {timeout: 20000});
        const [httpCode, sslResult, errMsg] = stdout.split('\n');
        return {httpCode: Number(httpCode), sslResult, errMsg, ok: stdout};
    } catch (err) {
        return {httpCode: 0, sslResult: 'error', errMsg: String(err)};
    }
}

/** Helper: wait until a TCP host:port is reachable inside the container. */
export async function waitForServiceUp(
    {containerName, port = 443, timeoutMs = 60_000},
) {
    const start = Date.now();
    while (Date.now() - start < timeoutMs) {
        const r = await curlInContainer({
            containerName,
            url: `https://localhost:${port}/`,
            extraArgs: ['--insecure', '--max-time', '2'],
        });
        if (r.httpCode > 0) return r;
        await sleep(2000);
    }
    throw new Error(`Service did not come up on :${port} within ${timeoutMs}ms`);
}
```

**Step B — Create `docker/local-uat/uat-driver/lib/tcpdump-check.mjs`:** Wraps tcpdump for AC-104-15 (D-104-RELAY-ZERO-DATA-PLANE assertion).

```javascript
// docker/local-uat/uat-driver/lib/tcpdump-check.mjs
// Phase 104 plan 104-07 — assert AC-104-15: hybrid mode page load produces ZERO
// Server5 traffic. Runs tcpdump inside the container, snaps the count, generates
// a single curl request, snaps the count again, asserts delta == 0.
import {spawn} from 'node:child_process';
import {setTimeout as sleep} from 'node:timers/promises';

const SERVER5_IP = '45.137.194.102';   // Hard-coded Server5 IP (per project memory)

/**
 * Run `tcpdump host SERVER5_IP` inside the container for `durationMs`, then return
 * the captured packet count.
 */
export async function countServer5PacketsDuring(
    {containerName, durationMs, triggerFn},
) {
    const tcpdumpArgs = [
        'exec', containerName,
        'timeout', String(Math.ceil(durationMs / 1000)),
        'tcpdump', '-i', 'any', '-nn', '-c', '100',
        '--immediate-mode',
        'host', SERVER5_IP,
    ];

    return new Promise((resolve, reject) => {
        const proc = spawn('docker', tcpdumpArgs, {stdio: ['ignore', 'pipe', 'pipe']});
        let captured = '';
        let errors = '';
        proc.stdout.on('data', (d) => { captured += d.toString(); });
        proc.stderr.on('data', (d) => { errors += d.toString(); });

        // Trigger function (e.g., a curl) AFTER tcpdump has started capturing
        sleep(1000).then(() => {
            triggerFn().catch((e) => { errors += `\nTRIGGER ERROR: ${e}`; });
        });

        proc.on('close', (code) => {
            // tcpdump exits with 124 on timeout, 0 on natural end — both OK
            // exit 1 means tcpdump itself errored (permission, interface missing)
            if (code !== 0 && code !== 124) {
                return reject(new Error(`tcpdump exited ${code}: ${errors}`));
            }
            // Count packets: each capture line starts with a timestamp
            const lines = captured.split('\n').filter((l) =>
                /^\d{2}:\d{2}:\d{2}\./.test(l),
            );
            resolve({
                packetCount: lines.length,
                rawOutput: captured,
                stderr: errors,
            });
        });
    });
}
```
