```

echo '--- Git HEAD on server (pre-deploy) ---' &&
cd /opt/livos && git rev-parse HEAD 2>/dev/null || echo 'NOT A GIT CHECKOUT — rsync deployed' &&
echo '--- Service health ---' &&
for svc in livos liv-core liv-worker liv-memory; do
  sudo systemctl is-active \$svc 2>/dev/null || echo \"\$svc: UNKNOWN\"
done &&
echo '--- SystemD restart counts ---' &&
for svc in livos liv-core liv-worker liv-memory; do
  sudo systemctl show \$svc --property=NRestarts 2>/dev/null
done &&
echo '--- Disk + RAM ---' &&
df -h /opt && free -h | head -2 &&
echo '--- Node + claude binary ---' &&
node --version && claude --version 2>/dev/null || which claude &&
echo '--- tmux version ---' &&
tmux -V 2>/dev/null || echo 'tmux: not installed' &&
echo '--- Pre-deploy SHA check ---' &&
cd /opt/livos && sudo sh scripts/check-sacred.sh 2>&1 | tail -5
"
```
