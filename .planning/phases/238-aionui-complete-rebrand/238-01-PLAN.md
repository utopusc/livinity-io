  # "${CURRENT_LINK}/static/assets/<aionui-logo-svg-2>"

if [[ -f "${LOGO_SRC}" ]] && [[ "${#LOGO_TARGETS[@]}" -gt 0 ]]; then

fi

Step 238-B (word-boundary sed) proposed shape (refine regex per Plan 238-02 Section H findings):
```bash
# ---------------------------------------------------------------------------
# Phase 238 Step B — Case-insensitive word-boundary Aion -> Liv sed pass
#
# Phase 234-03's sed pass was case-sensitive and pattern-matched the
# AionUi / aionui-web / aionui compound + lowercase variants. It MISSED
# orphan standalone tokens like `Aion`, `AION`, and `aion` that appear in
# UI copy (banner text, button labels, agent names) without the `Ui`
# suffix or `-web` compound.
#
# Word-boundary regex `\b(Aion|AION|aion)\b` ensures we don't touch
# false-positive tokens like `fashion`, `champion`, `companion`, etc.
# Plan 238-02 Section E pre-validated this against actual on-disk content
# and Section H specifies any refinements (e.g., excluding `aionic` if
# Section E found it).
#
# Idempotency: grep-pre-check for word-boundary matches; skip sed if zero.
# D-V43-APACHE-NOTICE: scope is ${REBRAND_TARGET}=${CURRENT_LINK}/static/,
# LICENSE + NOTICE structurally outside.
# ---------------------------------------------------------------------------
if [[ -d "${REBRAND_TARGET}" ]]; then
  WB_HITS="$(grep -rilE '\b(Aion|AION|aion)\b' "${REBRAND_TARGET}" \
    --include='*.html' --include='*.js' --include='*.css' 2>/dev/null | wc -l)"
  if [[ "${WB_HITS}" -gt 0 ]]; then
    log "Word-boundary rebrand: applying \\b(Aion|AION|aion)\\b -> Liv sed pass on ${WB_HITS} files"
    find "${REBRAND_TARGET}" \( -name '*.html' -o -name '*.js' -o -name '*.css' \) \
         -exec sed -E -i 's/\b(Aion|AION|aion)\b/Liv/g' {} +
    WB_POST="$(grep -rilE '\b(Aion|AION|aion)\b' "${REBRAND_TARGET}" \
      --include='*.html' --include='*.js' --include='*.css' 2>/dev/null | wc -l)"
    if [[ "${WB_POST}" -ne 0 ]]; then
      log "WARN: ${WB_POST} files still contain word-boundary Aion variants after sed pass"
    else
      log "Word-boundary rebrand: all standalone Aion/AION/aion tokens replaced (verified by post-grep)"
    fi
  else
    log "Word-boundary rebrand: no standalone Aion/AION/aion tokens found; skipping"
  fi
else
  log "Word-boundary rebrand: WARN ${REBRAND_TARGET} missing; skipping"
fi
```
