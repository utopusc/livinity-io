  ```ts
  // Phase 160-05 — file-read sandbox. LLM-controlled file read is a
  // jailbreak vector; restrict to user-owned paths only. Allowlist:
  //   /home/<user>/             — user home, read-only
  //   /tmp/luse-*/              — Luse-owned temp workspace
  //   /opt/livos/data/uploads/<userId>/  — user uploads
  // Symlinks resolved via realpath BEFORE allowlist check.
  function isPathAllowed(resolved: string, userSlug: string, userId: string): boolean {
    const allowlist = [
      `/home/${userSlug}/`,
      '/tmp/luse-',  // any /tmp/luse-* directory
      `/opt/livos/data/uploads/${userId}/`,
    ]
    return allowlist.some((prefix) => resolved.startsWith(prefix))
  }

  // Inside registerLuseTools handlers, replace existing computer_read_file with:
  computer_read_file: async (args) => {
    const requestedPath = String(args.path ?? '').trim()
    if (!requestedPath) {
      return {
        content: [{type: 'text', text: 'path is required'}],
        isError: true,
      }
    }

    // Phase 160-05 — sandbox: resolve symlinks then allowlist-check.
    let resolved: string
    try {
      const {promises: fs} = await import('node:fs')
      resolved = await fs.realpath(requestedPath)
    } catch (err) {
      return {
        content: [{type: 'text', text: `path not found or unreadable: ${requestedPath}`}],
        isError: true,
      }
    }

    const userSlug = process.env.LUSE_USER_ID ?? 'admin'
    const userId = process.env.LUSE_USER_ID ?? 'admin'  // for now slug == userId; tighten later when uuid mapping wires in
    if (!isPathAllowed(resolved, userSlug, userId)) {
      // Phase 160-05 — rejection. Include resolved + requested so the LLM
      // sees the symlink target if any (debugging) but never sees content.
      return {
        content: [{
          type: 'text',
          text: `path outside sandbox: requested=${requestedPath} resolved=${resolved} ` +
                `(allowed prefixes: /home/${userSlug}/, /tmp/luse-, /opt/livos/data/uploads/${userId}/)`,
        }],
        isError: true,
      }
    }

    // Phase 160-05 — sandbox passed; proceed with existing read logic.
    // (Existing implementation reading + base64-encoding the file goes here;
    // copy from current handler body, unchanged except for the guard above.)
    // ... existing read + return ...
  },
  ```
