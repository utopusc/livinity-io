### WS protocol envelopes

Client → server (JSON, one envelope per `ws.message`):
```json
{"type": "attach", "sessionId": "<uuid>"}
{"type": "stdin", "data": "<utf-8 string typed by user>"}
{"type": "resize", "cols": 120, "rows": 30}
{"type": "detach"}
```

Server → client (JSON):
```json
{"type": "attached", "session": {"id": "...", "tmuxName": "...", "cwd": "...", "title": "..."}}
{"type": "stdout", "data": "<base64-encoded chunk>"}
{"type": "exited", "code": 0}
{"type": "error", "message": "..."}
```

### Constants

```ts
const MAX_STDIN_BYTES = 1024 * 1024  // 1MB hard cap per stdin envelope
const WS_CLOSE_POLICY = 1008          // RFC 6455 "policy violation" — used for auth/auth-zation failures
const WS_CLOSE_OVERSIZE = 1009        // RFC 6455 "message too big"
```

### Handler factory signature

```ts
import type {WebSocket} from 'ws'
import type {IncomingMessage} from 'http'
import type {CcPtyManager} from './manager.js'
import type {SessionStore} from './session-store.js'

export interface CcPtyWsHandlerOptions {
  manager: CcPtyManager
  store: SessionStore  // for ownership check before attach
  logger: {log: (msg: string) => void; error: (msg: string, err?: unknown) => void}
  /**
   * Auth resolver — given the upgrade request, returns the authenticated user.
   * Implementation mirrors livos/packages/livinityd/source/modules/server/ws-agent.ts:
   *   const url = new URL(req.url ?? '', 'http://x')
   *   const token = url.searchParams.get('token')
   *              ?? req.headers['cookie']?.match(/livos_jwt=([^;]+)/)?.[1]
   *              ?? req.headers['authorization']?.replace(/^Bearer /, '')
   *   const payload = JSON.parse(Buffer.from(token.split('.')[1], 'base64').toString())
   *   → return {id: payload.userId ?? 'admin'}
   * Returns null on unauth (handler will close with 1008).
   */
  resolveUser: (req: IncomingMessage) => Promise<{id: string} | null>
}

export function createCcPtyWsHandler(opts: CcPtyWsHandlerOptions): (ws: WebSocket, req: IncomingMessage) => void
```

### Handler flow (pseudocode — implement verbatim)

```ts
return async (ws, req) => {
  const user = await opts.resolveUser(req).catch(() => null)
  if (!user) { ws.close(1008, 'unauthorized'); return }

  let attachHandle: {stdin: (s: string) => void; resize: (c: number, r: number) => void; detach: () => void} | null = null
  let attachedSessionId: string | null = null

  const send = (env: object) => { try { ws.send(JSON.stringify(env)) } catch { /* ws gone */ } }

  ws.on('message', async (raw) => {
    let env: any
    try { env = JSON.parse(raw.toString('utf-8')) } catch { send({type: 'error', message: 'malformed JSON'}); return }
    try {
      if (env.type === 'attach') {
        if (attachHandle) { send({type: 'error', message: 'already attached — detach first'}); return }
        const session = await opts.store.getById(env.sessionId)
        if (!session) { send({type: 'error', message: `session ${env.sessionId} not found`}); return }
        if (session.userId !== user.id) {
          opts.logger.log(`[cc-pty/ws] cross-user attach rejected: user=${user.id} target.userId=${session.userId}`)
          send({type: 'error', message: 'forbidden'})
          ws.close(1008, 'cross-user attach forbidden')
          return
        }
        attachHandle = await opts.manager.attachSession(session.id, (chunk) => {
          send({type: 'stdout', data: chunk.toString('base64')})
        })
        attachedSessionId = session.id
        send({type: 'attached', session: {id: session.id, tmuxName: session.tmuxName, cwd: session.cwd, title: session.title}})
      } else if (env.type === 'stdin') {
        if (!attachHandle) { send({type: 'error', message: 'not attached'}); return }
        const bytes = Buffer.byteLength(env.data ?? '', 'utf-8')
        if (bytes > MAX_STDIN_BYTES) {
          send({type: 'error', message: `stdin frame too large (${bytes} > ${MAX_STDIN_BYTES})`})
          ws.close(1009, 'stdin oversize')
          return
        }
        attachHandle.stdin(env.data)
      } else if (env.type === 'resize') {
        if (!attachHandle) { send({type: 'error', message: 'not attached'}); return }
        const cols = Number(env.cols), rows = Number(env.rows)
        if (!Number.isFinite(cols) || !Number.isFinite(rows) || cols < 1 || rows < 1 || cols > 1000 || rows > 1000) {
          send({type: 'error', message: 'invalid resize dims'}); return
        }
        attachHandle.resize(cols, rows)
      } else if (env.type === 'detach') {
        attachHandle?.detach()
        attachHandle = null
        attachedSessionId = null
      } else {
        send({type: 'error', message: `unknown envelope type: ${env.type}`})
      }
    } catch (err: any) {
      opts.logger.error('[cc-pty/ws] handler error', err)
      send({type: 'error', message: err?.message ?? String(err)})
    }
  })

  ws.on('close', () => {
    attachHandle?.detach()
    attachHandle = null
  })
}
```

// connection attaches to one tmux session by id; tmux OWNS the claude subprocess

  wss.on('connection', createCcPtyWsHandler({

    resolveUser: async (req) => {
      // Mirror /ws/agent pattern verbatim — see ws-agent.ts ~line 382
      try {
        const url = new URL(req.url ?? '', 'http://x')

          url.searchParams.get('token') ??
          req.headers['cookie']?.match(/livos_jwt=([^;]+)/)?.[1] ??
