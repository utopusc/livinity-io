  vaultPath: string
}

export class SessionStore {
  private filePath: string
  private writeQueue: Promise<void> = Promise.resolve()  // single-writer mutex

  constructor(opts: SessionStoreOptions) {
    this.filePath = path.join(opts.vaultPath, '.claude', 'livos-cc-sessions.json')
  }

  async load(): Promise<CcPtySession[]> { /* returns [] if file missing */ }

  async save(sessions: CcPtySession[]): Promise<void> { /* atomic: .tmp + rename */ }

  async getByUser(userId: string): Promise<CcPtySession[]> { /* filter on load */ }

  async getById(id: string): Promise<CcPtySession | null> { /* find on load */ }

  async add(session: CcPtySession): Promise<void> { /* load → push → save */ }

  async update(id: string, patch: Partial<CcPtySession>): Promise<void> { /* load → find → merge → save */ }

Atomic write recipe (mandatory):
```ts
async save(sessions: CcPtySession[]): Promise<void> {
  this.writeQueue = this.writeQueue.then(async () => {
    await fs.mkdir(path.dirname(this.filePath), {recursive: true})
    const tmp = this.filePath + '.tmp'
    const body = JSON.stringify({schemaVersion: 1, sessions}, null, 2)
    await fs.writeFile(tmp, body, 'utf-8')
    await fs.rename(tmp, this.filePath)
  })
  return this.writeQueue
}
```

Load recipe (schemaVersion guard):
```ts
async load(): Promise<CcPtySession[]> {
  try {
    const raw = await fs.readFile(this.filePath, 'utf-8')
    const parsed = JSON.parse(raw) as {schemaVersion?: number; sessions?: CcPtySession[]}
    if (parsed.schemaVersion !== 1) {
      throw new Error(`SessionStore: unsupported schemaVersion ${parsed.schemaVersion} (expected 1)`)
    }
    return parsed.sessions ?? []
  } catch (err: any) {
    if (err.code === 'ENOENT') return []
    throw err
  }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| fs writes inside vault | SessionStore writes `<vaultPath>/.claude/livos-cc-sessions.json`. vaultPath is OS-trusted (livinityd runs as bruce on Mini PC; vault path is owned by bruce). No external attacker writes to this file. |
| schema deserialization | `JSON.parse(load)` reads the on-disk file. If a corrupted/old-version file is on disk, schemaVersion guard rejects it BEFORE any consumer sees malformed data. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-166-02-01 | Tampering | livos-cc-sessions.json mid-save corruption | mitigate | Atomic `.tmp` write + `fs.rename` ensures readers either see the old complete file OR the new complete file — never a torn write. Acceptance criterion: assertion #8 verifies `.tmp` is gone after `save()` resolves. |
| T-166-02-02 | Tampering | Older schemaVersion file on disk after a future migration | mitigate | `load()` throws if `schemaVersion !== 1`. Future migrations will bump schemaVersion AND register a migration shim; for Phase 166 we hard-fail on mismatch. Acceptance criterion: assertion #9 exercises this path. |
| T-166-02-03 | Denial-of-Service | Concurrent writers losing data | mitigate | Single-writer Promise-chain mutex (`writeQueue`) serializes every write through one async tail. Assertion #11 exercises 5 parallel `add()` and asserts no lost rows. Concurrent reads are unsynchronized but fs reads are atomic post-rename. |
| T-166-02-04 | Denial-of-Service | Unbounded session list growth | accept | Phase 166-03 manager enforces cap `maxSessions ?? 10` per user BEFORE calling `store.add()`. SessionStore itself is not the gatekeeper. |
| T-166-02-05 | Information-Disclosure | Session metadata at rest | accept | File lives inside the user's vault dir; vault is bruce-owned (mode 0750). No secret material in the JSON (ccSessionId is not a credential; tmuxName is not sensitive). |
| T-166-02-06 | Elevation-of-Privilege | userId field is attacker-controlled at the manager layer | mitigate | SessionStore stores `userId` as-is; the threat surface is at the MANAGER layer (Plan 166-03) which validates `userId` against `/^[a-zA-Z0-9_-]+$/` BEFORE calling `store.add()`. SessionStore does not need its own userId regex. Cross-ref T-166-03-01. |
