<objective>
Add bounded cleanup for the display lifecycle: a sweep that kills any display whose last app activity (or creation time if no apps were ever attached) is more than 4 hours old. Reuses the Phase 246-05 TTL GC pattern verbatim — DI'd timer + nowFn + idempotent start/stop + drift-locked constants.

Purpose: Without this, an AI agent that creates a display and forgets to kill it leaves Xephyr/Xvfb processes + Redis state lingering. The 4h idle threshold gives the agent enough time for legitimate long-running flows (multi-step UAT walks, batch screenshot runs) while bounding the worst case. The 1h sweep matches the Phase 246 cadence for consistency. The TTL GC impersonates `owner_session` from the display record so it can kill any display it finds — bypass of the user-facing owner-scope is intentional and well-scoped to in-process GC code.

    ```typescript
    import type {DisplayManager} from './types.js'

    export const DISPLAY_TTL_GC_DEFAULT_IDLE_MS = 4 * 60 * 60 * 1000 // 4h
    export const DISPLAY_TTL_GC_DEFAULT_SWEEP_MS = 60 * 60 * 1000 // 1h

    export interface DisplayTtlGcDeps {
      displayManager: DisplayManager
      nowFn?: () => number
      idleMs?: number
      sweepMs?: number
      setIntervalFn?: typeof setInterval
      clearIntervalFn?: typeof clearInterval
      logger?: {info: (msg: string, ctx?: object) => void}
    }

    export interface IdleDisplaySweep {
      start(): void
      stop(): void
      sweepNow(): Promise<number>
    }

    export function createDisplayTtlGc(deps: DisplayTtlGcDeps): IdleDisplaySweep {
      const now = deps.nowFn ?? Date.now
      const idleMs = deps.idleMs ?? DISPLAY_TTL_GC_DEFAULT_IDLE_MS
      const sweepMs = deps.sweepMs ?? DISPLAY_TTL_GC_DEFAULT_SWEEP_MS
      const setIv = deps.setIntervalFn ?? setInterval
      const clearIv = deps.clearIntervalFn ?? clearInterval
      const logger = deps.logger ?? {info: () => {}}
      let handle: ReturnType<typeof setInterval> | null = null

      async function sweepNow(): Promise<number> {
        const cutoff = now() - idleMs
        const records = await deps.displayManager.list()
        let killed = 0
        for (const r of records) {
          const lastIso = r.last_app_at ?? r.created_at
          const lastMs = Date.parse(lastIso) || 0
          if (lastMs < cutoff) {
            try {
              const result = await deps.displayManager.kill({display: r.display, callerSession: r.owner_session})
              if (result.ok) {
                killed++
                logger.info('display-ttl-gc: killed idle display', {
                  display: r.display,
                  idleAgeMs: now() - lastMs,
                  owner_session: r.owner_session,
                })
              }
            } catch {
              // Display vanished between list and kill — best-effort, swallow.
            }
          }
        }
        return killed
      }

      function start(): void {
        if (handle !== null) clearIv(handle)
        handle = setIv(() => { sweepNow().catch(() => {}) }, sweepMs)
        logger.info('display-ttl-gc: started', {sweepMs, idleMs})
      }

      function stop(): void {
        if (handle !== null) {
          clearIv(handle)
          handle = null
        }
      }

      return {start, stop, sweepNow}
    }
    ```
