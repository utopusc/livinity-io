Proposed insertion (per 234-01-INVESTIGATION.md Section I Plan 234-03 spec):
```bash
# ---------------------------------------------------------------------------
# Phase 234-03 — vendored-binary brand rebrand (AionUi -> Liv AI)
#
# Idempotent sed-replace targeting ONLY HTML/JS/CSS files under
# ${CURRENT_LINK}/. EXCLUDES LICENSE + NOTICE files per D-V42-APACHE-NOTICE
# (Apache-2.0 obligation to preserve upstream attribution).
#
# Pre-check guard makes this a no-op on the second and later runs against a
# cmp-stable tarball: we grep for any remaining AionUi/aionui strings in the
# filtered file set; zero matches => skip the sed pass entirely.
#
# Why HTML/JS/CSS only: the upstream "AionUi" brand appears in the React SPA
# bundle text (HTML page titles, JS string literals, CSS comment blocks). It
# does NOT appear in LICENSE/NOTICE in a user-visible way (LICENSE/NOTICE
# files are operator-facing only; their AionUi references are legally
# required attribution and MUST survive).
#
# Why CURRENT_LINK and not VERSION_DIR: the symlink resolves to the same
# tree, but using CURRENT_LINK makes the find traversal stop at the
# aionui-web/ subtree (the symlink target) and not walk into INSTALL_ROOT's
# LICENSE/NOTICE/UPSTREAM.md siblings.
# ---------------------------------------------------------------------------
REBRAND_TARGETS=$(find "${CURRENT_LINK}/" -maxdepth 5 \
  \( -name '*.html' -o -name '*.js' -o -name '*.css' \) \
  -not -name 'LICENSE*' -not -name 'NOTICE*' \
  -not -path '*/LICENSE*' -not -path '*/NOTICE*' 2>/dev/null)

if [[ -z "${REBRAND_TARGETS}" ]]; then
  log "Rebrand: no HTML/JS/CSS files found under ${CURRENT_LINK}/; skipping"
else
  REBRAND_COUNT=$(echo "${REBRAND_TARGETS}" | xargs grep -l -e 'AionUi' -e 'aionui' 2>/dev/null | wc -l)
  if [[ "${REBRAND_COUNT}" -eq 0 ]]; then
    log "Rebrand: AionUi strings already replaced (or absent); skipping"
  else
    log "Rebrand: applying AionUi -> Liv AI / aionui -> liv-ai sed pass on ${REBRAND_COUNT} files"
    # shellcheck disable=SC2086
    echo "${REBRAND_TARGETS}" | xargs sed -i -e 's/AionUi/Liv AI/g' -e 's/aionui/liv-ai/g'
    POST_COUNT=$(echo "${REBRAND_TARGETS}" | xargs grep -lc 'AionUi\|aionui' 2>/dev/null | awk -F: '{s+=$NF} END{print s+0}')
    if [[ "${POST_COUNT}" -ne 0 ]]; then
      log "WARN: ${POST_COUNT} AionUi occurrences remain after sed pass (investigate);"
    else
      log "Rebrand: all AionUi strings replaced (verified by post-grep)"
    fi
  fi
fi

# LICENSE + NOTICE byte-identity check (defensive — sed should never touch them)
for guard in LICENSE NOTICE; do
  if [[ -f "${INSTALL_ROOT}/${guard}" ]]; then
    if grep -q 'AionUi' "${INSTALL_ROOT}/${guard}" 2>/dev/null; then
      log "OK: ${INSTALL_ROOT}/${guard} still contains AionUi attribution (Apache-2.0 preserved)"
    fi
  fi
done
```
