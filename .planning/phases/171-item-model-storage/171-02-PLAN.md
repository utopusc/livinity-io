export interface ItemStoreOptions {
  vaultRoot: string  // absolute path; caller passes resolveVaultRoot() result
}

export interface CreateInput {
  type: 'project' | 'agent' | 'chat'
  name: string
  parentId?: string | null
  // type-specific:
  cwd?: string          // project only
  schedule?: string     // agent only
  ccSessionId?: string  // chat only
}

export interface ListOptions {
  archived?: boolean      // default true (include archived); false hides archived
  parentId?: string | null  // optional filter (null = root-level items)
}

export class ItemStore {
  constructor(opts: ItemStoreOptions)
  create(input: CreateInput): Promise<Item>
  read(id: string): Promise<Item | null>
  update(id: string, patch: Partial<Omit<Item, 'id' | 'type' | 'createdAt' | 'schemaVersion'>>): Promise<Item>
  archive(id: string): Promise<Item>
  unarchive(id: string): Promise<Item>
  delete(id: string): Promise<boolean>  // true if removed, false if missing
  list(opts?: ListOptions): Promise<Item[]>
  // Helpers for plan 171-03 tree-resolver
  itemDir(id: string): string  // <vaultRoot>/items/<id>/
}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| caller (tRPC, future CLI) → ItemStore | `CreateInput.name`, `cwd`, `schedule`, `ccSessionId` are user-controlled strings that may flow into filesystem operations; vaultRoot is set by livinityd boot but `id` flows into `path.join` |
| ItemStore → filesystem | All writes restricted to `<vaultRoot>/items/<id>/`; id is server-generated via newItemId() (nanoid v7 alphabet `[0-9a-z_-]` is filesystem-safe) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-171-02-01 | Tampering | item.json on disk | mitigate | `read()` JSON.parse may throw on corrupt files; current behavior bubbles the throw, NOT silent recovery (zod parse comes in 171-04 — store stays strict). schemaVersion guard rejects future-version files at parse time. |
| T-171-02-02 | Tampering / Path traversal | `id` passed to read/update/delete | mitigate | id is NEVER caller-provided in normal flows — it's returned by `create()`. tRPC router (171-04) validates `z.string().regex(/^[0-9a-z_-]{20,}$/)` before calling. Defense-in-depth: ItemStore rejects ids containing `/`, `\`, `..`, or NUL by passing through `path.join` (which prevents segment escape because the id is always the LAST segment). |
| T-171-02-03 | Information Disclosure | items/ folder permissions | accept | Vault is per-user; Mini PC livinityd runs as root, vault is /root/livinity-vault (pre-173) or /root/liv/ (post-173). Filesystem perms inherit. Multi-user re-scoping is v38 Phase 173 + future work. |
| T-171-02-04 | Denial of Service | concurrent writes | mitigate | writeQueue serializes ALL mutations process-wide (mirror cc-pty/session-store.ts pattern). Concurrent reads do not block. |
| T-171-02-05 | Tampering | partial mid-write crash | mitigate | `.tmp` + `fs.rename` is POSIX-atomic on the same filesystem. Assertion 15 enforces. |
</threat_model>
