       ```ts
       import {Memory} from '@mastra/memory'
       import {PgStore, PgVector} from '@mastra/pg'

       /**
        * Scrub user:password from a postgres:// URL for safe logging.
        * Phase 197-03 T-197-03-02 mitigation.
        */
       export function redactPgUrl(url: string): string {
         // postgres://user:pass@host/db → postgres://***:***@host/db
         // postgres://user@host/db      → postgres://***@host/db
         return url.replace(/(postgres:\/\/)([^@/]+)(@)/, '$1***$3')
                   .replace(/(postgres:\/\/)([^:@/]+):([^@/]+)(@)/, '$1***:***$4')
       }

       export interface LivOSMemoryDeps {
         databaseUrl: string
       }

       /**
        * Phase 197-03 — Construct Mastra Memory backed by PgStore + PgVector.
        *
        * options.semanticRecall.scope === 'thread' is EXPLICIT — Mastra changed
        * the default to 'resource' in March 2026 which would leak conversation
        * context across threads.
        *
        * Errors during construction are re-thrown with the connection URL
        * redacted via redactPgUrl() — never log raw password.
        */
       export function createLivOSMemory(deps: LivOSMemoryDeps): Memory {
         try {
           const storage = new PgStore({connectionString: deps.databaseUrl})
           const vector = new PgVector({
             connectionString: deps.databaseUrl,
             indexConfig: {type: 'hnsw', metric: 'dotproduct'},
           })
           return new Memory({
             storage,
             vector,
             options: {
               lastMessages: 20,
               semanticRecall: {topK: 5, messageRange: 2, scope: 'thread'},
               workingMemory: {enabled: true, scope: 'thread'},
             },
           })
         } catch (err) {
           const redacted = redactPgUrl(deps.databaseUrl)
           throw new Error(
             `Phase 197-03 createLivOSMemory failed for ${redacted}: ${err instanceof Error ? err.message : String(err)}`,
           )
         }
       }
       ```

       ```ts
       import {readFile} from 'node:fs/promises'
       import {fileURLToPath} from 'node:url'
       import * as path from 'node:path'
       import {Client} from 'pg'
       import {redactPgUrl} from './memory.js'

       export interface MigrationResult {
         tablesCreated: number
         alreadyExisted: number
       }

       const MASTRA_TABLES = ['mastra_threads', 'mastra_messages', 'mastra_working_memory', 'mastra_workflow_runs']

       export async function runMastraMigrations(opts: {databaseUrl: string; dryRun?: boolean}): Promise<MigrationResult> {
         const client = new Client({connectionString: opts.databaseUrl})
         try {
           await client.connect()
           // Pre-check which tables exist
           const before = await client.query(
             `SELECT table_name FROM information_schema.tables WHERE table_schema='public' AND table_name = ANY($1)`,
             [MASTRA_TABLES],
           )
           const existedBefore = new Set(before.rows.map(r => r.table_name))
           const alreadyExisted = existedBefore.size

           if (opts.dryRun) return {tablesCreated: 0, alreadyExisted}

           // Run migration
           const sqlPath = path.join(path.dirname(fileURLToPath(import.meta.url)), 'migrations', '001-mastra-tables.sql')
           const sql = await readFile(sqlPath, 'utf-8')
           await client.query(sql)

           // Post-check
           const after = await client.query(
             `SELECT table_name FROM information_schema.tables WHERE table_schema='public' AND table_name = ANY($1)`,
             [MASTRA_TABLES],
           )
           const tablesCreated = after.rows.length - alreadyExisted
           return {tablesCreated, alreadyExisted}
         } catch (err) {
           const redacted = redactPgUrl(opts.databaseUrl)
           throw new Error(
             `Phase 197-03 runMastraMigrations failed for ${redacted}: ${err instanceof Error ? err.message : String(err)}`,
           )
         } finally {
           try { await client.end() } catch {}
         }
       }
       ```
