       // Phase 197-02 — shell_exec tool.
       //
       // Destructive by default: every invocation surfaces as a tool-call-approval
       // chunk in the Mastra agent stream. The operator must explicitly approve
       // before execFile fires.
       //
       // SECURITY (T-197-02-01):
       //   - inputSchema rejects shell metacharacters in `command` (no chaining)
       //   - banned-substring scan rejects /\brm\s+-rf\b/ literal regardless of approval
       //     (defense in depth: even an approved request must not nuke the host)
       //   - execFile (argv array) NOT exec — shell injection surface = 0
       import {createTool} from '@mastra/core/tools'
       import {z} from 'zod'
       import {execFile} from 'node:child_process'
       import {promisify} from 'node:util'

       const execFileP = promisify(execFile)

       const BANNED_PATTERNS = [
         /\brm\s+-rf\s+\//,           // catastrophic rm
         /[;&|`$()]/,                  // shell metacharacters
         /\b(reboot|shutdown|poweroff)\b/, // hard-power
       ] as const

       export class ShellExecAllowlistViolation extends Error {
         readonly code = 'SHELL_EXEC_ALLOWLIST_VIOLATION' as const
         constructor(pattern: string) {
           super(`shell_exec rejected: command matched banned pattern ${pattern}`)
         }
       }

       export const shellExecTool = createTool({
         id: 'shell_exec',
         description:
           'Execute a single shell command on the LivOS host via argv-array execFile (no shell). Destructive — requires operator approval. Use for one-off ops where no specialized tool exists.',
         inputSchema: z.object({
           command: z.string()
             .min(1).max(256)
             .regex(/^[A-Za-z0-9_\-./]+$/, 'command must be a single executable path with no spaces or shell metacharacters'),
           args: z.array(z.string().max(1024)).max(32).default([]),
         }),
         outputSchema: z.object({
           stdout: z.string(),
           stderr: z.string(),
           exitCode: z.number(),
         }),
         requireApproval: true,
         execute: async ({context: input}) => {
           const joined = [input.command, ...(input.args ?? [])].join(' ')
           for (const pat of BANNED_PATTERNS) {
             if (pat.test(joined)) {
               throw new ShellExecAllowlistViolation(pat.source)
             }
           }
           try {
             const {stdout, stderr} = await execFileP(input.command, input.args ?? [], {
               timeout: 30_000,
               maxBuffer: 1024 * 1024,
             })
             return {stdout, stderr, exitCode: 0}
           } catch (err) {
             const e = err as NodeJS.ErrnoException & {stdout?: string; stderr?: string; code?: number}
             return {
               stdout: e.stdout ?? '',
               stderr: e.stderr ?? String(e.message ?? e),
               exitCode: typeof e.code === 'number' ? e.code : 1,
             }
           }
         },
       })
       ```

       // Phase 197-02 — read_file + write_file.
       //
       // Both tools are path-jailed to /home/bruce/livinity/. write_file requires
       // operator approval; read_file does not (read-only).
       //
       // SECURITY (T-197-02-02):
       //   - path.resolve(input).startsWith('/home/bruce/livinity/') enforced
       //   - `..` segments rejected pre-resolve (belt + suspenders)
       //   - symlink target re-checked via fs.realpath inside the jail boundary
       import {createTool} from '@mastra/core/tools'
       import {z} from 'zod'
       import {readFile, writeFile, mkdir, realpath} from 'node:fs/promises'
       import {resolve, dirname} from 'node:path'

       const JAIL_ROOT = '/home/bruce/livinity/'

       export class FilePathJailViolation extends Error {
         readonly code = 'FILE_PATH_JAIL_VIOLATION' as const
         constructor(public path: string) {
           super(`file-tools rejected: path ${path} escapes jail ${JAIL_ROOT}`)
         }
       }

       async function enforceJail(inputPath: string): Promise<string> {
         if (inputPath.includes('..')) throw new FilePathJailViolation(inputPath)
         const resolved = resolve(inputPath)
         if (!resolved.startsWith(JAIL_ROOT)) throw new FilePathJailViolation(inputPath)
         try {
           const real = await realpath(resolved)
           if (!real.startsWith(JAIL_ROOT)) throw new FilePathJailViolation(inputPath)
         } catch (err: any) {
           // ENOENT on write_file is fine — parent will be mkdir'd. Rethrow others.
           if (err.code !== 'ENOENT') throw err
         }
         return resolved
       }

       export const readFileTool = createTool({
         id: 'read_file',
         description:
           'Read a UTF-8 file from /home/bruce/livinity/. Read-only; does not require approval.',
         inputSchema: z.object({path: z.string().min(1).max(4096)}),
         outputSchema: z.object({content: z.string(), byteLength: z.number()}),
         execute: async ({context: input}) => {
           const safe = await enforceJail(input.path)
           const content = await readFile(safe, 'utf8')
           return {content, byteLength: Buffer.byteLength(content, 'utf8')}
         },
       })

       export const writeFileTool = createTool({
         id: 'write_file',
         description:
           'Write a UTF-8 file under /home/bruce/livinity/. Destructive — requires operator approval. Parent directories are created.',
         inputSchema: z.object({path: z.string().min(1).max(4096), content: z.string()}),
         outputSchema: z.object({bytesWritten: z.number()}),
         requireApproval: true,
         execute: async ({context: input}) => {
           const safe = await enforceJail(input.path)
           await mkdir(dirname(safe), {recursive: true})
           await writeFile(safe, input.content, 'utf8')
           return {bytesWritten: Buffer.byteLength(input.content, 'utf8')}
         },
       })
       ```
