       ```ts
       // Phase 197-01 — xAI provider factory for Mastra agents.
       //
       // Every agent.generate() call goes through a custom fetch middleware that:
       //   1. Calls credentialsService.getToken() to obtain a non-expired access token
       //      (XaiCredentialsService.getToken() single-flights the refresh under the hood,
       //       so concurrent agent runs collapse to one refresh HTTP call).
       //   2. Sets Authorization: Bearer <token> on the outgoing request, overwriting any
       //      static value @ai-sdk/xai may have produced from its `apiKey` constructor prop.
       //
       // SECURITY (T-197-01-01): the token NEVER appears in a log line, error message,
       // or stringified value. Errors thrown from getToken() are wrapped so their
       // serialized form contains tokenLength metadata only.
       import {createXai} from '@ai-sdk/xai'
       import type {XaiCredentialsService} from '../../../../../livos/packages/livinityd/source/modules/xai-credentials/index.js'
       // ^ tsconfig path mapping may be required; alternatively, treat XaiCredentialsService as a structural type local to mastra/

       export interface XaiCredentialsLike {
         getToken(): Promise<string>
       }

       export class TokenAcquisitionError extends Error {
         readonly code = 'XAI_TOKEN_ACQUISITION_FAILED' as const
         readonly tokenLength: number
         constructor(message: string, opts: {tokenLength: number; cause?: unknown}) {
           super(message)
           this.tokenLength = opts.tokenLength
           if (opts.cause !== undefined) (this as any).cause = opts.cause
         }
       }

       export interface BuildXaiAgentModelOptions {
         modelId?: string
       }

       export function buildXaiAgentModel(
         creds: XaiCredentialsLike,
         opts: BuildXaiAgentModelOptions = {},
       ) {
         const modelId = opts.modelId ?? 'grok-4.20-0309-non-reasoning'
         const xai = createXai({
           apiKey: 'placeholder-overridden-by-fetch-middleware',
           fetch: async (url, init) => {
             let token: string
             try {
               token = await creds.getToken()
             } catch (err) {
               // Wrap WITHOUT leaking the token (it was never acquired anyway here).
               throw new TokenAcquisitionError(
                 'xAI Bearer token acquisition failed; see cause for credentials-service error code',
                 {tokenLength: 0, cause: err},
               )
             }
             const headers = new Headers(init?.headers)
             headers.set('Authorization', `Bearer ${token}`)
             return globalThis.fetch(url, {...init, headers})
           },
         })
         return xai(modelId)
       }
       ```
