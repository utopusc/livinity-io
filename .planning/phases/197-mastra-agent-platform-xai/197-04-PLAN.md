       // Minimal interface — Plan 197-05's ApprovalManager structurally satisfies this.
       // Plan 197-04 (Wave 2) does NOT import from Plan 197-05 (Wave 3) — only the contract surface lives here.
       export interface ApprovalGate {
         registerPending(toolCallId: string, runId: string): Promise<boolean>
       }

       /**
        * The rejected sentinel returned by a wrapped tool when the operator
        * rejects the approval. This is a NORMAL tool result — Mastra's agent
        * continuation surfaces it as a tool-result chunk; the agent then
        * naturally produces a text reply explaining the rejection.
        *
        * W-02 lock: this is the load-bearing alternative to throwing /
        * aborting the run. The agent stays alive; only this tool call is denied.
        */
       export const REJECTED_TOOL_RESULT = {
         rejected: true as const,
         reason: 'operator rejected this tool call' as const,
       }

       export interface MinimalTool {
         description?: string
         parameters?: unknown
         execute: (input: unknown, ctx: unknown) => Promise<unknown>
       }

       /**
        * Returns a NEW tool object that delegates description + parameters to
        * the original. The execute() function is replaced with a version that
        * first awaits gate.registerPending(toolCallId, runId). On approved=true
        * → delegate to originalTool.execute. On approved=false → return the
        * REJECTED_TOOL_RESULT sentinel.
        *
        * runId is read from ctx if present (Mastra surface — verify at exec
        * time); if absent, falls back to randomUUID() (still works, just no
        * cross-tool grouping for this run).
        */
       export function wrapToolWithApproval(
         originalTool: MinimalTool,
         toolName: string,
         gate: ApprovalGate,
       ): MinimalTool {
         return {
           description: originalTool.description,
           parameters: originalTool.parameters,
           async execute(input: unknown, ctx: unknown): Promise<unknown> {
             const toolCallId = randomUUID()
             // Runtime extraction of runId from ctx — Mastra may put it at ctx.runId
             // or expose it via getRunInfo(). Plan executor verifies at exec time.
             const runId =
               (typeof ctx === 'object' && ctx !== null && 'runId' in ctx && typeof (ctx as {runId?: unknown}).runId === 'string')
                 ? (ctx as {runId: string}).runId
                 : randomUUID()
             const approved = await gate.registerPending(toolCallId, runId)
             if (!approved) {
               return REJECTED_TOOL_RESULT
             }
             return originalTool.execute(input, ctx)
           },
         }
       }

       export function filterMcpTools(rawTools: Record<string, unknown>): Record<string, unknown> {
