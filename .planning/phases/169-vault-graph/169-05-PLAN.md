Create `wireup.test.ts` as an integration test (uses real Express app + real filesystem in OS tmp dir; mocks ONLY the auth middleware).

```ts
import { describe, it, expect, beforeEach, afterEach } from 'vitest';
import express from 'express';
import { mkdtemp, mkdir, writeFile, rm } from 'node:fs/promises';
import path from 'node:path';
import os from 'node:os';
import http from 'node:http';
import { createVaultGraphRouter } from './routes.js';

let tmpDir = '';
let server: http.Server;
let baseUrl = '';

async function startApp(authStub: express.RequestHandler) {
  const app = express();
  app.use(createVaultGraphRouter({ vaultRoot: tmpDir, authMiddleware: authStub }));
  return new Promise<void>((resolve) => {
    server = app.listen(0, () => {
      const addr = server.address();
      if (addr &amp;&amp; typeof addr === 'object') {
        baseUrl = `http://127.0.0.1:${addr.port}`;
      }
      resolve();
    });
  });
}

beforeEach(async () => {
  tmpDir = await mkdtemp(path.join(os.tmpdir(), 'vault-graph-test-'));
  await mkdir(path.join(tmpDir, 'memory'), { recursive: true });
  await writeFile(path.join(tmpDir, 'foo.md'), '# foo\n[[bar]]\n');
  await writeFile(path.join(tmpDir, 'memory', 'bar.md'), '# bar\n');
  await writeFile(path.join(tmpDir, 'memory', 'baz.md'), '# baz\n');
  await writeFile(path.join(tmpDir, '.deleted-old.md'), '# tomb\n');
});

afterEach(async () => {
  await new Promise<void>((r) => server.close(() => r()));
  await rm(tmpDir, { recursive: true, force: true });
});

describe('vault-graph integration', () => {
  it('GET /api/vault/graph returns nodes excluding tombstones', async () => {
    await startApp((_req, _res, next) => next());
    const res = await fetch(`${baseUrl}/api/vault/graph`);
    expect(res.status).toBe(200);
    const body = await res.json();
    expect(body.nodes).toHaveLength(3); // tombstone excluded
    expect(body.truncated).toBe(false);
    expect(body.totalFiles).toBe(3);
  });

  it('GET /api/vault/file returns seeded content', async () => {
    await startApp((_req, _res, next) => next());
    const res = await fetch(`${baseUrl}/api/vault/file?path=foo.md`);
    expect(res.status).toBe(200);
    const body = await res.json();
    expect(body.content).toContain('# foo');
  });

  it('GET /api/vault/file rejects path traversal', async () => {
    await startApp((_req, _res, next) => next());
    const res = await fetch(`${baseUrl}/api/vault/file?path=../etc/passwd`);
    expect(res.status).toBe(400);
  });

  it('Routes are gated by auth middleware', async () => {
    await startApp((_req, res) => {
      res.status(401).json({ error: 'unauthenticated' });
    });
    const res = await fetch(`${baseUrl}/api/vault/graph`);
    expect(res.status).toBe(401);
  });
});
```
