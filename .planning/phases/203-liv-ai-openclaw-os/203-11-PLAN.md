    Create `livos/packages/liv-claw-os/lib/fetch-openui-app.ts`:
    ```ts
    export interface OpenUiApp { slug: string; name: string; content: string; version: number }

    export async function fetchOpenUiApp(slug: string, cookieHeader: string): Promise<OpenUiApp | null> {
      const url = `http://127.0.0.1:8080/trpc/openclawos.apps.get?batch=1&input=${encodeURIComponent(JSON.stringify({0:{json:{slug}}}))}`
      const res = await fetch(url, { headers: { cookie: cookieHeader }, cache: 'no-store' })
      if (!res.ok) return null
      const env = await res.json()
      // tRPC v10/v11 batch envelope
      const result = env?.[0]?.result?.data?.json ?? env?.[0]?.result?.data ?? null
      return result ?? null
    }
    ```
    Server-component fetch only — never client-side (avoids leaking the cookie into the iframe DOM).

  <action>
    Create `livos/packages/liv-claw-os/app/apps/[slug]/page.tsx`:
    ```tsx
    import { cookies } from 'next/headers'
    import { notFound } from 'next/navigation'
    import { fetchOpenUiApp } from '@/lib/fetch-openui-app'
    import { OpenUiRenderer } from '@livos/openui-validator'

    export default async function OpenUiAppPage({ params }: { params: Promise<{ slug: string }> }) {
      const { slug } = await params
      const cookieHeader = (await cookies()).toString()
      const app = await fetchOpenUiApp(slug, cookieHeader)
      if (!app) notFound()
      return (
        <main className="h-screen w-screen overflow-auto bg-background p-4">
          <OpenUiRenderer content={app.content} />
        </main>
      )
    }
    ```
