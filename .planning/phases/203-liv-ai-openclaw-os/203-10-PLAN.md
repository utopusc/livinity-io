  <name>Task 1: desktop-registrar.ts — wrap NativeAppConfigStore for openui apps</name>
  <action>
    Create `livos/packages/livinityd/source/modules/openclawos/desktop-registrar.ts`:
    ```ts
    import { randomUUID } from 'node:crypto'
    import { createHash } from 'node:crypto'
    import { NativeAppConfigStore, type NativeAppConfig } from '../apps/native-app-config.js'

    const OPENUI_ICON_URL = '/liv-ai-app/icons/liv-ai-placeholder.svg'
    const OPENUI_BINARY = '/usr/bin/true'  // synthetic — windows mount via OpenUiAppContent, not via process spawn

    function deterministicUuid(slug: string): string {
      // Stable across boots so re-creating the same slug gives the same uuid (T-203-05 idempotent)
      const h = createHash('sha1').update(`openui-app:${slug}`).digest('hex')
      return `${h.slice(0,8)}-${h.slice(8,12)}-5${h.slice(13,16)}-${h.slice(16,20)}-${h.slice(20,32)}`
    }

    export async function registerOpenUiAppAsDesktopIcon(
      store: NativeAppConfigStore,
      slug: string,
      name: string,
    ): Promise<void> {
      const cfg: NativeAppConfig = {
        id: deterministicUuid(slug),
        name,
        iconUrl: OPENUI_ICON_URL,
        binaryPath: OPENUI_BINARY,
        wmClassHint: `liv-openui-${slug}`.replace(/[^\w-]/g, '-').slice(0, 64),
      }
      await store.upsert(cfg)
    }

    export async function unregisterOpenUiApp(
      store: NativeAppConfigStore,
      slug: string,
    ): Promise<void> {
      await store.delete(deterministicUuid(slug))
    }
    ```

    Idempotency (T-203-05): deterministic uuid means repeated `create` calls for the same slug hit the same Redis key — `NativeAppConfigStore.upsert` is already idempotent.
