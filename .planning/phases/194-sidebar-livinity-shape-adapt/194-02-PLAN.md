```json

Non-agent items (`project`, `chat`) get `mcps: []` and `tools: []` — the file does not exist for them, fallback path returns empty arrays.

           const tools = Array.isArray(parsed?.tools) ? parsed.tools.filter((s: unknown): s is string => typeof s === 'string') : []
           return {mcps, tools}
         } catch {
           return {mcps: [], tools: []}
         }
       }
