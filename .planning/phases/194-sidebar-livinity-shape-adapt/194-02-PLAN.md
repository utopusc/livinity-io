The shape is forward-compatible; Phase 193+ will populate the empty lists from frontmatter.

    ```typescript
    // Phase 194-02 — Inline agent settings panel below SidebarTree.
    //
    // Closes Bug #10 deferred from v38.2. Read-only peek view of agent
    // metadata; the canonical edit surface is AiChatSettingsPanel (v38.2
    // hotfix) — this component deep-links there via onOpenSettings prop.
    //
    // Adapter for Phase 193: today, MCPs + tools lists are not exposed by
    // vault.items.get; we render empty list placeholders ('—') in those
    // slots. Phase 193 will populate them from claude.md frontmatter
    // without component-API change.
    //
    // Sacred SHA f3538e1d811992b782a9bb057d1b7f0a0189f95f UNCHANGED (we
    // do not touch sdk-agent-runner.ts).

    import {Bot, Settings as SettingsIcon} from 'lucide-react'
    import {trpcReact} from '@/trpc/trpc'

    export interface AgentInlineSettingsProps {
      agentId: string
      onOpenSettings: () => void
    }

    export function AgentInlineSettings({agentId, onOpenSettings}: AgentInlineSettingsProps) {
      const query = trpcReact.vault.items.get.useQuery({id: agentId}, {enabled: !!agentId})
      const item = query.data?.item as
        | {id: string; type: string; name: string; schedule?: string}
        | undefined

      return (
        <div
          data-testid='agent-inline-settings-card'
          className='flex flex-col gap-2 border-t border-border px-3 py-3'
        >
          <div className='flex items-center gap-2'>
            <Bot size={14} className='text-accent-blue' />
            <span className='truncate text-sm font-medium'>{item?.name ?? '—'}</span>
          </div>

          <div className='flex flex-col gap-1 text-xs text-text-secondary'>
            <div>
              <span className='font-medium text-text-primary'>MCPs:</span>{' '}
              <span data-testid='agent-inline-mcps'>—</span>
            </div>
            <div>
              <span className='font-medium text-text-primary'>Tools:</span>{' '}
              <span data-testid='agent-inline-tools'>—</span>
            </div>
            <div>
              <span className='font-medium text-text-primary'>Schedule:</span>{' '}
              <span data-testid='agent-inline-schedule'>{item?.schedule ?? '—'}</span>
            </div>
          </div>

          <button
            type='button'
            data-testid='agent-inline-edit-settings'
            onClick={onOpenSettings}
            className='mt-1 flex items-center gap-1 self-start rounded px-2 py-1 text-xs text-accent-blue hover:bg-surface-2'
          >
            <SettingsIcon size={12} />
            Edit in Settings
          </button>
        </div>
      )
    }
    ```
