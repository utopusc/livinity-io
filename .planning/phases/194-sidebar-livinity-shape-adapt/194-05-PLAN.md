       ```typescript
       // Phase 194-05 — slug preview + collision check.
       // Read the existing items list to compare against the typed name's slug.
       const listQuery = trpcReact.vault.items.list.useQuery(undefined, {staleTime: 10_000})
       const existingItems = listQuery.data?.items ?? []

       const trimmedName = name.trim()
       const previewSlug = useMemo(() => slugify(trimmedName), [trimmedName])
       const slugInvalid = trimmedName.length > 0 && previewSlug.length === 0
       const slugCollides = useMemo(() => {

         return existingItems.some((it: {slug?: string; name?: string}) => {
           const otherSlug = (it.slug ?? slugify(it.name ?? '')).toLowerCase()
           return otherSlug === previewSlug.toLowerCase()
         })
       }, [previewSlug, existingItems])
       ```
