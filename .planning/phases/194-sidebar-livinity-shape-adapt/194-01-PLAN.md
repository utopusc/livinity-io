    Create `livos/packages/ui/src/features/sidebar-tree/slug.ts` exporting a single function:
    ```typescript
    // Phase 194-01 — slugify name → kebab-case ascii slug.
    // Used by AddItemModal slug preview (Plan 194-05) and ItemTreeRow fallback.
    // ASCII-folds common Turkish + Latin diacritics so 'Çağrı' → 'cagri'.
    // Output is empty string when input has zero alphanumerics after folding.
    export function slugify(input: string): string {
      if (typeof input !== 'string') return ''
      const fold = input
        .normalize('NFD')
        .replace(/[̀-ͯ]/g, '') // diacritic strip
        // Turkish-specific mappings NFD doesn't catch
        .replace(/[ıİ]/g, 'i')
        .replace(/[şŞ]/g, 's')
        .replace(/[ğĞ]/g, 'g')
        .replace(/[çÇ]/g, 'c')
        .replace(/[öÖ]/g, 'o')
        .replace(/[üÜ]/g, 'u')
        .toLowerCase()
      const ascii = fold.replace(/[^a-z0-9]+/g, '-')
      const trimmed = ascii.replace(/^-+|-+$/g, '')
      return trimmed.slice(0, 64)
    }
    ```
    Create `slug.test.ts` covering the 6 behaviors above plus the empty-string + non-string-input edges.
