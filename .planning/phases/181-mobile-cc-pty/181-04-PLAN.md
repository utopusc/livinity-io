    === 4. MobileBubbleChat.tsx (NEW) ===

    ```tsx
    import { useState, useEffect, useRef, useCallback } from 'react'
    import { CcPtyWsClient } from '@/features/cc-terminal/terminal-ws-client'
    import { JWT_LOCAL_STORAGE_KEY } from '@/modules/auth/shared'
    import { cn } from '@/lib/utils'

    interface Bubble {
      id: string
      role: 'user' | 'assistant'
      text: string
    }

    function wsUrl(): string {
      const base = `${location.protocol === 'https:' ? 'wss:' : 'ws:'}//${location.host}/ws/cc-pty`
      const jwt = typeof localStorage !== 'undefined'
        ? localStorage.getItem(JWT_LOCAL_STORAGE_KEY)
        : null
      return jwt ? `${base}?token=${encodeURIComponent(jwt)}` : base
    }

    export function MobileBubbleChat({ sessionId, className }: { sessionId: string; className?: string }) {
      const [bubbles, setBubbles] = useState<Bubble[]>([])
      const [input, setInput] = useState('')
      const wsRef = useRef<CcPtyWsClient | null>(null)
      const scrollRef = useRef<HTMLDivElement>(null)

      useEffect(() => {
        const ws = new CcPtyWsClient({
          url: wsUrl(),
          sessionId,
          onStdout: (text) => {
            // Strip ANSI escape sequences for bubble display
            const clean = text.replace(/\x1b\[[0-9;]*[mGKHFJABCDsu]/g, '')
            setBubbles((prev) => [
              ...prev,
              { id: `${Date.now()}-${Math.random()}`, role: 'assistant', text: clean }
            ])
          },
          onAttached: () => { /* session ready */ },
          onError: (msg) => {
            setBubbles((prev) => [
              ...prev,
              { id: `err-${Date.now()}`, role: 'assistant', text: `[error] ${msg}` }
            ])
          },
        })
        wsRef.current = ws
        return () => { ws.detach() }
      }, [sessionId])

      // Auto-scroll to bottom on new bubbles
      useEffect(() => {
        scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: 'smooth' })
      }, [bubbles])

      const submit = useCallback(() => {
        const text = input.trim()
        if (!text) return
        wsRef.current?.sendStdin(text + '\r')
        setBubbles((prev) => [
          ...prev,
          { id: `user-${Date.now()}`, role: 'user', text }
        ])
        setInput('')
      }, [input])

      return (
        <div className={cn('flex h-full flex-col bg-bg', className)}>
          {/* Bubble scroll area */}
          <div ref={scrollRef} className='flex-1 overflow-y-auto p-3 space-y-2'>
            {bubbles.map((b) => (
              <div
                key={b.id}
                className={cn(
                  'max-w-[85%] rounded-xl px-3 py-2 text-sm font-mono whitespace-pre-wrap',
                  b.role === 'user'
                    ? 'ml-auto bg-primary text-primary-foreground'
                    : 'mr-auto bg-muted text-foreground'
                )}
                data-role={b.role}
              >
                {b.text}
              </div>
            ))}
          </div>
          {/* Input bar */}
          <div className='flex gap-2 border-t border-border p-2'>
            <textarea
              className='flex-1 resize-none rounded-md bg-input px-3 py-2 text-sm outline-none min-h-[2.5rem] max-h-24'
              placeholder='Type a command...'
              value={input}
              rows={1}
              onChange={(e) => setInput(e.target.value)}
              onKeyDown={(e) => {
                if (e.key === 'Enter' && !e.shiftKey) {
                  e.preventDefault()
                  submit()
                }
              }}
            />
            <button
              className='rounded-md bg-primary px-3 py-2 text-sm text-primary-foreground'
              onClick={submit}
            >
              Send
            </button>
          </div>
        </div>
      )
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser→server WS | ping/pong heartbeat; server validates session ownership before any pong |
| PTY stdin | MobileBubbleChat textarea → CcPtyWsClient.sendStdin → MAX_STDIN_BYTES guard (existing) |
| tmux capture-pane | runs in already-authorized tmux session; no new privilege path |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-181-04-01 | Spoofing | visibilitychange reconnect | mitigate | Reconnect path re-authenticates via JWT in URL (same as initial connect path in ccPtyWsUrl()); no auth bypass |
| T-181-04-02 | Denial of Service | Heartbeat ping flood | mitigate | Ping only sent when WS readyState === OPEN; interval cleared on close/detach; single interval per client instance |
| T-181-04-03 | Tampering | capture-pane output injected into client | accept | capture-pane runs against an already-owned tmux session; output is the same content the user already has access to via the terminal; non-fatal on error |
| T-181-04-04 | Tampering | MobileBubbleChat textarea → PTY stdin | mitigate | Input goes through CcPtyWsClient.sendStdin which enforces MAX_STDIN_BYTES (64KB) guard; server ws-handler enforces MAX_STDIN_BYTES (1MB) as second layer |
| T-181-04-05 | Information Disclosure | ANSI strip in MobileBubbleChat | accept | Stripping escape sequences reduces terminal control char exposure in bubble UI; no sensitive data added |
| T-181-04-06 | Elevation of Privilege | capture-pane execSync | mitigate | Uses shell-escaped tmux session name (shellEscape); timeout: 3000 prevents indefinite block; same pattern as existing execSync calls in manager.ts (T-166-03 threat model applies) |
</threat_model>
