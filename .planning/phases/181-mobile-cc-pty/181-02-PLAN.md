<!-- Escape sequence reference — use EXACTLY these strings, no deviation -->
Row 1 keys:
- ESC  → '\x1b'
- TAB  → '\x09'
- CTRL → sticky modifier (see sticky-Ctrl state machine below)
- /    → '/'
- |    → '|'
- "    → '"'
- '    → "'"
- -    → '-'
- ↑    → '\x1b[A'

Row 2 keys:
- ⌘ (Cmd/Meta)  → sticky modifier (sends \x1b prefix to next key if non-modifier, e.g. ESC+key for meta combos)
- ←             → '\x1b[D'
- ↓             → '\x1b[B'
- →             → '\x1b[C'
- PGUP          → '\x1b[5~'
- PGDN          → '\x1b[6~'
- HOME          → '\x1b[H'
- END           → '\x1b[F'
- ⏎ (Enter)    → '\r'

<!-- Sticky-Ctrl state machine:
  state: 'off' | 'latched' | 'locked'
  'off'     → tap CTRL → 'latched'  (visual: dim cyan background)
  'latched' → next non-modifier key press:
                  seq = Ctrl+key (e.g. 'a' → '\x01', 'c' → '\x03')
                  → state returns to 'off'
  'latched' → tap CTRL again → 'off'  (cancel latch)
  'off'     → long-press CTRL (>600ms) → 'locked'  (visual: solid cyan background)
  'locked'  → every non-modifier key: seq = Ctrl+key, state stays 'locked'
  'locked'  → tap CTRL → 'off'

  Ctrl+letter encoding: String.fromCharCode(letter.charCodeAt(0) - 64)
    → 'a' (0x61) - 64 = 0x01 → '\x01'
    → 'c' (0x63) - 64 = 0x03 → '\x03'
  For non-letter keys that don't map to a control code, send the raw sequence as-is.
-->

    Structure of MobileTerminalKeyBar.tsx:

    ```tsx
    import { useState, useCallback, useRef } from 'react'
    import { cn } from '@/lib/utils'

    type CtrlState = 'off' | 'latched' | 'locked'

    interface Key {
      label: string
      seq: string | 'ctrl-modifier' | 'meta-modifier'
      wide?: boolean
    }

    const ROW_1: Key[] = [
      { label: 'ESC', seq: '\x1b' },
      { label: 'TAB', seq: '\x09' },
      { label: 'CTRL', seq: 'ctrl-modifier' },
      { label: '/', seq: '/' },
      { label: '|', seq: '|' },
      { label: '"', seq: '"' },
      { label: "'", seq: "'" },
      { label: '-', seq: '-' },
      { label: '↑', seq: '\x1b[A' },
    ]

    const ROW_2: Key[] = [
      { label: '⌘', seq: 'meta-modifier' },
      { label: '←', seq: '\x1b[D' },
      { label: '↓', seq: '\x1b[B' },
      { label: '→', seq: '\x1b[C' },
      { label: 'PGUP', seq: '\x1b[5~', wide: true },
      { label: 'PGDN', seq: '\x1b[6~', wide: true },
      { label: 'HOME', seq: '\x1b[H' },
      { label: 'END', seq: '\x1b[F' },
      { label: '⏎', seq: '\r' },
    ]

    const CTRL_HOLD_MS = 600

    export function MobileTerminalKeyBar({ onKey, className }: MobileTerminalKeyBarProps) {
      const [ctrlState, setCtrlState] = useState<CtrlState>('off')
      const ctrlHoldTimer = useRef<ReturnType<typeof setTimeout> | null>(null)

      const handleKey = useCallback((seq: string) => {
        if (seq === 'ctrl-modifier') return  // handled by touch handlers
        if (seq === 'meta-modifier') return  // future: meta prefix

        if (ctrlState !== 'off') {
          // Apply Ctrl encoding for single lowercase letters
          const char = seq.length === 1 && seq >= 'a' && seq <= 'z'
            ? String.fromCharCode(seq.charCodeAt(0) - 64)
            : seq
          onKey(char)
          if (ctrlState === 'latched') setCtrlState('off')
          return
        }
        onKey(seq)
      }, [ctrlState, onKey])

      const handleCtrlTouchStart = useCallback(() => {
        ctrlHoldTimer.current = setTimeout(() => {
          setCtrlState('locked')
        }, CTRL_HOLD_MS)
      }, [])

      const handleCtrlTouchEnd = useCallback(() => {
        if (ctrlHoldTimer.current) {
          clearTimeout(ctrlHoldTimer.current)
          ctrlHoldTimer.current = null
        }
        // Only toggle latch/off if timer did NOT fire (i.e. not a long-press)
        setCtrlState((prev) => {
          if (prev === 'locked') return 'off'
          if (prev === 'latched') return 'off'
          return 'latched'
        })
      }, [])

      // Render: sticky bottom bar, 2 rows, each row flex-row, touch-action: manipulation
      // CSS: position sticky bottom-0; z-index 50; bg-bg border-t border-border
      // Each key: min-w-[2.5rem] h-9 rounded-md text-xs font-mono touch-action-manipulation
      // CTRL button: aria-pressed={ctrlState !== 'off'} aria-label={`CTRL ${ctrlState}`}
      //   bg-cyan-500/20 when latched, bg-cyan-500/60 when locked
    }
    ```
