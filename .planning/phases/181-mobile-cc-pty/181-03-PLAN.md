    4. Pinch-zoom gesture handler (add inside the sessionId useEffect, before return):
    ```typescript
    let touchStartDist = 0

    function getTouchDist(touches: TouchList): number {
      if (touches.length < 2) return 0
      const dx = touches[0].clientX - touches[1].clientX
      const dy = touches[0].clientY - touches[1].clientY
      return Math.sqrt(dx * dx + dy * dy)
    }

    const onTouchStart = (e: TouchEvent) => {
      if (e.touches.length === 2) {
        touchStartDist = getTouchDist(e.touches)
      }
    }

    const onTouchMove = (e: TouchEvent) => {
      if (e.touches.length === 2 && touchStartDist > 0) {
        const dist = getTouchDist(e.touches)
        const delta = dist - touchStartDist
        const steps = Math.round(delta / 20)  // 1pt per 20px spread
        if (steps !== 0) {
          const current = (term.options as any).fontSize ?? 13
          const next = Math.max(10, Math.min(22, current + steps))
          if (next !== current) {
            ;(term.options as any).fontSize = next
            fitRef.current?.fit()
            try { localStorage.setItem('cc-pty-font-size', String(next)) } catch { /* ignore */ }
          }
          touchStartDist = dist  // reset baseline so next move is incremental
        }
      }
    }
    ```
