  <name>Task 5: Detail page</name>
  <action>
    `app/agents/[id]/page.tsx`:

    ```tsx
    'use client'
    import { use } from 'react'
    import { useAgent } from '@/src/lib/agents/use-agent'
    import { AgentEditForm } from '@/components/agents/AgentEditForm'
    import { RecentTasksList } from '@/components/agents/RecentTasksList'
    import { RunNowButton } from '@/components/agents/RunNowButton'

    export default function AgentDetailPage({ params }: { params: Promise<{ id: string }> }) {
      const { id } = use(params)
      const { agent, isLoading, refetch } = useAgent(id)

      if (isLoading) return <p className="p-8 text-muted-foreground">Loading…</p>
      if (!agent) return <p className="p-8">Agent not found.</p>

      return (
        <div className="container mx-auto max-w-4xl px-6 py-8 space-y-8">
          <header className="flex items-center justify-between">
            <div>
              <h1 className="text-2xl font-semibold">{agent.name}</h1>
              <p className="text-sm text-muted-foreground">{agent.modelName}</p>
            </div>
            <RunNowButton agentId={agent.id} />
          </header>
          <section>
            <h2 className="mb-3 text-lg font-medium">Configuration</h2>
            <AgentEditForm agent={agent} onSaved={refetch} />
          </section>
          <section>
            <h2 className="mb-3 text-lg font-medium">Recent runs</h2>
            <RecentTasksList agentId={agent.id} />
          </section>
        </div>
      )
    }
    ```
  </action>
