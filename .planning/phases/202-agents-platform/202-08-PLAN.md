    Extend `livinityd/source/modules/mastra/agents/built-in-tools.ts`:

    ```ts
    import { createTool } from '@mastra/core/tools'
    import { z } from 'zod'

    export const uiRenderTool = createTool({
      id: 'ui_render',
      description:
        'Render a custom inline UI in the chat (card, list, form, layout). ' +
        'Use when the user asks to "show", "display", "design" something visual, ' +
        'or when structured data is best presented as UI rather than markdown text. ' +
        'Accepts an OpenUI Lang JSON tree.',
      inputSchema: z.object({
        // Loose schema — actual OpenUI Lang validation happens client-side.
        // We forward the tree opaquely; the renderer validates the shape.
        tree: z.unknown(),
        title: z.string().optional(),
      }),
      outputSchema: z.object({
        rendered: z.literal(true),
        title: z.string().optional(),
      }),
      execute: async ({ context }) => {
        // No-op server-side — the UI is rendered by the client via the
        // tool-ui makeAssistantToolUI registration. We just acknowledge.
        return { rendered: true as const, title: context.title }
      },
    })
    ```

  <name>Task 4: OpenUI renderer wrapper</name>
  <action>
    `src/lib/openui/openui-renderer.tsx`:

    ```tsx
    'use client'
    import { makeAssistantToolUI } from '@assistant-ui/react'
    import { Renderer } from '@openuidev/renderer'
    import { OPENUI_COMPONENTS } from './openui-components'

    type Args = { tree: unknown; title?: string }
    type Result = { rendered: true; title?: string }

    export const UiRenderTool = makeAssistantToolUI<Args, Result>({
      toolName: 'ui_render',
      render: ({ args }) => (
        <div className="my-2 rounded-md border bg-card p-3">
          {args.title && <div className="mb-2 text-sm font-medium">{args.title}</div>}
          <Renderer tree={args.tree as never} components={OPENUI_COMPONENTS} />
        </div>
      ),
    })
    ```

    Per skill:openui, `OPENUI_COMPONENTS` is a `defineComponent` map of allowed primitives. Whitelist (T-202-06 mitigation): text, heading, paragraph, button, list, card, image, link, divider, layout-stack, layout-row, badge, input (read-only display), table. NO raw HTML, NO script, NO iframe.
  </action>
  <acceptance_criteria>
    - UiRenderTool exported
    - Whitelist covers ≥10 component types
    - No `dangerouslySetInnerHTML` in the source
  </acceptance_criteria>
</task>
