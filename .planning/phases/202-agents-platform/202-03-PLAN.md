  <action>
    Create `mastra/scheduler.ts`:

    ```ts
    import cron, { type ScheduledTask } from 'node-cron'
    import type { Redis } from 'ioredis'
    import type { AgentRegistry } from './agents/agent-registry.js'
    import type { AgentRepository } from './agents/agent-repository.js'

    interface SchedulerDeps {
      registry: AgentRegistry
      repo: AgentRepository
      memory: unknown // narrowed via cast (B-02)
      redis: Redis
      logger: { info: (m: string) => void; warn: (m: string, e?: unknown) => void }
    }

    export class AgentScheduler {
      private tasks = new Map<string, ScheduledTask>()
      constructor(private deps: SchedulerDeps) {}

      async init(): Promise<void> { await this.refresh() }

      async refresh(): Promise<void> {
        for (const [, t] of this.tasks) t.stop()
        this.tasks.clear()

        for (const row of this.deps.repo /* … listAll() */) {
          if (!row.enabled || !row.scheduleCron) continue
          if (!cron.validate(row.scheduleCron)) {
            this.deps.logger.warn(`agent ${row.name}: invalid cron "${row.scheduleCron}" — skipped`)
            continue
          }
          const lockKey = `livos:agent:${row.id}:lock`
          const ttlMs = this.lockTtlForCron(row.scheduleCron)
          const task = cron.schedule(row.scheduleCron, async () => {
            const acquired = await this.deps.redis.set(lockKey, '1', 'PX', ttlMs, 'NX')
            if (!acquired) {
              this.deps.logger.info(`agent ${row.name}: prior run still active — skipping`)
              return
            }
            try {
              await this.runOnce(row.id, 'cron')
            } finally {
              await this.deps.redis.del(lockKey).catch(() => {})
            }
          })
          this.tasks.set(row.id, task)
          this.deps.logger.info(`agent ${row.name}: scheduled "${row.scheduleCron}"`)
        }
      }

      async runOnce(agentId: string, triggeredBy: 'cron' | 'manual' | 'parent_agent', parentTaskId?: string): Promise<string> {
        const row = await this.deps.repo.getById(agentId)
        if (!row) throw new Error(`Agent ${agentId} not found`)
        const agent = this.deps.registry.get(agentId)
        if (!agent) throw new Error(`Agent ${row.name} not registered`)
        const memory = this.deps.memory as { createThread: (opts: any) => Promise<{ id: string }> }
        const threadId = `task-${row.id}-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`
        await memory.createThread({
          threadId, resourceId: 'system',
          title: `Scheduled run: ${row.name}`,
          metadata: { taskId: threadId, agentId: row.id, triggeredBy, triggeredAt: new Date().toISOString(), parentTaskId },
        })
        // Background drain — caller does not await
        ;(async () => {
          try {
            const stream = await (agent as any).stream(row.instructions, {
              memory: { thread: threadId, resource: 'system' },
            })
            await stream.text
          } catch (err) {
            this.deps.logger.warn(`agent ${row.name} run failed`, err)
          }
        })()
        return threadId
      }

      destroy(): void {
        for (const [, t] of this.tasks) t.stop()
        this.tasks.clear()
      }

      private lockTtlForCron(_cronExpr: string): number {
        // Conservative — 14 minutes is fine for any cron with cadence ≥ 15 min.
        // For sub-15-min cadences, reduce here.
        return 14 * 60 * 1000
      }
    }
    ```
