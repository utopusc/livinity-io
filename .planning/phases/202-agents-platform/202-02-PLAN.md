  <name>Task 2: AgentRegistry</name>
  <action>
    Create `agents/agent-registry.ts`:

    ```ts
    export class AgentRegistry {
      private agents = new Map<string, Agent>()
      private rowsById = new Map<string, LivosAgent>()

      constructor(private deps: {
        repo: AgentRepository
        providerRouter: ProviderRouter
        memory: unknown
        mcpBridge: McpBridge
        approvalManager: ApprovalGate
        logger: McpBridgeLogger
      }) {}

      async init(): Promise<void> { await this.refresh() }

      async refresh(): Promise<void> {
        const rows = await this.deps.repo.listAll()
        this.rowsById = new Map(rows.map((r) => [r.id, r]))

        // 1st pass: instantiate all rows without Supervisor map (children only)
        const flat = new Map<string, Agent>()
        for (const row of rows) {
          if (!row.enabled) continue
          flat.set(row.id, createAgentFromRow(row, { ...this.deps, registry: this }))
        }
        // 2nd pass: rebuild parents with agents:{} map referencing flat children
        const final = new Map<string, Agent>()
        for (const row of rows) {
          if (!row.enabled) continue
          const children = rows.filter((c) => c.parentAgentId === row.id && c.enabled)
          if (children.length > 0) {
            // T-202-04 runtime double-check: refuse if any child has its own children
            for (const c of children) {
              if (rows.some((g) => g.parentAgentId === c.id)) {
                this.deps.logger.warn(
                  `Agent depth > 2 detected for ${row.name} > ${c.name} > ...; skipping deeper levels`,
                )
              }
            }
            final.set(
              row.id,
              createAgentFromRow(row, {
                ...this.deps,
                registry: this,
                subAgents: Object.fromEntries(
                  children.map((c) => [c.name, flat.get(c.id)!]),
                ),
              }),
            )
          } else {
            final.set(row.id, flat.get(row.id)!)
          }
        }
        this.agents = final
        this.deps.logger.info(`AgentRegistry refreshed — ${final.size} live agents`)
      }

      get(id: string): Agent | undefined { return this.agents.get(id) }
      getByName(name: string): Agent | undefined {
        const row = [...this.rowsById.values()].find((r) => r.name === name)
        return row ? this.agents.get(row.id) : undefined
      }
      listAll(): { id: string; name: string; agent: Agent }[] {
        return [...this.agents.entries()].map(([id, agent]) => ({
          id, name: this.rowsById.get(id)!.name, agent,
        }))
      }
      rowsAll(): LivosAgent[] { return [...this.rowsById.values()] }
    }
    ```

    Extend `createAgentFromRow` to accept an optional `subAgents: Record<string, Agent>` and pass it to the Mastra `Agent` constructor as `agents`.
  </action>
