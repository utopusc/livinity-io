<objective>
Server5 Dashboard wizard needs **multi-key per user** API key management. Current `api_keys` table has a `UNIQUE (user_id)` constraint that limits each user to ONE active key (per `\d api_keys` 2026-05-13: `api_keys_user_id_key UNIQUE CONSTRAINT, btree (user_id)`). The current `POST /api/dashboard` route works around this by DELETE-then-INSERT (destructive regeneration). The wizard needs **purely additive** key creation so a user can:
- have a long-lived production key
- generate a fresh ephemeral key for each new install
- revoke individual keys without nuking the others

This plan ships:
1. **Drizzle migration 0010** that drops the UNIQUE constraint (drizzle-kit generate-then-apply pattern)
2. **Drizzle schema.ts additive declaration** for `api_keys` (for future drizzle-kit consistency; the table exists already via raw SQL, this just documents it)
3. **Three Next.js App Router endpoints** under `/api/account/api-keys` for create/list/revoke

Purpose: API plumbing the wizard UI (Plan 111-03) calls into. Without these endpoints the wizard cannot generate the `liv_k_*` token it bakes into the install command.
Output: 4 new/modified files on Server5 + Drizzle migration applied + local PLAN+SUMMARY.

import { NextRequest, NextResponse } from "next/server";
import { nanoid } from "nanoid";
import bcrypt from "bcryptjs";
import pool from "@/lib/db";
import { getSession, SESSION_COOKIE_NAME } from "@/lib/auth";

/**
 * Phase 111-02: multi-key-per-user CRUD endpoints for the install wizard.
 *
 * D-111-KEY-NEVER-RE-SHOWN: POST returns the plain liv_k_* token EXACTLY ONCE.
 * GET returns metadata only (prefix + timestamps). The plain key is bcrypted
 * (cost 10) before persist — the column key_hash stores only the hash.
 *
 * D-111-EXISTING-AUTH: every handler resolves the session cookie via
 * @/lib/auth.getSession — no parallel auth path.
 */

async function getUser(req: NextRequest) {
  const token = req.cookies.get(SESSION_COOKIE_NAME)?.value;
  if (!token) return null;
  return getSession(token);
}

/** POST /api/account/api-keys — generate a NEW key (additive, does not delete others). */
export async function POST(req: NextRequest) {
  const user = await getUser(req);
  if (!user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  if (!user.emailVerified) {
    return NextResponse.json(
      { error: "Please verify your email before generating an API key" },
      { status: 403 },
    );
  }

  // Generate fresh key
  const rawKey = `liv_k_${nanoid(20)}`;
  const prefix = rawKey.substring(0, 14);
  const keyHash = await bcrypt.hash(rawKey, 10);

  const result = await pool.query<{ id: string; created_at: string }>(
    `INSERT INTO api_keys (user_id, key_hash, prefix)
     VALUES ($1, $2, $3)
     RETURNING id, created_at`,
    [user.userId, keyHash, prefix],
  );

  const row = result.rows[0];

  // SECURITY: plain key returned ONLY in this response. Never logged, never re-fetchable.
  return NextResponse.json({
    id: row.id,
    apiKey: rawKey,            // shown once — Plan 111-03 bakes this into the install command
    prefix,
    created_at: row.created_at,
  });
}

/** GET /api/account/api-keys — list user keys (metadata only, NO key_hash, NO plain). */
export async function GET(req: NextRequest) {
  const user = await getUser(req);
  if (!user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });

  const result = await pool.query<{
    id: string;
    prefix: string;
    created_at: string;
    last_used_at: string | null;
  }>(
    `SELECT id, prefix, created_at, last_used_at
     FROM api_keys
     WHERE user_id = $1
     ORDER BY created_at DESC`,
    [user.userId],
  );

  return NextResponse.json({
    keys: result.rows.map((r) => ({
      id: r.id,
      prefix: r.prefix,
      created_at: r.created_at,
      last_used_at: r.last_used_at,
    })),
  });
}

 *  CROSS-USER ISOLATION: WHERE user_id = $2 prevents another user from deleting
 *  someone else'\''s key by guessing the UUID.

export async function DELETE(
  req: NextRequest,
  { params }: { params: Promise<{ id: string }> },
) {
  const user = await getUser(req);

  );

  if (result.rowCount === 0) {
