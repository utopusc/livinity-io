- **Plan 111-02** — `POST /api/account/api-keys` generates the liv_k_* baked into the command

Wizard steps:
1. **Mode selection** — 4 cards (Local, Hybrid, Own-Cloud[Coming Soon], Cloud[Coming Soon])
2. **Mode-specific form**:
   - **Local:** auto-fills `<hostname>.local`, no other inputs
   - **Hybrid:** domain + cf-token; on-blur of token → resolves zone-id via `/api/cf/resolve-zone`, displays it inline
3. **Generated install command** — single shell line with copy button; behind the scenes POST `/api/account/api-keys` to mint a fresh `liv_k_*`
4. **(Deferred to P112 — verification polling)** — Plan 111 ships steps 1-3 only. Step 3 ends with "Run the command on your VPS. We'll show your instance under My Servers once it connects." (no auto-poll; user manually returns to /dashboard to see device come online — this matches the existing dashboard's 10-second poll cycle).

D-111-INSTALL-CMD-COPY-FRIENDLY: install command is exactly ONE shell line, NO `\`-continuations. Use `bash -s -- <args>` form so the entire flag list stays on one line.
D-111-KEY-NEVER-RE-SHOWN: the plain `liv_k_*` shows ONLY in step 3 (one render); navigating back to step 2 and forward again **regenerates** a fresh key (old one revoked via DELETE then new POST). Make this transparent in the UI with a small "Regenerate Key" warning.
D-111-CF-TOKEN-NEVER-PERSISTED: cf-token lives in React state only; never sent to any endpoint except `/api/cf/resolve-zone` (which already audits non-persistence per Plan 111-03).
D-111-RELAY-DATA-PLANE-DOC: in step 3, below the install command, show a small advisory: "Your primary URL is https://<your-domain>. The legacy <username>.livinity.io URL routes through our relay — prefer your own domain to keep your data off Server5."

import { useState, useEffect } from "react";
import { useRouter } from "next/navigation";
import ModeCards, { type WizardMode } from "./components/mode-cards";
import HybridForm, { type HybridFormState } from "./components/hybrid-form";
import LocalForm, { type LocalFormState } from "./components/local-form";
import InstallCommandDisplay from "./components/install-command-display";
import WizardStepper from "./components/wizard-stepper";

/**
 * Phase 111-04 wizard page — 3 step flow (verification polling deferred).
 *
 * D-111-INSTALL-CMD-COPY-FRIENDLY: command is ONE shell line, no \\-continuations.
 * D-111-KEY-NEVER-RE-SHOWN: plain liv_k_* only stored in component state during
 *   step 3 render; navigating back+forward regenerates (old key revoked).
 * D-111-CF-TOKEN-NEVER-PERSISTED: cfToken only in component state; never sent
 *   anywhere except /api/cf/resolve-zone (which itself does not persist).
 */
export default function OnboardingInstallPage() {
  const router = useRouter();
  const [step, setStep] = useState<1 | 2 | 3>(1);
  const [mode, setMode] = useState<WizardMode>("hybrid");
  const [hybrid, setHybrid] = useState<HybridFormState>({
    domain: "", cfToken: "", cfZoneId: "", cfStatus: "idle", cfError: null,
  });
  const [local, setLocal] = useState<LocalFormState>({ hostname: "livinity" });

  // Step-3 minted key. Cleared on back-nav.
  const [genState, setGenState] = useState<{
    status: "idle" | "minting" | "ready" | "error";
    keyId?: string;
    plainKey?: string;
    error?: string;
  }>({ status: "idle" });

  // When entering step 3, mint a fresh API key.
  useEffect(() => {
    if (step !== 3) return;
    if (genState.status === "ready" || genState.status === "minting") return;
    setGenState({ status: "minting" });
    (async () => {
      try {
        const res = await fetch("/api/account/api-keys", {
          method: "POST",
          credentials: "same-origin",
          headers: { "Content-Type": "application/json" },
        });
        if (!res.ok) {
          const body = await res.json().catch(() => ({}));
          setGenState({ status: "error", error: body.error ?? `HTTP ${res.status}` });
          return;
        }
        const body = await res.json();
        setGenState({ status: "ready", keyId: body.id, plainKey: body.apiKey });
      } catch (e) {
        setGenState({ status: "error", error: String(e) });
      }
    })();
  }, [step, genState.status]);

  function handleBack() {
    if (step === 3 && genState.keyId) {
      // Revoke the minted key — security hygiene; user will get a fresh one if they re-enter step 3.
      fetch(`/api/account/api-keys/${genState.keyId}`, { method: "DELETE", credentials: "same-origin" }).catch(() => {});
    }
    setGenState({ status: "idle" });
    setStep(Math.max(1, step - 1) as 1 | 2 | 3);
  }

  function handleCancel() {
    if (step === 3 && genState.keyId) {
      fetch(`/api/account/api-keys/${genState.keyId}`, { method: "DELETE", credentials: "same-origin" }).catch(() => {});
    }
    router.push("/dashboard");
  }

  function canAdvanceFromStep2(): boolean {
    if (mode === "local-lan") return !!local.hostname.trim();
    if (mode === "hybrid") {
      return !!(hybrid.domain.trim() && hybrid.cfToken.trim() && hybrid.cfZoneId.trim() && hybrid.cfStatus === "ok");
    }
    return false;
  }

  return (
    <main className="mx-auto max-w-3xl px-6 py-12">
      <header className="mb-8">
        <h1 className="text-2xl font-bold text-zinc-900 dark:text-zinc-50">Install LivOS on Your Server</h1>
        <p className="mt-1 text-sm text-zinc-500">We will generate a ready-to-paste install command for you.</p>
      </header>

      <WizardStepper current={step} total={3} />

      <div className="mt-8 rounded-xl border border-zinc-200 bg-white p-8 shadow-sm dark:border-zinc-800 dark:bg-zinc-900">
        {step === 1 && (
          <>
            <ModeCards value={mode} onChange={setMode} />
            <div className="mt-8 flex items-center justify-between">
              <button onClick={handleCancel} className="rounded-lg border border-zinc-200 px-4 py-2 text-sm text-zinc-600 hover:bg-zinc-100 dark:border-zinc-700 dark:text-zinc-300 dark:hover:bg-zinc-800">
                Cancel
              </button>
              <button
                onClick={() => setStep(2)}
                disabled={mode !== "local-lan" && mode !== "hybrid"}
                className="rounded-lg bg-zinc-900 px-6 py-2 text-sm font-medium text-white hover:bg-zinc-800 disabled:opacity-50 dark:bg-zinc-50 dark:text-zinc-900 dark:hover:bg-zinc-200"
              >
                Next
              </button>
            </div>
          </>
        )}

        {step === 2 && (
          <>
            {mode === "hybrid" && <HybridForm state={hybrid} onChange={setHybrid} />}
            {mode === "local-lan" && <LocalForm state={local} onChange={setLocal} />}
            <div className="mt-8 flex items-center justify-between">
              <button onClick={handleBack} className="rounded-lg border border-zinc-200 px-4 py-2 text-sm text-zinc-600 hover:bg-zinc-100 dark:border-zinc-700 dark:text-zinc-300 dark:hover:bg-zinc-800">
                Back
              </button>
              <button
                onClick={() => setStep(3)}
                disabled={!canAdvanceFromStep2()}
                className="rounded-lg bg-zinc-900 px-6 py-2 text-sm font-medium text-white hover:bg-zinc-800 disabled:opacity-50 dark:bg-zinc-50 dark:text-zinc-900 dark:hover:bg-zinc-200"
              >
                Generate Install Command
              </button>
            </div>
          </>
        )}

        {step === 3 && (
          <>
            <InstallCommandDisplay
              mode={mode}
              hybrid={hybrid}
              local={local}
              gen={genState}
            />
            <div className="mt-8 flex items-center justify-between">
              <button onClick={handleBack} className="rounded-lg border border-zinc-200 px-4 py-2 text-sm text-zinc-600 hover:bg-zinc-100 dark:border-zinc-700 dark:text-zinc-300 dark:hover:bg-zinc-800">

              >
                I have copied my command — Done
              </button>
            </div>
          </>
        )}
      </div>
    </main>
  );
}

"use client";

export default function WizardStepper({ current, total }: { current: number; total: number }) {
  return (
    <div className="flex items-center gap-2">
      {Array.from({ length: total }, (_, i) => i + 1).map((n) => (
        <div key={n} className="flex flex-1 items-center gap-2">
          <div
            className={`flex h-8 w-8 items-center justify-center rounded-full text-xs font-medium ${
              n === current
                ? "bg-zinc-900 text-white dark:bg-zinc-50 dark:text-zinc-900"
                : n < current
                ? "bg-emerald-500 text-white"
                : "bg-zinc-200 text-zinc-500 dark:bg-zinc-700 dark:text-zinc-400"
            }`}
          >
            {n < current ? "✓" : n}
          </div>
          {n < total && (
            <div
              className={`h-0.5 flex-1 ${
                n < current ? "bg-emerald-500" : "bg-zinc-200 dark:bg-zinc-700"
              }`}
            />
          )}
        </div>
      ))}
    </div>
  );
