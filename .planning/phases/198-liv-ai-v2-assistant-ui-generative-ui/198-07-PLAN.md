       ```tsx
       /**
        * Phase 198-07 — Dev-only DevTools panel.
        *
        * T-198-07-01: gated behind import.meta.env.DEV so production bundle
        * tree-shakes out @assistant-ui/react-devtools entirely.
        */

       import {lazy, Suspense} from 'react'

       const LazyDevTools = lazy(async () => {
         // Dynamic import — only resolves at runtime in dev
         if (!import.meta.env.DEV) {
           // Production: return a no-op component
           return {default: () => null}
         }
         try {
           const mod = await import('@assistant-ui/react-devtools')
           // assistant-ui exports may use AssistantDevTools or default; pick what's available
           const Component = (mod as Record<string, unknown>).AssistantDevTools ?? (mod as Record<string, unknown>).default
           return {default: (Component as React.ComponentType) ?? (() => null)}
         } catch {
           return {default: () => null}
         }
       })

       export function DevToolsMount() {
         if (!import.meta.env.DEV) return null
         return (
           <Suspense fallback={null}>
             <LazyDevTools />
           </Suspense>
         )
       }
       ```
    2. **NOTE:** If `@assistant-ui/react-devtools` isn't a separate npm package (it's bundled in @assistant-ui/react or only a browser extension), this component should just return null and the DevTools experience comes from the installed browser extension. In that case, the file still ships as a stub so future revisions can mount it cleanly.
