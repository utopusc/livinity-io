**Step F — Synthetic WS probe (the live runtime gate):**

This is a Node.js one-liner that the operator runs on the Mini PC (avoids browser dependency). It mints a JWT, opens a WS, sends a `start` envelope, captures 8 seconds of messages, then closes.

```bash
sudo node -e '
const ws = require("ws");
const crypto = require("crypto");
const fs = require("fs");
const jwt = require("jsonwebtoken");  // ships with livinityd node_modules

const secret = fs.readFileSync("/opt/livos/data/secrets/jwt", "utf8").trim();
const token = jwt.sign({ loggedIn: true, userId: "admin", role: "admin" }, secret, { algorithm: "HS256" });

const sock = new ws("ws://localhost:8080/ws/agent?token=" + encodeURIComponent(token));
const start = Date.now();

sock.on("open", () => {
  console.log("WS_OPEN");
  sock.send(JSON.stringify({
    type: "start",
    conversationId: "conv_phase162smoke",
    prompt: "What vault are you running from? Reply in one sentence."
  }));
});

sock.on("message", (data) => {
  const t = ((Date.now() - start) / 1000).toFixed(2);
  const msg = JSON.parse(data.toString());
  if (msg.type === "session_ready") console.log(`[+${t}s] SESSION_READY sessionId=${msg.sessionId}`);
  else if (msg.type === "sdk_message" && msg.data?.type === "system" && msg.data?.subtype === "init")
    console.log(`[+${t}s] SDK_INIT model=${msg.data.model}`);
  else if (msg.type === "sdk_message" && msg.data?.type === "stream_delta")
    process.stdout.write(msg.data.text);
  else if (msg.type === "sdk_message" && msg.data?.type === "result")
    console.log(`\n[+${t}s] RESULT`);
});

sock.on("close", () => console.log("WS_CLOSED"));

setTimeout(() => sock.close(), 10000);
' 2>&1 | tee /tmp/phase162-probe.log
```
