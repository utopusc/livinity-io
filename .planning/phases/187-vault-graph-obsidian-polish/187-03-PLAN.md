<!-- VaultGraph handleNavigateTo implementation shape -->
```typescript
function handleNavigateTo(id: string) {
  const data = fgRef.current?.graphData()
  const node = data?.nodes.find((n: any) => n.id === id)
  if (!node) return  // graceful no-op for unknown id
  fgRef.current!.centerAt(node.x ?? 0, node.y ?? 0, 500)
  fgRef.current!.zoom(2.5, 500)
  // Also enter local mode focused on the target:
  setLocalFocusId(id)
  setGraphMode('local')
}
```
