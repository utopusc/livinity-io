```typescript
// NEW FILE: livos/packages/ui/src/features/vault-graph/graph-stats.ts

export interface HubEntry {
  id: string
  label: string
  degree: number
}

export interface GraphStats {
  nodeCount: number
  edgeCount: number
  orphanCount: number   // nodes where wikiDegree === 0
  topHubs: HubEntry[]  // top 3 nodes sorted by degree descending; may be < 3 if fewer nodes
}

/**
 * computeGraphStats — pure function, no side effects.
 * nodes must have { id, label, degree: number, wikiDegree: number }.
 * edges can be any array with a .length (only count used).
 */
export function computeGraphStats(
  nodes: Array<{ id: string; label: string; degree: number; wikiDegree: number }>,
  edges: Array<unknown>,
): GraphStats {
  const nodeCount = nodes.length
  const edgeCount = edges.length
  const orphanCount = nodes.filter((n) => n.wikiDegree === 0).length
  const topHubs = [...nodes]
    .sort((a, b) => b.degree - a.degree)
    .slice(0, 3)
    .map(({ id, label, degree }) => ({ id, label, degree }))
  return { nodeCount, edgeCount, orphanCount, topHubs }
}
```
