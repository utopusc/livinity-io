linkWidth={(link: any) => {
  // Hover override stays at top:
  if (hoveredLink && ...) return 1.4

  // Type-aware thickness:
  // link._edge is injected in graphData links mapping (see below)
  const edgeType = link._edge?.type ?? 'wikilink'
  const weight   = link._edge?.weight ?? 1
  return edgeType === 'wikilink'
    ? 1.2 + weight * 0.3   // weight=1 → 1.5px; weight=2 → 1.8px
    : 0.3                  // directory: thin scaffolding
}}

<!-- linkColor directory opacity -->
```typescript
// Current linkColor baseline (non-hover):
return getEdgeColor(theme)   // 'var(--line-strong)'

// Target (187-04) — directory edges get reduced opacity:
const edgeType = link._edge?.type ?? 'wikilink'
if (edgeType === 'directory') {
  // Append '55' (33% opacity) to the CSS var — force-graph resolves CSS vars before canvas paint.
  // Since CSS vars are used as-is by canvas, we switch to a concrete hex for directory edges.
  // Use a muted neutral matching the theme:
  return theme === 'dark'
    ? 'rgba(200,200,200,0.2)'
    : 'rgba(100,100,100,0.2)'
}
return getEdgeColor(theme)
```

NOTE on CSS var + opacity: Canvas 2D cannot directly mix a CSS var with opacity. For
directory edges only, use a concrete rgba() value instead of var(--line-strong). Wikilink
edges keep var(--line-strong) (existing behavior).
