```js
/**
 * verify-umd.cjs — assert window.LivKit shape from livkit.umd.js
 *
 * Runs the UMD bundle in a vm context with shimmed `window`, `self`,
 * `React`, and `ReactDOM` globals (matching what a browser provides).
 * After execution, asserts every expected named export is present
 * and is a function (component) or expected value.
 */
const fs = require("node:fs");
const path = require("node:path");
const vm = require("node:vm");
const React = require("react");
const ReactDOM = require("react-dom");

const umdPath = path.resolve(__dirname, "../dist/umd/livkit.umd.js");
if (!fs.existsSync(umdPath)) {
  console.error("MISSING:", umdPath);
  process.exit(2);
}
const code = fs.readFileSync(umdPath, "utf8");

const sandbox = {
  window: {},
  self: {},
  globalThis: {},
  React,
  ReactDOM,
  module: { exports: {} },
  exports: {},
  console,
};
sandbox.window.React = React;
sandbox.window.ReactDOM = ReactDOM;
sandbox.self.React = React;
sandbox.globalThis.window = sandbox.window;

vm.createContext(sandbox);
try {
  vm.runInContext(code, sandbox, { filename: "livkit.umd.js" });
} catch (err) {
  console.error("UMD execution threw:", err);
  process.exit(3);
}

// UMD writes either to window or to module.exports — pick whichever has LivKit
const LK = sandbox.window.LivKit || sandbox.self.LivKit || sandbox.module.exports;
if (!LK) {
  console.error("LivKit global not found after UMD execution");
  console.error("window keys:", Object.keys(sandbox.window));
  console.error("module.exports keys:", Object.keys(sandbox.module.exports || {}));
  process.exit(4);
}

const requiredComponents = [
  "Button", "Card", "Pill", "Input", "PasswordInput",
  "Stepper", "CommandBox", "Modal", "NavBar", "ThemeToggle",
  "ToastProvider",
];
const requiredFns = [
  "useToast", "cn", "applyLivTheme", "readLivTheme",
];
const requiredValues = [
  "LIV_THEME_STORAGE_KEY", "LIV_THEMES", "__ui_kit_version__",
];

const failures = [];
for (const name of requiredComponents) {
  if (typeof LK[name] !== "function" && typeof LK[name] !== "object") {
    failures.push(`${name}: expected component (function/object), got ${typeof LK[name]}`);
  }
}
for (const name of requiredFns) {
  if (typeof LK[name] !== "function") {
    failures.push(`${name}: expected function, got ${typeof LK[name]}`);
  }
}
for (const name of requiredValues) {
  if (LK[name] === undefined) {
    failures.push(`${name}: expected defined value, got undefined`);
  }
}

if (failures.length > 0) {
  console.error("UMD shape verification FAILED:");
  failures.forEach((f) => console.error("  -", f));
  process.exit(5);
}

console.log("UMD shape OK — verified", requiredComponents.length + requiredFns.length + requiredValues.length, "exports:");
console.log("  components:", requiredComponents.join(", "));
console.log("  functions: ", requiredFns.join(", "));
console.log("  values:    ", requiredValues.join(", "));
console.log("  version:   ", LK.__ui_kit_version__);
```
