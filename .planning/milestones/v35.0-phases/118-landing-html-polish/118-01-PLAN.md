    ```bash
    SSH_BIN='/c/Windows/System32/OpenSSH/ssh.exe'

    INJECT='<!-- p118-01: shared design tokens -->\n  <link rel="stylesheet" href="/_shared/tokens.css">\n  <link rel="stylesheet" href="/_shared/fonts.css">'

    import sys, re

    print(f'linked:{p}')
    ```
