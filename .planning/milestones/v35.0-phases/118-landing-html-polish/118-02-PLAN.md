<!-- nav.jsx component contract (the executor implements this; published here so reviewers
     and future maintainers can confirm shape without reading Server5). -->

The component must:

1. Be a single JSX file using @babel/standalone in-browser compilation (`<script type="text/babel" data-presets="react">`).
2. Expose a global `window.LivNav` constructor / render function.
3. Auto-mount itself into the first `#__liv_nav__` element on `DOMContentLoaded` — no per-page mount boilerplate.
4. Render structure (semantic):

   ```jsx
   <nav className="liv-nav" role="navigation" aria-label="Livinity main">
     <a className="liv-nav-brand" href="/">Livinity</a>
     <div className="liv-nav-spacer" />
     <div className="liv-nav-theme" role="group" aria-label="Theme">
       <button className="h-btn" data-theme="light"      aria-pressed={...}>Light</button>
       <button className="h-btn" data-theme="dark"       aria-pressed={...}>Dark</button>
       <button className="h-btn" data-theme="iridescent" aria-pressed={...}>Iridescent</button>
     </div>
     <a className="h-btn solid liv-nav-cta" href={ctaHref}>{ctaLabel}</a>
   </nav>
   ```

5. `ctaHref` / `ctaLabel` defaults:
   - if `localStorage.liv_authed === '1'` → `href="/dashboard"`, label `"Dashboard"`
   - else → `href="/login"`, label `"Sign in"`
6. Theme application:
   ```js
   function applyTheme(t) {
     document.body.classList.remove('dark','iridescent');
     if (t === 'dark') document.body.classList.add('dark');
     else if (t === 'iridescent') document.body.classList.add('iridescent');
     // 'light' = no class
     localStorage.setItem('liv_theme', t);
   }
   const initial = localStorage.getItem('liv_theme') || 'light';
   applyTheme(initial);
   ```
7. The component consumes `.h-btn` and `.h-btn.solid` classes provided by canonical tokens.css (Plan 118-01). It MUST NOT introduce inline color hex literals (cascade through `var(--accent-*)`).
8. Inline minimal layout styles (flex row, height, padding) are acceptable, but ALL colors / radii / spacings come from `var(--*)` tokens.

    **Step 1 — Author `.work/nav.jsx` locally.** Required structure (see `<interfaces>` for full contract):

    ```jsx
    /*!
     * Livinity shared landing nav — Phase 118-02
     * Mounted into the first <div id="__liv_nav__"></div> on DOMContentLoaded.
     * Reads/writes localStorage 'liv_theme' (values: 'light' | 'dark' | 'iridescent').
     * Consumes canonical .h-btn / .h-btn.solid from /_shared/tokens.css (118-01).
     * NO hex literals — colors flow through CSS variables.
     */
    (function () {
      "use strict";

      function applyTheme(t) {
        document.body.classList.remove('dark', 'iridescent');
        if (t === 'dark') document.body.classList.add('dark');
        else if (t === 'iridescent') document.body.classList.add('iridescent');
        try { localStorage.setItem('liv_theme', t); } catch (_) {}
      }

      function readTheme() {
        try { return localStorage.getItem('liv_theme') || 'light'; }
        catch (_) { return 'light'; }
      }

      function readAuthed() {
        try { return localStorage.getItem('liv_authed') === '1'; }
        catch (_) { return false; }
      }

      function LivNav() {
        const [theme, setTheme] = React.useState(readTheme());
        const authed = readAuthed();
        React.useEffect(() => { applyTheme(theme); }, [theme]);

        const themes = [
          { id: 'light',      label: 'Light' },
          { id: 'dark',       label: 'Dark' },
          { id: 'iridescent', label: 'Iridescent' },
        ];

        const ctaHref  = authed ? '/dashboard' : '/login';
        const ctaLabel = authed ? 'Dashboard' : 'Sign in';

        return (
          <nav className="liv-nav" role="navigation" aria-label="Livinity main"
               style={{
                 display:'flex', alignItems:'center', gap:'12px',
                 padding:'14px var(--dash-pad, 28px)',
                 borderBottom:'1px solid var(--dash-line, rgba(0,0,0,0.07))',
                 background:'var(--card-bg, #ffffff)',
                 fontFamily:'system-ui, -apple-system, "Geist", sans-serif',
               }}>
            <a className="liv-nav-brand" href="/"
               style={{ fontWeight:700, fontSize:'1.05rem', textDecoration:'none',
                        color:'inherit', letterSpacing:'-0.01em' }}>Livinity</a>
            <div className="liv-nav-spacer" style={{ flex:1 }} />
            <div className="liv-nav-theme" role="group" aria-label="Theme"
                 style={{ display:'flex', gap:'6px' }}>
              {themes.map(t => (
                <button key={t.id}
                        className="h-btn"
                        aria-pressed={theme === t.id}
                        onClick={() => setTheme(t.id)}
                        style={{
                          opacity: theme === t.id ? 1 : 0.7,
                          borderRadius:'10px',
                          padding:'6px 10px',
                          fontSize:'0.85rem',
                        }}>{t.label}</button>
              ))}
            </div>
            <a className="h-btn solid liv-nav-cta" href={ctaHref}
               style={{
                 borderRadius:'10px', padding:'8px 14px', textDecoration:'none',
                 fontSize:'0.9rem',
               }}>{ctaLabel}</a>
          </nav>
        );
      }

      function mount() {
        const el = document.getElementById('__liv_nav__');
        if (!el) return;
        if (typeof React === 'undefined' || typeof ReactDOM === 'undefined') {
          el.innerHTML = '<!-- LivNav: React UMD missing -->';
          return;
        }
        // React 18 root API preferred; fall back to legacy render
        if (ReactDOM.createRoot) {
          ReactDOM.createRoot(el).render(<LivNav />);
        } else {
          ReactDOM.render(<LivNav />, el);
        }
      }

      if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', mount);
      } else {
        mount();
      }

      window.LivNav = { mount: mount, applyTheme: applyTheme, readTheme: readTheme };
    })();
    ```
